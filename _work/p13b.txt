
// ============ PATCH 13b: list hooks (literals, indexing, len/sum, append, print, for-in) ============
(function patchListHooks() {
  B.setCtx('Interpreter');
  const I = (op, opts) => B.B('Interpreter', op, opts);
  const iSet = (n, v) => I('data_setvariableto', { fields: B.VARF(n), inputs: { VALUE: B.REP(v, [4, '0']) } });
  const iChg = (n, v) => I('data_changevariableby', { fields: B.VARF(n), inputs: { VALUE: B.REP(v, [4, '0']) } });
  const iCmp = (op, a, b) => I('operator_' + op, { inputs: { OPERAND1: B.REP(a, [4, '0']), OPERAND2: B.REP(b, [4, '0']) } });
  const AND = (a, b) => I('operator_and', { inputs: { OPERAND1: B.BOOL(a), OPERAND2: B.BOOL(b) } });
  const OR = (a, b) => I('operator_or', { inputs: { OPERAND1: B.BOOL(a), OPERAND2: B.BOOL(b) } });
  const NOT = (a) => I('operator_not', { inputs: { OPERAND: B.BOOL(a) } });
  const JOIN = (a, b) => I('operator_join', { inputs: { STRING1: B.REP(a, [10, '']), STRING2: B.REP(b, [10, '']) } });
  const LEN = (s) => I('operator_length', { inputs: { STRING: B.REP(s, [10, '']) } });
  const ADD = (a, b) => I('operator_add', { inputs: { NUM1: B.REP(a, [4, '0']), NUM2: B.REP(b, [4, '0']) } });
  const SUB = (a, b) => I('operator_subtract', { inputs: { NUM1: B.REP(a, [4, '0']), NUM2: B.REP(b, [4, '0']) } });
  const LET = (i, s) => I('operator_letter_of', { inputs: { LETTER: B.REP(i, [4, '0']), STRING: B.REP(s, [10, '']) } });
  const SEPV = () => B.VAR('lsep');
  const iIf = (c, ids) => { const id = I('control_if', { inputs: { CONDITION: B.BOOL(c) } }); B.sub('Interpreter', id, 'SUBSTACK', B.chain('Interpreter', ids)); return id; };
  const iIfElse = (c, a, b) => { const id = I('control_if_else', { inputs: { CONDITION: B.BOOL(c) } }); B.sub('Interpreter', id, 'SUBSTACK', B.chain('Interpreter', a)); B.sub('Interpreter', id, 'SUBSTACK2', B.chain('Interpreter', b)); return id; };
  function callExisting(proccode, args) {
    const t = B.T['Interpreter'];
    const defId = findDef('Interpreter', proccode);
    const proto = t.blocks[t.blocks[defId].inputs.custom_block[1]];
    const argIds = JSON.parse(proto.mutation.argumentids);
    const inputs = {};
    for (let i = 0; i < argIds.length; i++) inputs[argIds[i]] = B.REP(args[i], [4, '0']);
    return I('procedures_call', { inputs, mutation: { tagName: 'mutation', children: [], proccode: proccode, argumentids: proto.mutation.argumentids, warp: 'true' } });
  }
  const fail = (m) => callExisting('fail %s', [B.txt(m)]);
  const interp = B.T['Interpreter'];
  function findInProc(targetName, defId, pred) {
    const t = B.T[targetName];
    const seen = new Set();
    const stack = [t.blocks[defId].next];
    while (stack.length) {
      const id = stack.pop();
      if (!id || seen.has(id)) continue;
      seen.add(id);
      const b = t.blocks[id];
      if (!b) continue;
      if (pred(b, id)) return id;
      if (b.next) stack.push(b.next);
      for (const inp of Object.values(b.inputs || {})) {
        if (Array.isArray(inp)) for (let i = 1; i < inp.length; i++) if (typeof inp[i] === 'string') stack.push(inp[i]);
      }
    }
    return null;
  }
  const argLo = () => B.argReporter('Interpreter', 'lo');
  const argHi = () => B.argReporter('Interpreter', 'hi');
  function insertBeforeBlock(targetId, newId) {
    const t = interp;
    if (!t.blocks[targetId]) return false;
    const blk = t.blocks[targetId];
    const pid = blk.parent;
    const p = pid ? t.blocks[pid] : null;
    if (p && p.next === targetId) { p.next = newId; t.blocks[newId].parent = pid; t.blocks[newId].next = targetId; blk.parent = newId; return true; }
    if (p) {
      for (const [slot, inp] of Object.entries(p.inputs || {})) {
        if (Array.isArray(inp) && inp[1] === targetId) { p.inputs[slot] = [2, newId]; t.blocks[newId].parent = pid; t.blocks[newId].next = targetId; blk.parent = newId; return true; }
      }
    }
    return false;
  }

  // ---- 1) eval: [ ... ] literal ----
  {
    const evalDef = findDef('Interpreter', 'eval %s %s');
    // the branch that dispatches function calls: if_else( Tok[lo+1] = "(" and Tok[hi] = ")" )
    // the call branch is the PARENT of the index-dispatch if_else added earlier (it tests Tok[lo+1] = "[")
    const idxDispatch = findInProc('Interpreter', evalDef, (b) => {
      if (b.opcode !== 'control_if_else' || !b.inputs.CONDITION || !b.inputs.CONDITION[1]) return false;
      let hasBracket = false;
      const walk = (id, d) => { if (!id || d > 5 || hasBracket) return; const bb = interp.blocks[id]; if (!bb) return; for (const inp of Object.values(bb.inputs || {})) { if (!Array.isArray(inp)) continue; if (Array.isArray(inp[1]) && inp[1][0] === 10 && inp[1][1] === '[') hasBracket = true; if (typeof inp[1] === 'string') walk(inp[1], d + 1); } };
      walk(b.inputs.CONDITION[1], 0);
      return hasBracket;
    });
    if (!idxDispatch) throw new Error('index dispatch branch not found');
    const callIf = interp.blocks[idxDispatch].parent;
    if (!callIf) throw new Error('eval call branch not found');
    const litIf = iIf(AND(iCmp('equals', I('data_itemoflist', { fields: B.LISTF('Tok'), inputs: { INDEX: B.REP(argLo(), [4, '0']) } }), B.txt('[')),
                          iCmp('equals', I('data_itemoflist', { fields: B.LISTF('Tok'), inputs: { INDEX: B.REP(argHi(), [4, '0']) } }), B.txt(']'))),
      [callExisting('listlit %s %s', [ADD(argLo(), B.num(1)), SUB(argHi(), B.num(1))])]);
    if (!insertBeforeBlock(callIf, litIf)) throw new Error('could not splice the list literal branch');
    console.log('[patch13b] list literal branch installed');
  }

  // ---- 2) indexget: allow lists as the base ----
  {
    const igDef = findDef('Interpreter', 'indexget %s %s');
    const typeCheck = findInProc('Interpreter', igDef, (b) => {
      if (b.opcode !== 'control_if' || !b.inputs.CONDITION || !b.inputs.CONDITION[1]) return false;
      const c = interp.blocks[b.inputs.CONDITION[1]];
      return c && c.opcode === 'operator_not';
    });
    if (!typeCheck) throw new Error('indexget type check not found');
    // relax the "must be a string" guard to "string or list"
    const oldCond = interp.blocks[typeCheck].inputs.CONDITION[1];
    const okType = OR(iCmp('equals', B.VAR('xb'), B.txt('S')), iCmp('equals', B.VAR('xb'), B.txt('L')));
    interp.blocks[typeCheck].inputs.CONDITION = [2, NOT(okType)];
    // split the single-index branch: lists take a different path
    const idxIf = findInProc('Interpreter', igDef, (b) => b.opcode === 'control_if_else' && b.inputs.CONDITION && b.inputs.CONDITION[1] &&
      (function chk(id, d) { if (!id || d > 5) return false; const bb = interp.blocks[id]; if (!bb) return false; for (const inp of Object.values(bb.inputs || {})) { if (!Array.isArray(inp)) continue; if (Array.isArray(inp[1]) && inp[1][0] === 12 && inp[1][1] === 'xcol') return true; if (typeof inp[1] === 'string' && chk(inp[1], d + 1)) return true; } return false; })(b.inputs.CONDITION[1], 0));
    if (!idxIf) throw new Error('indexget dispatch not found');
    const oldA = interp.blocks[idxIf].inputs.SUBSTACK[1];
    // list path: evaluate the index, normalise negatives, fetch the element
    const listBranch = [
      callExisting('eval %s %s', [ADD(argLo(), B.num(2)), SUB(argHi(), B.num(1))]),
      iIf(NOT(iCmp('equals', B.VAR('error'), B.txt(''))), [I('control_stop', { fields: { STOP_OPTION: ['this script', null] }, mutation: { tagName: 'mutation', children: [], hasnext: 'false' } })]),
      iIf(NOT(iCmp('equals', B.VAR('evtype'), B.txt('N'))), [fail('TypeError: list index must be an integer')]),
      iSet('lidx', B.VAR('ev')),
      callExisting('listlen %s', [B.VAR('xa')]),
      iSet('ln', B.VAR('ev')),
      iIf(iCmp('lt', B.VAR('lidx'), B.num(0)), [iChg('lidx', B.VAR('ln'))]),
      iIf(OR(iCmp('lt', B.VAR('lidx'), B.num(0)), NOT(iCmp('lt', B.VAR('lidx'), B.VAR('ln')))), [fail('IndexError: list index out of range')]),
      callExisting('listitem %s %s', [B.VAR('xa'), ADD(B.VAR('lidx'), B.num(1))])
    ];
    const newA = iIfElse(iCmp('equals', B.VAR('xb'), B.txt('L')), listBranch, [oldA]);
    interp.blocks[oldA].parent = newA;
    interp.blocks[idxIf].inputs.SUBSTACK = [2, newA];
    interp.blocks[newA].parent = idxIf;
    console.log('[patch13b] list indexing installed');
  }

  // ---- 3) callfn: len / sum for lists, str() renders a list ----
  {
    const cfDef = findDef('Interpreter', 'callfn %s %s %s');
    const lenIf = findInProc('Interpreter', cfDef, (b) => b.opcode === 'control_if' && b.inputs.CONDITION && b.inputs.CONDITION[1] &&
      (function chk(id, d) { if (!id || d > 4) return false; const bb = interp.blocks[id]; if (!bb) return false; for (const inp of Object.values(bb.inputs || {})) { if (!Array.isArray(inp)) continue; if (Array.isArray(inp[1]) && inp[1][0] === 10 && inp[1][1] === 'len') return true; if (typeof inp[1] === 'string' && chk(inp[1], d + 1)) return true; } return false; })(b.inputs.CONDITION[1], 0));
    if (!lenIf) throw new Error('callfn len branch not found');
    const lenBranch = [
      iIfElse(iCmp('equals', B.VAR('evtype'), B.txt('L')), [
        callExisting('listlen %s', [B.VAR('ev')]),
        iSet('evtype', B.txt('N'))
      ], [
        iIf(NOT(iCmp('equals', B.VAR('evtype'), B.txt('S'))), [fail('TypeError: len requires string or list')]),
        iSet('ev', LEN(B.VAR('ev'))),
        iSet('evtype', B.txt('N'))
      ]),
      iSet('pending', B.num(1))
    ];
    const oldLenSub = interp.blocks[lenIf].inputs.SUBSTACK[1];
    const newLen = I('control_if_else', { inputs: { CONDITION: B.BOOL(iCmp('equals', B.VAR('evtype'), B.txt('L'))) } });
    const seq = B.chain('Interpreter', lenBranch);
    interp.blocks[lenIf].inputs.SUBSTACK = [2, seq];
    interp.blocks[oldLenSub].parent = null;
    // sum / str(list): insert before the len branch
    const sumIf = iIf(iCmp('equals', B.argReporter('Interpreter', 'fn'), B.txt('sum')), [
      iIf(NOT(iCmp('equals', B.VAR('evtype'), B.txt('L'))), [fail('TypeError: sum requires a list')]),
      iSet('lacc', B.num(0)),
      iSet('ln', B.num(0)),
      callExisting('listlen %s', [B.VAR('ev')]),
      iSet('lcount', B.VAR('ev')),
      iSet('li', B.num(1)),
      (function () {
        const loop = I('control_repeat_until', { inputs: { CONDITION: B.BOOL(iCmp('gt', B.VAR('li'), B.VAR('lcount'))) } });
        B.sub('Interpreter', loop, 'SUBSTACK', B.chain('Interpreter', [
          callExisting('listitem %s %s', [B.VAR('ev'), B.VAR('li')]),
          iSet('lacc', ADD(B.VAR('lacc'), B.VAR('ev'))),
          iChg('li', B.num(1))
        ]));
        return loop;
      })(),
      iSet('ev', B.VAR('lacc')),
      iSet('evtype', B.txt('N')),
      iSet('pending', B.num(1))
    ]);
    const strIf = iIf(AND(iCmp('equals', B.argReporter('Interpreter', 'fn'), B.txt('str')), iCmp('equals', B.VAR('evtype'), B.txt('L'))), [
      callExisting('listfmt %s', [B.VAR('ev')]),
      iSet('pending', B.num(1))
    ]);
    insertBeforeBlock(lenIf, sumIf);
    insertBeforeBlock(sumIf, strIf);
    console.log('[patch13b] len/sum/str for lists installed');
  }

  // ---- 4) print: format list arguments ----
  {
    const logDef = findDef('Interpreter', 'log %s');
    const paDef = findDef('Interpreter', 'printargs');
    const concat = findInProc('Interpreter', paDef, (b) => {
      if (b.opcode !== 'data_setvariableto' || !b.fields.VARIABLE || b.fields.VARIABLE[0] !== 'printbuf') return false;
      return true;
    });
    if (!concat) throw new Error('printargs concat not found');
    // before appending ev to printbuf, turn a list into [a, b, c]
    const fmtIf = iIf(iCmp('equals', B.VAR('evtype'), B.txt('L')), [callExisting('listfmt %s', [B.VAR('ev')])]);
    insertBeforeBlock(concat, fmtIf);
    console.log('[patch13b] print(list) installed');
  }

  // ---- 5) for x in <list> ----
  {
    const flDef = findDef('Interpreter', 'forloop');
    const strIf = findInProc('Interpreter', flDef, (b) => b.opcode === 'control_if_else' && b.inputs.SUBSTACK2 && b.inputs.SUBSTACK2[1] &&
      (function chk(id, d) { if (!id || d > 5) return false; const bb = interp.blocks[id]; if (!bb) return false; for (const inp of Object.values(bb.inputs || {})) { if (!Array.isArray(inp)) continue; if (Array.isArray(inp[1]) && inp[1][0] === 10 && inp[1][1] === 'range') return true; if (typeof inp[1] === 'string' && chk(inp[1], d + 1)) return true; } return false; })(b.inputs.CONDITION[1], 0));
    if (!strIf) throw new Error('forloop else branch not found');
    const tokAt2 = I('data_itemoflist', { fields: B.LISTF('Tok'), inputs: { INDEX: B.num(2) } });
    const pushText = (typ, cnt) => I('data_replaceitemoflist', { fields: B.LISTF('LoopText'), inputs: { INDEX: B.REP(I('data_lengthoflist', { fields: B.LISTF('LoopText') }), [4, '0']), ITEM: B.REP(B.VAR('ev'), [10, '']) } });
    const listCase = [
      callExisting('listlen %s', [B.VAR('ev')]),
      iSet('ln', B.VAR('ev')),
      iIfElse(iCmp('gt', B.VAR('ln'), B.num(0)), [
        callExisting('listitem %s %s', [B.VAR('ev'), B.num(1)]),
        callExisting('assign %s %s %s', [B.VAR('name'), B.VAR('ev'), B.VAR('evtype')]),
        callExisting('pushLoop %s %s %s %s %s', [B.txt('list'), B.VAR('name'), B.num(1), B.VAR('ln'), B.num(1)]),
        pushText()
      ], [
        iSet('dest', I('data_itemoflist', { fields: B.LISTF('Ends'), inputs: { INDEX: B.VAR('pc') } }))
      ])
    ];
    const strCase = [
      iIf(NOT(iCmp('equals', B.VAR('evtype'), B.txt('S'))), [fail('TypeError: for expects a string or list')]),
      iIfElse(iCmp('gt', LEN(B.VAR('ev')), B.num(0)), [
        callExisting('assign %s %s %s', [B.VAR('name'), LET(B.num(1), B.VAR('ev')), B.txt('S')]),
        callExisting('pushLoop %s %s %s %s %s', [B.txt('str'), B.VAR('name'), B.num(1), LEN(B.VAR('ev')), B.num(1)]),
        pushText()
      ], [
        iSet('dest', I('data_itemoflist', { fields: B.LISTF('Ends'), inputs: { INDEX: B.VAR('pc') } }))
      ])
    ];
    const body = [
      iSet('name', tokAt2),
      callExisting('eval %s %s', [B.num(4), SUB(B.VAR('count'), B.num(1))]),
      iIf(NOT(iCmp('equals', B.VAR('error'), B.txt(''))), [I('control_stop', { fields: { STOP_OPTION: ['this script', null] }, mutation: { tagName: 'mutation', children: [], hasnext: 'false' } })]),
      iIfElse(iCmp('equals', B.VAR('evtype'), B.txt('L')), listCase, strCase)
    ];
    const oldBody = interp.blocks[strIf].inputs.SUBSTACK2[1];
    const head = B.chain('Interpreter', body);
    interp.blocks[strIf].inputs.SUBSTACK2 = [2, head];
    interp.blocks[head].parent = strIf;
    void oldBody;
    console.log('[patch13b] for-in-string/list installed');
  }

  // ---- 6) boundary: iterate list elements ----
  {
    const bdDef = findDef('Interpreter', 'boundary');
    const kindIf = findInProc('Interpreter', bdDef, (b) => b.opcode === 'control_if_else' && b.inputs.SUBSTACK2 && b.inputs.SUBSTACK2[1] &&
      (function chk(id, d) { if (!id || d > 5) return false; const bb = interp.blocks[id]; if (!bb) return false; for (const inp of Object.values(bb.inputs || {})) { if (!Array.isArray(inp)) continue; if (Array.isArray(inp[1]) && inp[1][0] === 10 && inp[1][1] === 'while') return true; if (typeof inp[1] === 'string' && chk(inp[1], d + 1)) return true; } return false; })(b.inputs.CONDITION[1], 0));
    if (!kindIf) throw new Error('boundary kind branch not found');
    const forBranch = interp.blocks[kindIf].inputs.SUBSTACK2[1];
    const listBody = [
      iSet('lv', ADD(I('data_itemoflist', { fields: B.LISTF('LoopValue'), inputs: { INDEX: B.VAR('lc') } }), B.num(1))),
      iIfElse(iCmp('gt', B.VAR('lv'), I('data_itemoflist', { fields: B.LISTF('LoopStop'), inputs: { INDEX: B.VAR('lc') } })), [
        callExisting('popLoop', [])
      ], [
        I('data_replaceitemoflist', { fields: B.LISTF('LoopValue'), inputs: { INDEX: B.REP(B.VAR('lc'), [4, '0']), ITEM: B.REP(B.VAR('lv'), [4, '0']) } }),
        callExisting('listitem %s %s', [I('data_itemoflist', { fields: B.LISTF('LoopText'), inputs: { INDEX: B.VAR('lc') } }), B.VAR('lv')]),
        callExisting('assign %s %s %s', [I('data_itemoflist', { fields: B.LISTF('LoopName'), inputs: { INDEX: B.VAR('lc') } }), B.VAR('ev'), B.VAR('evtype')]),
        iSet('pc', ADD(I('data_itemoflist', { fields: B.LISTF('LoopHead'), inputs: { INDEX: B.VAR('lc') } }), B.num(1)))
      ])
    ];
    const newIf = iIfElse(iCmp('equals', I('data_itemoflist', { fields: B.LISTF('LoopKind'), inputs: { INDEX: B.VAR('lc') } }), B.txt('list')), listBody, [forBranch]);
    interp.blocks[forBranch].parent = newIf;
    interp.blocks[kindIf].inputs.SUBSTACK2 = [2, newIf];
    interp.blocks[newIf].parent = kindIf;
    console.log('[patch13b] list loop advancement installed');
  }
  console.log('[patch13b] hooks done');
})();
