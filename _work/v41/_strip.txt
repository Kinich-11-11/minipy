### strip spaces
print(repr('  a b  '.strip()))
### strip tab newline
print(repr('\t\na\n\t'.strip()))
### strip carriage return
print(repr('\r\nx\r\n'.strip()))
### lstrip tab
print(repr('\ta'.lstrip()))
### rstrip tab
print(repr('a\t'.rstrip()))
### strip only whitespace
print(repr('   \t\n  '.strip()))
### strip mixed
print(repr(' \t a b \n '.strip()))
### strip charset still works
print(repr('xxaxx'.strip('x')))
### strip charset space
print(repr('  a  '.strip(' ')))
### strip charset multiple chars
print(repr('abc'.strip('cba')))
### lstrip charset
print(repr('xxhixx'.lstrip('x')))
### rstrip charset
print(repr('xxhixx'.rstrip('x')))
### strip on empty
print(repr(''.strip()))
### strip both ends different
print(repr('\n\n a \t\t'.strip()))
### splitlines with crlf
print('a\r\nb'.splitlines())
### splitlines trailing newline
print('a\nb\n'.splitlines())
### split whitespace unaffected
print('a b  c'.split())
### strip in fstring
print(f'[{ " x ".strip() }]')
### strip result used
s = '  hi  '
print(s.strip() + '!')
### strip with tab only
print(repr('\t\t'.strip()))
