### two attr method calls one print
class N:
    def __init__(self, v, a, b):
        self.v = v
        self.a = a
        self.b = b
    def vals(self):
        return [self.v]
l0 = N(0, None, None)
r2 = N(2, None, None)
l1 = N(1, l0, r2)
top = N(5, l1, None)
print(top.a.vals(), top.b.vals())
### two attr method calls two prints
print(top.a.vals())
print(top.b.vals())
### var then attr
t = top.a.vals()
u = top.b.vals()
print(t, u)
