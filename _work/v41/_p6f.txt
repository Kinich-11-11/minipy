### decorator manual wrap
def shout(fn):
    def wrapped(x):
        return '!!' + str(fn(x)) + '!!'
    return wrapped
def double(x):
    return x * 2
loud = shout(double)
print(loud(21))
def compose(f, g):
    return lambda x: f(g(x))
inc = lambda x: x + 1
print(compose(double, inc)(5), compose(inc, double)(5))
