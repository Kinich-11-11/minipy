### d-echo passthrough
def echo(f):
    return f
@echo
def g(n):
    return n + 1
print(g(5))
### d-echo print mid
def echo(f):
    print("in echo", f)
    return f
@echo
def g(n):
    return n + 1
print(g(5))
### d-factory stages
def double_result(f):
    print("dr f=", f)
    def wrapper(n):
        return f(n) * 2
    print("made wrapper", wrapper)
    return wrapper
@double_result
def val(n):
    return n + 1
print(val(5))
