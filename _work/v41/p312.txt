// ---- P312: 内建名当值（Tier-2 #8 的最后一块，第三版） ----
//
// 实测（探针 _lm 的 l3/l6）：
//     def apply(f, v): return f(v)
//     print(apply(str, 4))      # CPython 4 ｜ MiniPy: NameError: name 'str' is not defined
//     h = abs ; print(h(-3))    # CPython 3 ｜ MiniPy: NameError: name 'abs' is not defined
//
// 第 38/40 轮的两条教训：
//   * 把内建名"注册成变量"（`assign(名, 名, "F")`）会**破坏用户函数调用**
//     （func 31→28、_oo 22→18，全是 "too many arguments"）——所以不能走变量表；
//   * 失败发生在**求值那个裸名字**时（PZ72 探针都没触发），消息由 P141 拼出
//     （p141:235 `failMsg("NameError: name '" & Tok[lo] & "' is not defined")`）。
//
// 本版（不碰变量表）：
//   1. patch 时扫 `callfn`/`builtin2`/`builtin2b` 的条件链，收集 `fn == "<名>"` 成新列表 `BifNames`；
//   2. 把 P141 那个"裸名字未定义"的 fail 包一层：若名字在 `BifNames` 里，
//      就当**函数值**返回（`ev = 名字`、`evK=""`、`evtype="F"`、`pending=1`，结束本次 eval），
//      否则照旧抛 NameError；
//   3. `calluser` 里"函数名找不到"的 fail 同样包一层：若 `fn` 在 `BifNames` 里，
//      转交内建分派 `callfn(fn, lo, hi)`（`calluser` 的 lo/hi 本身就是实参区间，与 callfn 约定一致）。
patch('P312 builtin names as values', () => {
  const litOf = (d) => {
    if (!Array.isArray(d)) return null;
    if ((d[0] === 10 || d[0] === 4) && typeof d[1] !== 'object') return String(d[1]);
    if ((d[0] === 1 || d[0] === 2 || d[0] === 3) && Array.isArray(d[1])) return litOf(d[1]);
    return null;
  };
  const isFnSide = (v) => {
    const id = Array.isArray(v) && typeof v[1] === 'string' ? v[1] : null;
    const q = id ? BL()[id] : null;
    return !!q && q.opcode === 'argument_reporter_string_number' && q.fields.VALUE[0] === 'fn';
  };
  const stopThis = () => I('control_stop', { fields: { STOP_OPTION: ['this script', null] }, mutation: { tagName: 'mutation', children: [], hasnext: 'false' } });
  const inBif = (v) => I('data_listcontainsitem', { fields: B.LISTF('BifNames'), inputs: { ITEM: B.REP(v, [10, '']) } });

  // (1) 收集内建名
  const names = new Set();
  for (const pc of ['callfn %s %s %s', 'builtin2 %s %s %s %s', 'builtin2b %s %s %s %s %s']) {
    let def = null;
    try { def = defIdOf(pc); } catch (e) { continue; }
    for (const id of reach(def)) {
      const b = BL()[id];
      if (b.opcode !== 'operator_equals' || !b.inputs) continue;
      const a = b.inputs.OPERAND1, c = b.inputs.OPERAND2;
      const la = litOf(a), lc = litOf(c);
      if (isFnSide(a) && lc && /^[A-Za-z_][A-Za-z_0-9]*$/.test(lc)) names.add(lc);
      else if (isFnSide(c) && la && /^[A-Za-z_][A-Za-z_0-9]*$/.test(la)) names.add(la);
    }
  }
  const list = [...names].sort();
  if (!list.length) throw new Error('[P312] no builtin names collected');
  console.log('[P312] builtin names = ' + list.length);
  const have = new Set(Object.values(T(IT).lists).map((x) => x[0]));
  if (have.has('BifNames')) throw new Error('[P312] BifNames taken');
  B.addList(IT, 'BifNames', list);

  // 通用的"包一层"工具
  const wrap = (site, cond, thenBranch) => {
    const nx = BL()[site].next;
    BL()[site].next = null;
    const w = IFELSE(cond, thenBranch, [site]);
    insertBeforeStmt(site, [w]);
    BL()[w].next = nx;
    if (nx) BL()[nx].parent = w;
  };
  const msgHas = (id, needle) => {
    let hit = false;
    const walk = (x, d) => {
      if (!x || d > 14 || typeof x !== 'string' || hit) return;
      const q = BL()[x];
      if (!q) return;
      if (JSON.stringify(q).indexOf(needle) >= 0) { hit = true; return; }
      for (const v of Object.values(q.inputs || {})) if (Array.isArray(v) && typeof v[1] === 'string') walk(v[1], d + 1);
      walk(q.next, d + 1);
    };
    for (const v of Object.values(BL()[id].inputs || {})) if (Array.isArray(v) && typeof v[1] === 'string') walk(v[1], 0);
    return hit;
  };
  const findFail = (scope, needles) => {
    for (const id of scope) {
      const b = BL()[id];
      if (b.opcode !== 'procedures_call' || !b.mutation || b.mutation.proccode !== 'fail %s') continue;
      if (!needles.every((n) => msgHas(id, n))) continue;
      return id;
    }
    return null;
  };

  // (2) 裸名字未定义 -> 内建名当函数值
  const ev = reach(defIdOf('eval %s %s'));
  const bare = findFail(ev, ["NameError: name '", 'is not defined']);
  if (!bare) throw new Error('[P312] bare-name NameError site not found');
  console.log('[P312] bare-name site=' + bare);
  wrap(bare, inBif(item('Tok', argOf('lo'))), [
    setV('ev', item('Tok', argOf('lo'))),
    setV('evK', txt('')),
    setV('evtype', txt('F')),
    setV('pending', num(1)),
    stopThis()
  ]);

  // (3) calluser：函数名找不到 -> 转交内建分派
  const cu = reach(defIdOf('calluser %s %s %s'));
  const cuFail = findFail(cu, ['is not defined']);
  if (!cuFail) throw new Error('[P312] calluser NameError site not found');
  console.log('[P312] calluser site=' + cuFail);
  wrap(cuFail, inBif(argOf('fn')), [
    callProc('callfn %s %s %s', [argOf('fn'), argOf('lo'), argOf('hi')]),
    stopThis()
  ]);
  console.log('[P312] builtin names as values installed');
});
