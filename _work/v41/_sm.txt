### m1 split
print('a,b,c'.split(','), 'a b  c'.split())
### m2 split maxsplit
print('a-b-c'.split('-', 1))
### m3 rsplit
print('a-b-c'.rsplit('-', 1))
### m4 strip variants
print('  x  '.strip(), 'xxaxx'.strip('x'), 'ab'.lstrip('a'))
### m5 case
print('Ab'.lower(), 'aB'.upper(), 'a b'.title(), 'x'.capitalize())
### m6 find index
print('abc'.find('b'), 'abc'.index('c'), 'abc'.rfind('a'))
### m7 replace
print('abc'.replace('a', 'z'), 'aaa'.replace('a', 'b', 2))
### m8 join
print('-'.join(['a', 'b']), ''.join(['x', 'y']))
### m9 startswith tuple
print('abc'.startswith(('a', 'z')), 'abc'.endswith('c'))
### m10 zfill center ljust
print('42'.zfill(4), 'x'.center(3), 'x'.ljust(2, '.'), 'x'.rjust(2, '.'))
### m11 slicing
print('abcdef'[1:3], 'abcdef'[::2], 'abcdef'[::-1], 'abcdef'[-2:])
### m12 percent format
print('%s-%d' % ('a', 2))
### m13 list sort inplace
x = [3, 1, 2]
x.sort()
print(x)
### m14 list sort reverse
x = [3, 1, 2]
x.sort(reverse=True)
print(x)
### m15 list append returns None
x = []
print(x.append(1), x)
### m16 list extend insert pop
x = [1, 2]
x.extend([3])
x.insert(0, 0)
print(x, x.pop(), x)
### m17 list index count remove
x = [1, 2, 2]
print(x.index(2), x.count(2))
x.remove(2)
print(x)
### m18 dict get setdefault
d = {'a': 1}
print(d.get('b', 2), d.setdefault('b', 3), d)
### m19 dict pop keys values
d = {'a': 1, 'b': 2}
print(d.pop('a'), list(d.keys()), list(d.values()))
### m20a dict items sorted
d = {'b': 1, 'a': 2}
print(sorted(d.items()))
### m20b dict items sorted key
d = {'b': 1, 'a': 2}
print(sorted(d.items(), key=lambda kv: -kv[1]))
### m21 dict update
d = {'a': 1}
d.update({'b': 2})
print(d)
### m22 list concat repeat
print([1, 2] + [3], [0] * 3)
### m23 tuple unpack in for
for a, b in [(1, 2), (3, 4)]:
    print(a, b)
### m24 list comprehension
print([i * i for i in range(4)])
### m25 dict comprehension
print({i: i * 2 for i in range(3)})
### m26 set comprehension
print(sorted({i % 2 for i in range(5)}))
### m27 generator to join
print(','.join(str(i) for i in range(3)))
### m28 nested comprehension
print([[i * j for j in range(2)] for i in range(2)])
### m29 conditional comprehension
print([i for i in range(5) if i % 2 == 0])
### m30 slicing assign
x = [1, 2, 3, 4]
x[1:3] = [9]
print(x)
### m31 del slice
x = [1, 2, 3, 4]
del x[1:3]
print(x)
### m32 str partition
print('a=b'.partition('='), 'a=b'.split('='))
