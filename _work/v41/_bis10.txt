### closure counter
def make_counter():
    n = 0
    def inc():
        nonlocal n
        n += 1
        return n
    return inc
c1 = make_counter()
c2 = make_counter()
print(c1(), c1(), c2(), c1())
### function factory compose
def multiplier(k):
    def apply(x):
        return x * k
    return apply
def compose(f, g):
    return lambda x: f(g(x))
double = multiplier(2)
triple = multiplier(3)
both = compose(double, triple)
print(double(5), triple(5), both(5), multiplier(10)(4))
### dispatch table lambdas
ops = {'add': lambda a, b: a + b, 'sub': lambda a, b: a - b, 'mul': lambda a, b: a * b}
def calc(op, a, b):
    if op in ops:
        return ops[op](a, b)
    return None
print(calc('add', 3, 4), calc('mul', 3, 4), calc('sub', 3, 4), calc('div', 3, 4))

### global keyword battery
count = 0
items = []
def record(v):
    global count
    count += 1
    items.append(v)
    return count
record('a')
record('b')
print(count, items)
### nested function nonlocal layers
def outer():
    x = 'o'
    def middle():
        nonlocal x
        def inner():
            nonlocal x
            x = 'i'
        inner()
        return x
    r = middle()
    return x, r
print(outer())
### decorator basic
def double_result(f):
    def wrapper(n):
        return f(n) * 2
    return wrapper
@double_result
def val(n):
    return n + 1
print(val(5), val(0))
