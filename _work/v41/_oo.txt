### o1 class basic
class A:
    def __init__(self, x):
        self.x = x
    def get(self):
        return self.x
print(A(3).get())
### o2 str dunder
class A:
    def __init__(self, x):
        self.x = x
    def __str__(self):
        return 'A(' + str(self.x) + ')'
print(A(1), str(A(2)))
### o3 repr dunder

### o4 eq dunder
class A:
    def __init__(self, x):
        self.x = x
    def __eq__(self, o):
        return self.x == o.x
print(A(1) == A(1), A(1) == A(2))

### o6 inheritance
class A:
    def hi(self):
        return 'a'
class B(A):
    def hi(self):
        return 'b'
print(A().hi(), B().hi())
### o7 super
class A:
    def hi(self):
        return 'a'
class B(A):
    def hi(self):
        return super().hi() + '!'
print(B().hi())

### o20 nested def call
def outer():
    def inner(x):
        return x * 2
    return inner(3)
print(outer())
### o21 unpack call
def f(a, b):
    return a - b
print(f(*[5, 2]), f(**{'a': 5, 'b': 2}))
### o22 property-ish method chain
class A:
    def __init__(self):
        self.v = 1
    def inc(self):
        self.v = self.v + 1
        return self
print(A().inc().inc().v)
