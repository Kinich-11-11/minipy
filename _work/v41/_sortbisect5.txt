### sort_key_stable
class A:
    def __init__(self,n,k):
        self.n=n
        self.k=k
a=[A('a',1),A('b',2),A('c',1),A('d',2)]
a.sort(key=lambda x:x.k,reverse=True)
print([x.n for x in a])

### sorted_key_stable
a=[('a',1),('b',2),('c',1),('d',2)]
print(sorted(a,key=lambda x:x[1],reverse=True))

### sorted_object
class A:
    def __init__(self,n,k):
        self.n=n
        self.k=k
a=[A('a',1),A('b',2)]
print([x.n for x in sorted(a,key=lambda x:x.k,reverse=True)])

### sort_no_key
a=[3,1,2,1]
a.sort(reverse=True)
print(a)
a.sort()
print(a)

### sort_builtin_key
a=['aaaa','b','cc']
a.sort(key=len)
print(a)
