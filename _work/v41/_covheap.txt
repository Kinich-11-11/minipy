### name alias append
a = [1]
b = a
b.append(2)
print(a)
### name alias setitem
a = [1]
b = a
b[0] = 9
print(a)
### name alias del
a = [1, 2]
b = a
del b[0]
print(a)
### name alias sort
a = [2, 1]
b = a
b.sort()
print(a)
### dict alias
d = {'k': 1}
e = d
e['k'] = 2
print(d)
### rebind after alias
a = [1]
b = a
a = [9]
print(a, b)
### alias clear
a = [1]
b = a
b.clear()
print(a, len(a))
### no alias for copy
a = [1]
b = a[:]
b.append(2)
print(a, b)
### alias via function return same object
items = []
def additem(v):
    items.append(v)
    return items
x = additem(1)
y = additem(2)
print(x, y)
