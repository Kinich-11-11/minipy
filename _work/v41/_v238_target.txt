### sort_key_stable
class A:
    def __init__(self,n,k):
        self.n=n
        self.k=k
a=[A('a',1),A('b',2),A('c',1),A('d',2)]
a.sort(key=lambda x:x.k,reverse=True)
print([x.n for x in a])

### sorted_key_stable
a=[('a',1),('b',2),('c',1),('d',2)]
print(sorted(a,key=lambda x:x[1],reverse=True))

### with_normal
class C:
    def __enter__(self):
        print('enter')
        return 7
    def __exit__(self,t,v,b):
        print('exit',t is None,v is None,b is None)
with C() as x:
    print(x)
print('after')

### with_suppress
class C:
    def __enter__(self):
        print('enter')
        return self
    def __exit__(self,t,v,b):
        print(t.__name__,str(v))
        return True
with C():
    raise ValueError('bad')
print('after')

### with_propagate
class C:
    def __enter__(self):
        print('enter')
    def __exit__(self,t,v,b):
        print('exit',str(v))
        return False
try:
    with C():
        raise ValueError('bad')
except ValueError:
    print('caught')

### with_nested
class C:
    def __init__(self,n):
        self.n=n
    def __enter__(self):
        print('enter',self.n)
        return self.n
    def __exit__(self,t,v,b):
        print('exit',self.n)
with C(1) as x:
    with C(2) as y:
        print(x,y)
print('after')

### with_return
class C:
    def __enter__(self):
        print('enter')
        return 5
    def __exit__(self,t,v,b):
        print('exit')
def f():
    with C() as x:
        return x
print(f())

### with_break_continue
class C:
    def __enter__(self):
        print('enter')
    def __exit__(self,t,v,b):
        print('exit')
for i in range(4):
    with C():
        if i==0:
            continue
        if i==2:
            break
        print(i)
print('after')

### with_inner_exception
class C:
    def __enter__(self):
        print('enter')
    def __exit__(self,t,v,b):
        print('exit',t is None)
with C():
    try:
        raise ValueError('bad')
    except ValueError:
        print('caught inside')
    print('body')
