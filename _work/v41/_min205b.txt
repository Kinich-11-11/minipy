### N2 add plus mul and eq
class V3:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    def __add__(self, o):
        return V3(self.x + o.x, self.y + o.y)
    def __mul__(self, k):
        return V3(self.x * k, self.y * k)
    def __eq__(self, o):
        return self.x == o.x and self.y == o.y
    def __str__(self):
        return '(' + str(self.x) + ',' + str(self.y) + ')'
print(V3(1, 2) + V3(3, 4))
print(V3(1, 2) * 3)
print(V3(1, 2) == V3(1, 2))
### N3 qsort pieces
def q2(a):
    if len(a) <= 1:
        return a
    p = a[0]
    left = []
    right = []
    for x in a[1:]:
        if x < p:
            left.append(x)
        else:
            right.append(x)
    return left + [p] + right
print(q2([8, 3, 9, 1]))
### N4 qsort with recursion in plus expr
def q3(a):
    if len(a) <= 1:
        return a
    p = a[0]
    left = []
    right = []
    for x in a[1:]:
        if x < p:
            left.append(x)
        else:
            right.append(x)
    return q3(left) + [p] + q3(right)

### N6 nested recursive flatten simple
def fl(x):
    out = []
    for item in x:
        if isinstance(item, list):
            for sub in fl(item):
                out.append(sub)
        else:
            out.append(item)
    return out
print(fl([1, [2, 3], 4]))

### N13 linked list append head
class Nd:
    def __init__(self, val):
        self.val = val
        self.next = None
class LL:
    def __init__(self):
        self.head = None
    def append(self, v):
        n = Nd(v)
        if self.head is None:
            self.head = n
            return
        cur = self.head
        while cur.next is not None:
            cur = cur.next
        cur.next = n
    def get(self):
        return self.head
ll = LL()
ll.append(1)
ll.append(2)
h = ll.get()
print(h is None, h.val, h.next.val)

### N17 compose returning lambda calling fns
def compose(f, g):
    return lambda x: f(g(x))
def dbl(x):
    return x * 2
def inc(x):
    return x + 1

print(h(5))
