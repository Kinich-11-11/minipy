### p1 len of generator
def g():
    yield 1
    yield 2
print(len(g()))
### p2 index of generator
def g():
    yield 7
    yield 8
print(list(g())[1])
### p3 list then len
def g():
    yield 1
    yield 2
    yield 3
print(len(list(g())))
### p4 next three times
def g():
    yield 'a'
    yield 'b'
    yield 'c'
it = g()
print(next(it), next(it), next(it))
