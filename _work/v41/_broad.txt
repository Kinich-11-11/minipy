### recursion factorial
def fact(n):
    if n <= 1:
        return 1
    return n * fact(n - 1)
print(fact(5))
### class basics with str
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    def __str__(self):
        return "(" + str(self.x) + "," + str(self.y) + ")"
    def dist2(self):
        return self.x * self.x + self.y * self.y
p = Point(3, 4)
print(p, p.dist2())
### class inheritance super
class Animal:
    def __init__(self, name):
        self.name = name
    def speak(self):
        return "..."
class Dog(Animal):
    def speak(self):
        return self.name + " says woof"
print(Dog("Rex").speak(), Animal("x").speak())
