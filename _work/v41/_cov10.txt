### float printing whole values
print(1.0, 2.0, -0.0, 0.0, 100.0)
### float printing after arithmetic
print(1.0 + 1.0, 2.5 * 2, 10.0 / 5, 3.0 - 1.0)
### float printing fractional
print(0.5, 0.25, 1.5, 2.75, 1.0 / 8)
### float printing trailing zeros from literal
print(2.50, 3.00, 0.10)
### round returns
print(round(2.5), round(2.5, 0), round(2.5, 1), round(2.55, 1))
### round of int
print(round(7), round(7, 2), round(-7))
### round whole float with ndigits
print(round(26.0, 2), round(26.0), round(-26.0, 2))
### float from int division
print(6 / 2, 6 / 4, 1 / 3, 10 / 4)
### int division whole results
print(6 // 2, 7 // 2, -6 // 2, 0 // 5)
### mixing float into int operations
print(1 + 1.0, 2 * 1.5, 3 - 0.5, 4 / 1)
### float comparison and equality
print(0.1 + 0.2 == 0.3, abs(0.1 + 0.2 - 0.3) < 1e-9)
### int float str format round trip
print(str(1.5), float('1.5'), int('3'), int(3.9), int(-3.9))
### scientific notation literals
print(1e3, 1E3, 1.5e-3, 2e0, 1e-10)
### large and small floats
print(1e16, 1e17, 1e-5, 123456789.123456789)
### float precision loss demonstration
print(0.1 + 0.2, 0.3, 0.1 + 0.2 == 0.3)
### sum of floats
print(sum([0.1, 0.2, 0.3]), sum([1.5, 2.5]), sum([1, 2.5]))
### format floats with e and g
print(f'{1234.5678:.2e}', f'{0.0001234:.2e}', f'{1234.5678:g}', f'{0.000012345:g}')
### format floats width and fill
print(f'{3.14159:10.3f}|', f'{3.14159:<10.3f}|', f'{3.14159:^10.3f}|', f'{3.14159:010.3f}')
### !format negative and sign
print(f'{-3.14159:10.2f}|', f'{-3.14159:+.2f}', f'{3.0:+d}', f'{3.0: d}')
### format percent and comma
print(f'{0.5:.1%}', f'{0.12345:.2%}', f'{1234567.891:,.2f}', f'{-1234567:,.0f}')
### format int bases with width and fill
print(f'{255:08x}', f'{255:#010x}', f'{5:08b}', f'{64:08o}')
### percent operator float precision
print('%.3f' % 1.0, '%.0f' % 1.5, '%8.2f' % -1.5, '%-8.2f|' % 1.5)
### percent operator with repr
print('%r' % 1.0, '%s' % 1.0, '%r' % 'a', '%s' % [1])
### int formatting with separators
print(f'{123456789:,}', f'{123456789:_}', f'{1234:,d}')
### string formatting of containers
print('{}'.format([1, 2]), '{}'.format({'a': 1}), '{}'.format((1, 2)), '{}'.format(None))
print(str({'a': [1, 2]}), str([{'a': 1}]))
### bool formatting
print(f'{True}', f'{False}', '%s' % True, str(True), int(True))
### none formatting
print(f'{None}', '%s' % None, str(None))
### nested fstring in format
w = 8
print(f'{3.14159:{w}.2f}', f'{42:{w}d}|')
### chained format calls
print('{:{}}'.format(3.14159, '.2f'))
### formatting very small and large numbers
print(f'{1e-10:.3e}', f'{1e10:.3e}', f'{10 ** 20}')
### int to str of very large
print(str(10 ** 30), 10 ** 30 % 7, len(str(10 ** 30)))
### floor division and modulo consistency
a = 17
b = 5
print(a // b, a % b, a // b * b + a % b)
a = -17
print(a // b, a % b, a // b * b + a % b)
b = -5
print(a // b, a % b, a // b * b + a % b)
### divmod consistency with big numbers
for a in [17, -17, 1000000007]:
    for b in [5, -5]:
        q, r = divmod(a, b)
        print(a, b, q, r, q * b + r)
### numeric string parsing variants
print(int('  -42  '), int('+42'), float(' 1.5 '), float('1e2'), float('.5'))
### numeric parse errors caught
for s in ['', '  ', '1.2.3', 'abc', '1,000']:
    try:
        print('int', s, int(s))
    except ValueError as e:
        print('bad int', repr(s))
### numeric conversion of bools and containers
print(int(True), float(False), bool(0.0), bool(0.1))
### string to number round trip
print(str(int('42')) == '42', float(str(1.5)) == 1.5)
