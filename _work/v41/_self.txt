### s1 method uses self
class A:
    def __init__(self, x):
        self.x = x
    def get(self):
        return self.x
print(A(5).get())
### s2 method with arg uses self
class B:
    def __init__(self, x):
        self.x = x
    def add(self, o):
        return self.x + o
print(B(5).add(3))
### s3 method two instances
class C:
    def __init__(self, x):
        self.x = x
    def plus(self, other):
        return self.x + other.x
print(C(1).plus(C(2)))
