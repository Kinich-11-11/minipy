// ---- P182: `join` accepts a tuple or a dict's keys (v4.109) ----
//
// P181 taught `join` to reject non-string elements, but its argument still had to be a *list*:
// `'|'.join(('x', 'y'))` reported `TypeError: join needs a list` although CPython joins a tuple,
// and `','.join({'a': 1, 'b': 2})` likewise although CPython joins a dict's keys.  (A string --
// `'-'.join('ab')` -- is a third case that stays a gap for now.)
//
// Both changes sit in the join branch of the string dispatch (P18):
//   * the guard `if smt1 != "L" then fail` also lets a tuple through -- P140 encodes tuples
//     exactly like lists, so `sjoin` needs no change at all (nor does P181's element check, which
//     runs after the guard and therefore covers the new shapes);
//   * a dict is turned into its keys with `dictkeys` just before that guard.
patch('P182 join tuple and dict', () => {
  const F = (v) => VAR(v);
  const mc = defIdOf('methodcall %s %s');
  // NB: find() wants the *reachable set*, not the definition id
  const mcSet = reach(mc);
  const litOf = (d) => {
    if (!d) return null;
    if (Array.isArray(d) && (d[0] === 10 || d[0] === 4) && typeof d[1] !== 'object') return String(d[1]);
    if (Array.isArray(d[1]) && (d[1][0] === 10 || d[1][0] === 4)) return String(d[1][1]);
    return null;
  };
  const vnOf = (d) => (Array.isArray(d) && Array.isArray(d[1]) && d[1][0] === 12) ? d[1][1] : null;
  // expression blocks carry no parent during the build, so walk the statement chain by hand
  const inChain = (stackId, target) => {
    let cur = stackId, n = 0;
    while (typeof cur === 'string' && cur && n++ < 5000) {
      if (cur === target) return true;
      cur = BL()[cur] && BL()[cur].next;
    }
    return false;
  };

  const joinFail = find(mcSet, (b) => b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === 'fail %s'
    && Object.values(b.inputs).some((v) => Array.isArray(v[1]) && v[1][1] === 'TypeError: join needs a list'));
  if (joinFail.length !== 1) throw new Error('[P182] join guard = ' + joinFail.length);
  // the *innermost* if holding it: exactly one of them keeps the fail in its own substack chain
  const guardIf = find(mcSet, (b) => b.opcode === 'control_if'
    && b.inputs.SUBSTACK && typeof b.inputs.SUBSTACK[1] === 'string'
    && inChain(b.inputs.SUBSTACK[1], joinFail[0]));
  if (guardIf.length !== 1) throw new Error('[P182] join guard if = ' + guardIf.length);
  const condId = BL()[guardIf[0]].inputs.CONDITION[1];

  // ---------- (a) the guard also lets a tuple through (P140 gives tuples the list encoding)
  const eqs = [...reach(condId)].filter((x) => {
    const q = BL()[x];
    if (!q || q.opcode !== 'operator_equals') return false;
    return (vnOf(q.inputs.OPERAND1) === 'smt1' && litOf(q.inputs.OPERAND2) === 'L')
      || (vnOf(q.inputs.OPERAND2) === 'smt1' && litOf(q.inputs.OPERAND1) === 'L');
  });
  if (eqs.length !== 1) throw new Error('[P182] smt1 == L test = ' + eqs.length);
  const eqId = eqs[0];
  // the guard reads `if not (smt1 == "L") then fail`, so widen the test it negates -- never add
  // another `not` around the comparison itself, that would invert the whole guard
  const orId = OR(eqId, EQ(F('smt1'), txt('T')));
  const gcond = BL()[condId];
  if (gcond.opcode === 'operator_not') {
    gcond.inputs.OPERAND = [2, orId];
  } else if (condId === eqId) {
    BL()[guardIf[0]].inputs.CONDITION = [2, NOT(orId)];
  } else {
    throw new Error('[P182] unexpected join guard ' + gcond.opcode);
  }

  // ---------- (b) a dict joins its keys
  insertBeforeStmt(guardIf[0], [
    IFELSE(EQ(F('smt1'), txt('D')), [
      callProc('dictkeys %s', [F('sma1')]), setV('sma1', F('ev')), setV('smt1', txt('L'))
    ], [])
  ]);
  console.log('[P182] join accepts a dict (keys) and a tuple');
});
