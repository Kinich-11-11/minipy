### palindrome check
def pal(s):
    t = ''
    for ch in s.lower():
        if ch.isalnum():
            t += ch
    return t == t[::-1]
print(pal('A man a plan a canal Panama'), pal('racecar'), pal('hello'), pal(''))

### manual title case
def titlecase(s):
    out = []
    for w in s.split(' '):
        if w:
            out.append(w[0].upper() + w[1:].lower())
        else:
            out.append(w)
    return ' '.join(out)
print(titlecase('the great GATSBY was HERE'))
print(titlecase('  spaced   out  '))

### caesar cipher
def caesar(s, k):
    out = ''
    for ch in s:
        if 'a' <= ch <= 'z':
            out += chr((ord(ch) - 97 + k) % 26 + 97)
        elif 'A' <= ch <= 'Z':
            out += chr((ord(ch) - 65 + k) % 26 + 65)
        else:
            out += ch
    return out
enc = caesar('Attack at Dawn!', 3)
print(enc)
print(caesar(enc, -3) == 'Attack at Dawn!', caesar(enc, 23))
### roman numerals
def to_roman(n):
    vals = [1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1]
    syms = ['M', 'CM', 'D', 'CD', 'C', 'XC', 'L', 'XL', 'X', 'IX', 'V', 'IV', 'I']
    res = ''
    i = 0
    while n > 0:
        while n >= vals[i]:
            res += syms[i]
            n -= vals[i]
        i += 1
    return res
print(to_roman(4), to_roman(9), to_roman(14), to_roman(2024), to_roman(3999))

### vowels and consonants
def counts(s):
    v = 'aeiou'
    vc = 0
    cc = 0
    for ch in s.lower():
        if 'a' <= ch <= 'z':
            if ch in v:
                vc += 1
            else:
                cc += 1
    return [vc, cc]
print(counts('Hello World'))
print(counts('PYTHON Programming'))
print(counts('xyz'), counts(''))

### compress and decompress run length
def compress(s):
    if s == '':
        return ''
    out = ''
    cur = s[0]
    n = 1
    for ch in s[1:]:
        if ch == cur:
            n += 1
        else:
            out += cur + str(n)
            cur = ch
            n = 1
    return out + cur + str(n)
print(compress('aaabbbcccd'), compress('abc'), compress(''))
def expand(s):
    out = ''
    i = 0
    while i < len(s):
        ch = s[i]
        i += 1
        num = ''
        while i < len(s) and '0' <= s[i] <= '9':
            num += s[i]
            i += 1
        out += ch * int(num)
    return out
print(expand('a3b4c1'), expand(compress('xxxyyyzz')))
### bracket matcher
def match(s):
    pairs = {')': '(', ']': '[', '}': '{'}
    st = []
    for ch in s:
        if ch in '([{':
            st.append(ch)
        elif ch in pairs:
            if not st or st.pop() != pairs[ch]:
                return False
    return len(st) == 0
print(match('{[()]}'), match('(]'), match('([)]'), match('((()))'), match(''))
