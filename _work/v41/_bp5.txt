### min max any all key
nums = [3, -1, 4, -1, 5, -9, 2, 6]
print(min(nums), max(nums), min(nums, key=abs), max(nums, key=abs))
print(any(n > 5 for n in nums), all(n > -10 for n in nums), any([]), all([]))
### decorator basic
def double_result(f):
    def wrapper(n):
        return f(n) * 2
    return wrapper
@double_result
def val(n):
    return n + 1
print(val(5), val(0))
