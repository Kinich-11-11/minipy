### c1 returned closure
def adder(n):
    def add(x):
        return x + n
    return add
add5 = adder(5)
print(add5(1))
### c2 immediate call
def make():
    def f():
        return 7
    return f
print(make()())
### c3 function as arg
def twice(f, v):
    return f(f(v))
def inc(x):
    return x + 1
print(twice(inc, 3))
### c4 two levels
def a(n):
    def b(m):
        def c(k):
            return n + m + k
        return c
    return b
print(a(1)(2)(3))
### c5 closure read then call
def counter():
    n = 10
    def inc():
        return n + 1
    return inc
print(counter()())
### c6 store in list then call
def mk(n):
    def g(x):
        return x * n
    return g
fs = [mk(2), mk(3)]
print(fs[0](5), fs[1](5))
