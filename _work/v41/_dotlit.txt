### string in string no dot
print('a' in 'a')
### string in dotted string
print('a' in '1.5')
### dotted string in string
print('1.5' in 'a')
### dotted string in dotted string
print('1.5' in '1.5')
### dotted string equality
print('1.5' == '1.5')
### dotted string equality with other
print('1.5' == 'x')
### dotted string concat
print('1.5' + 'x')
### dotted string len
print(len('1.5'))
### string in string via variable
x = '1.5'
print('a' in x)
### dotted literal in list
print('1.5' in ['1.5'])
### dotted literal list membership both
a = ['1.5']
print('1.5' in a)
### dotted string upper
print('1.5'.upper())
### dotted string index
print('1.5'[1])
### dotted string in tuple
print('1.5' in ('1.5',))
### string with dot find
print('1.5'.find('.'))
### string with dot count
print('1.5'.count('.'))
### two dotted strings in one expr
print('1.5' in '1.5', '2.5' in '2.5')
### dotted string as dict key
d = {'1.5': 'x'}
print(d['1.5'])
### dotted string key membership plain
print('a' in {'a': 'b'})
### dotted string value membership
print('a' in {'a': '1.5'})
