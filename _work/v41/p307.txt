// ---- P306: `tuplelit` 认识 lambda 的参数逗号（Tier-2 #8 的 l1） ----
//
// 第 30/31 轮定位：`(lambda a, b: a + b)` 被 `( … )` 分支当元组，`tuplelit` 按顶层逗号切成
// `lambda a` 与 `b: a + b` → 第一段没有冒号 → `SyntaxError: lambda needs a colon`。
// （裸的 `g = lambda a, b: …` 是对的；括号 + 单参也是对的 —— 只有"括号 + 多参 lambda"错。）
//
// 上一版（P305）想改"是否走 tuplelit"的判定，结果连括号单表达式都崩 → 撤链。
// 本版改成**让 `tuplelit` 自己跳过 lambda 参数表里的逗号**：
//   * 循环内遇到深度 0 的 `lambda` → 置 `tqlf = 1`；
//   * 遇到深度 0 的 `:` → 置 `tqlf = 0`（lambda 体里再出现的顶层逗号**仍然是分隔符**，
//     与 CPython 一致：`lambda a, b: a, b` 解析为元组）；
//   * 只有 `tqlf == 0` 时那个深度 0 的逗号才算元素分隔符。
patch('P307 tuplelit skips lambda params + grouping when no comma', () => {
  const taken = new Set(Object.values(T(IT).variables).map((x) => x[0]));
  if (taken.has('tqlf')) throw new Error('[P307] tqlf taken');
  if (taken.has('tqnc')) throw new Error('[P307] tqnc taken');
  newVar('tqlf', 0);
  newVar('tqnc', 0);
  const tl = reach(defIdOf('tuplelit %s %s'));
  // 定位：找到 `tqhi = tqek - 1` 那条，再沿 parent 上溯到最近的 control_if（分隔符判定）
  let tqhiSite = null;
  for (const id of tl) {
    const b = BL()[id];
    if (b.opcode === 'data_setvariableto' && b.fields.VARIABLE[0] === 'tqhi') { tqhiSite = id; break; }
  }
  if (!tqhiSite) throw new Error('[P306] tqhi assignment not found');
  let commaIf = null, cur = tqhiSite;
  for (let i = 0; i < 10 && cur; i++) {
    const q = BL()[cur];
    if (q.opcode === 'control_if' && q.inputs.CONDITION) { commaIf = cur; break; }
    cur = q.parent;
  }
  if (!commaIf) throw new Error('[P306] comma site not found');
  console.log('[P306] comma site=' + commaIf);
  insertBeforeStmt(commaIf, [
    IF(EQ(VAR('tqeop'), txt('lambda')), [setV('tqlf', num(1))]),
    IF(AND(EQ(VAR('tqlf'), num(1)), EQ(VAR('tqeop'), txt(':'))), [setV('tqlf', num(0))])
  ]);
  // 初始化：循环开始前 tqlf = 0（插在 tqek = lo 那条之后）
  let initSite = null;
  for (const id of tl) {
    const b = BL()[id];
    if (b.opcode === 'data_setvariableto' && b.fields.VARIABLE[0] === 'tqek') { initSite = id; break; }
  }
  if (!initSite) throw new Error('[P306] init site not found');
  spliceAfter(BL()[defIdOf('tuplelit %s %s')].next, chain([setV('tqlf', num(0)), setV('tqnc', num(0))]));
  // 真正收集到一个顶层逗号时计数 +1（插在分隔符判定体最前面）
  insertBeforeStmt(BL()[commaIf].inputs.SUBSTACK[1], [chgV('tqnc', num(1))]);
  // 分隔符条件再加上 tqlf == 0
  const condId = BL()[commaIf].inputs.CONDITION[1];
  BL()[commaIf].inputs.CONDITION = B.BOOL(AND(condId, EQ(VAR('tqlf'), num(0))));
  // 元素循环结束后：若一个真正的顶层逗号都没有，说明括号只是分组 →
  // 用原来的内层区间再 eval 一次，让 ev/evtype 直接等于那个唯一元素（而不是 1 元组）。
  { const bh = BL()[defIdOf('tuplelit %s %s')].next; let last = bh;    for (let i = 0; i < 200 && BL()[last].next; i++) last = BL()[last].next;    spliceAfter(last, chain([IF(AND(EQ(VAR('tqnc'), num(0)), GT(argOf('hi'), argOf('lo'))), [callProc('eval %s %s', [argOf('lo'), argOf('hi')])])])); }
  console.log('[P307] parenthesised single expression returns the value itself');
});
