// ---- P54: list aliasing  x = y = [0]; x[0] = 1; print(y) -> [1]   [gap row B5]
//
// Values are strings, so `x = y = [0]` stores two copies of the same list text and a
// later x[0]=1 only rewrites x.  Full object identity would mean a heap table keyed by
// id -- far too invasive here.  Instead mimic the observable behaviour:
//   * record `T aliases S` when a plain variable-to-variable assignment carries a list
//     (y = x) and when a chained assignment hands the same list to several names;
//   * every in-place mutation ends with assign(name, newtext, 'L'), so propagating to
//     the alias targets inside assign() covers indexset/indexdel/methodcall/slicewrite;
//   * a rebinding assignment (execute's `name = expr`, assignlist) drops the name's
//     alias entries first, so `x = [5]` leaves y pointing at the old list, as in Python.
// Out of scope (documented): aliases through function arguments, through container
// elements (b = a[0]), and augmented assignment.
patch('P54 list aliasing', () => {
  for (const v of ['ali', 'aln', 'alz', 'alT', 'alT2', 'alS', 'ald']) newVar(v, 0);
  for (const l of ['AliasN', 'AliasT']) B.addList(IT, l, []);
  const delLast = (list) => I('data_deleteoflist', { fields: B.LISTF(list), inputs: { INDEX: txt('last') } });
  const delAt = (list, idx) => I('data_deleteoflist', { fields: B.LISTF(list), inputs: { INDEX: B.REP(idx, [4, '0']) } });
  const at = (list, idx) => item(list, idx);
  const tokAt = (i) => item('Tok', i);

  // ---------- procs
  const ad = newProc('aldrop %s', ['vn']);
  attachBody(ad.defId, [
    setV('ali', num(1)),
    REPEATU(GT(VAR('ali'), llen('AliasN')), [
      IFELSE(OR(EQ(at('AliasN', VAR('ali')), argOf('vn')), EQ(at('AliasT', VAR('ali')), argOf('vn'))), [
        delAt('AliasN', VAR('ali')),
        delAt('AliasT', VAR('ali'))
      ], [
        chgV('ali', num(1))
      ])
    ])
  ]);
  const ar = newProc('alrec %s %s', ['target', 'src']);
  attachBody(ar.defId, [
    callProc('aldrop %s', [argOf('target')]),
    setV('ali', num(1)),
    REPEATU(GT(VAR('ali'), llen('AliasN')), [
      IFELSE(AND(EQ(at('AliasN', VAR('ali')), argOf('target')), EQ(at('AliasT', VAR('ali')), argOf('src'))), [
        delAt('AliasN', VAR('ali')),
        delAt('AliasT', VAR('ali'))
      ], [
        chgV('ali', num(1))
      ])
    ]),
    addTo('AliasN', argOf('src')),
    addTo('AliasT', argOf('target'))
  ]);

  // ---------- assign(): propagate a list value to every name that aliases this one
  const asg = inDef('assign %s %s %s');
  const ifElse = find(asg, (b) => b.opcode === 'control_if_else');
  if (ifElse.length !== 1) throw new Error('[P54] assign if-else = ' + ifElse.length);
  spliceAfter(ifElse[0], chain([
    IF(AND(EQ(argOf('vt'), txt('L')), LT(VAR('ald'), num(3))), [
      // ping-pong between a and b is stopped by the depth guard; two levels cover
      // chains like c -> b -> a
      chgV('ald', num(1)),
      callProc('lookup %s', [argOf('vn')]),
      IF(NOT(EQ(VAR('found'), num(0))), [
        setV('alS', item('Values', VAR('found'))),
        setV('alT2', item('Types', VAR('found'))),
        setV('ali', num(1)),
        setV('aln', llen('AliasN')),
        REPEATU(GT(VAR('ali'), VAR('aln')), [
          // both directions: y = x then mutating either name must update the other
          IF(OR(EQ(at('AliasN', VAR('ali')), argOf('vn')), EQ(at('AliasT', VAR('ali')), argOf('vn'))), [
            setV('alT', at('AliasT', VAR('ali'))),
            IF(EQ(VAR('alT'), argOf('vn')), [setV('alT', at('AliasN', VAR('ali')))]),
            // the nested assign reuses our scratch vars -- keep the loop state
            addTo('VStack', VAR('ali')),
            addTo('VStack', VAR('aln')),
            callProc('assign %s %s %s', [VAR('alT'), VAR('alS'), VAR('alT2')]),
            setV('aln', item('VStack', llen('VStack'))),
            delLast('VStack'),
            setV('ali', item('VStack', llen('VStack'))),
            delLast('VStack')
          ]),
          chgV('ali', num(1))
        ])
      ]),
      chgV('ald', num(-1))
    ])
  ]));
  console.log('[P54] assign propagates list aliases');

  // ---------- execute(): `y = x` records an alias, and a rebind drops stale ones
  const ex = inDef('execute');
  const eqTok = (orId, idx, lit) => [...reach(orId)].some((id) => {
    const b = BL()[id];
    if (!b || b.opcode !== 'operator_equals') return false;
    const itemOf = (d, want) => {
      const bb = (d && typeof d[1] === 'string') ? BL()[d[1]] : null;
      if (!bb || bb.opcode !== 'data_itemoflist' || !bb.fields.LIST || bb.fields.LIST[0] !== 'Tok') return false;
      const ix = bb.inputs.INDEX;
      return (Array.isArray(ix) && ix[0] === 4 && String(ix[1]) === String(want)) || (Array.isArray(ix[1]) && ix[1][0] === 4 && String(ix[1][1]) === String(want));
    };
    const litOf = (d) => {
      if (!d) return null;
      if (Array.isArray(d) && (d[0] === 10 || d[0] === 4) && typeof d[1] !== 'object') return String(d[1]);
      if (Array.isArray(d[1]) && (d[1][0] === 10 || d[1][0] === 4)) return String(d[1][1]);
      return null;
    };
    const a = b.inputs.OPERAND1, c = b.inputs.OPERAND2;
    return (itemOf(a, idx) && litOf(c) === lit) || (itemOf(c, idx) && litOf(a) === lit);
  });
  const eqCands = find(ex, (b) => b.opcode === 'control_if' && b.inputs.CONDITION && typeof b.inputs.CONDITION[1] === 'string' && eqTok(b.inputs.CONDITION[1], 2, '='));
  // two branches compare Tok[2] with '='; the plain assignment is the one that calls assign itself
  const inSub = (id) => (BL()[id].inputs.SUBSTACK && typeof BL()[id].inputs.SUBSTACK[1] === 'string') ? reach(BL()[id].inputs.SUBSTACK[1]) : new Set();
  const asgBranch = eqCands.filter((id) => [...inSub(id)].some((x) => {
    const b = BL()[x];
    return b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === 'assign %s %s %s';
  }));
  if (asgBranch.length !== 1) throw new Error('[P54] execute assignment branch = ' + asgBranch.length + ' (cands ' + eqCands.length + ')');
  const callIn = (hostId, proccode) => find(inSub(hostId), (b) => b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === proccode);
  const asgCalls = callIn(asgBranch[0], 'assign %s %s %s');
  if (asgCalls.length !== 1) throw new Error('[P54] assignment assign call = ' + asgCalls.length);
  insertBeforeStmt(asgCalls[0], [callProc('aldrop %s', [VAR('name')])]);
  spliceAfter(asgCalls[0], chain([
    IF(AND(EQ(VAR('evtype'), txt('L')), AND(EQ(VAR('count'), num(3)), EQ(tokAt(num(3)), tokAt(num(3))))), [
      callProc('alrec %s %s', [VAR('name'), tokAt(num(3))])
    ])
  ]));
  console.log('[P54] execute records y = x aliases');

  // ---------- assignlist(): chained x = y = <list> aliases all names to the first
  const al = inDef('assignlist %s');
  const alAssign = find(al, (b) => b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === 'assign %s %s %s');
  if (alAssign.length !== 1) throw new Error('[P54] assignlist assign call = ' + alAssign.length);
  insertBeforeStmt(alAssign[0], [callProc('aldrop %s', [at('AName', VAR('mi'))])]);
  const alLoop = find(al, (b) => b.opcode === 'control_repeat_until' && b.inputs.SUBSTACK && typeof b.inputs.SUBSTACK[1] === 'string'
    && [...reach(b.inputs.SUBSTACK[1])].some((x) => x === alAssign[0]));
  if (alLoop.length !== 1) throw new Error('[P54] assignlist loop = ' + alLoop.length);
  const loopId = alLoop[0];
  spliceAfter(loopId, chain([
    IF(AND(EQ(at('AType', num(1)), txt('L')), GT(llen('AName'), num(1))), [
      // alrec/aldrop reset `ali`, so this loop counts in `alz`
      setV('alz', num(2)),
      REPEATU(GT(VAR('alz'), llen('AName')), [
        callProc('alrec %s %s', [at('AName', VAR('alz')), at('AName', num(1))]),
        chgV('alz', num(1))
      ])
    ])
  ]));
  console.log('[P54] assignlist records chained aliases');

  // ---------- reset
  const runInit = find(allIds(), (b) => b.opcode === 'data_deletealloflist' && b.fields.LIST && b.fields.LIST[0] === 'LoopText');
  if (runInit.length !== 1) throw new Error('[P54] run init anchor = ' + runInit.length);
  insertBeforeStmt(runInit[0], [clearL('AliasN'), clearL('AliasT')]);
  console.log('[P54] aliasing ready');
});
