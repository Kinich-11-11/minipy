    g.append(2)
g = [1]
f()
print(g)
### g2 global list assign
def f():
    g[0] = 9

### o1 add overload
class V:
    def __init__(self, x): self.x = x
    def __add__(self, o): return V(self.x + o.x)
print((V(1) + V(2)).x)
### o2 eq overload
class W:
    def __init__(self, x): self.x = x
    def __eq__(self, o): return self.x == o.x
print(W(1) == W(1))
### o3 lt overload
class U:
    def __init__(self, x): self.x = x
    def __lt__(self, o): return self.x < o.x
print(sorted([U(3), U(1)])[0].x)
### o4 len getitem
class C:
    def __init__(self, n): self.n = n
    def __len__(self): return 3
    def __getitem__(self, i): return i * 2
c = C(1)
print(len(c), c[2])
