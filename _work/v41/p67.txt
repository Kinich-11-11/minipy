// ---- P67: finish the non-warp chain (the "first demo is still slow" fix) ----
//
// v4.31 flipped the call + expression chain, but demo1 is `print(area(3, 4))` and the
// argument evaluation happens inside `printargs` -- still warp, sitting right above
// `eval`, so everything below it inherited warp again and a step inside area() still
// cost 501ms.
//
// Found by walking the block graph: `_work/v41/callgraph.cjs` computes every procedure
// that can transitively reach `calluser`; only three were still warp:
//   printargs  (print),  minmax (min/max & friends),  regdef (default arguments)
// After flipping them: demo1 step 501ms -> 47ms.
//
// On top of that this patch *checks* the closure at build time: if any procedure that
// can reach calluser is still warp, the build fails -- so a future syntax path cannot
// silently bring back the half-second step.
patch('P67 print chain', () => {
  const MORE = ['printargs', 'minmax %s %s %s', 'regdef %s'];
  const protos = {};
  for (const b of Object.values(BL())) {
    if (b.opcode === 'procedures_prototype' && b.mutation) protos[b.mutation.proccode] = b;
  }
  let n = 0;
  for (const code of MORE) {
    const b = protos[code];
    if (!b) throw new Error('[P67] procedure not found: ' + code);
    if (b.mutation.warp !== 'false') n++;
    b.mutation.warp = 'false';
  }
  console.log('[P67] ' + MORE.length + ' more procedures are non-warp (' + n + ' changed)');
  // build-time closure check: everything that can reach calluser must be non-warp
  const defs = {};
  for (const [id, b] of Object.entries(BL())) {
    if (b.opcode === 'procedures_definition' && b.inputs.custom_block) {
      const pr = BL()[b.inputs.custom_block[1]];
      if (pr && pr.mutation) defs[id] = pr.mutation.proccode;
    }
  }
  const edges = {};
  for (const [defId, code] of Object.entries(defs)) {
    const calls = new Set();
    const seen = new Set();
    const stack = [BL()[defId].next];
    while (stack.length) {
      const id = stack.pop();
      if (!id || seen.has(id)) continue;
      const b = BL()[id];
      if (!b) continue;
      seen.add(id);
      if (b.opcode === 'procedures_call' && b.mutation) calls.add(b.mutation.proccode);
      for (const v of Object.values(b.inputs || {})) if (v && typeof v[1] === 'string') stack.push(v[1]);
      if (b.next) stack.push(b.next);
    }
    edges[code] = calls;
  }
  const reach = new Set(['calluser %s %s %s']);
  let grew = true;
  while (grew) {
    grew = false;
    for (const [code, calls] of Object.entries(edges)) {
      if (reach.has(code)) continue;
      for (const c of calls) if (reach.has(c)) { reach.add(code); grew = true; break; }
    }
  }
  const still = [...reach].filter((code) => protos[code] && protos[code].mutation.warp !== 'false');
  if (still.length) throw new Error('[P67] these can still reach calluser but stay warp: ' + still.join(', '));
  console.log('[P67] checked: all ' + reach.size + ' procedures on the path to calluser are non-warp');
});
