### k1 two bases methods
class A:
    def a(self):
        return 'a'
class B:
    def b(self):
        return 'b'
class C(A, B):
    pass
c = C()
print(c.a(), c.b())
### k2 isinstance second base
class A:
    pass
class B:
    pass
class C(A, B):
    pass
c = C()
print(isinstance(c, A), isinstance(c, B))
### k3 issubclass second base
class A:
    pass
class B:
    pass
class C(A, B):
    pass
print(issubclass(C, A), issubclass(C, B))
### k4 three bases
class A:
    pass
class B:
    pass
class D:
    pass
class C(A, B, D):
    pass
c = C()
print(isinstance(c, D), issubclass(C, D))
### k5 second base init
class A:
    def __init__(self):
        self.x = 1
class B:
    def __init__(self):
        self.y = 2
class C(A, B):
    def __init__(self):
        A.__init__(self)
        B.__init__(self)
c = C()
print(c.x, c.y)
### k6 second base only
class B:
    def b(self):
        return 'b'
class C(B):
    pass
print(C().b())
