### divmod negatives
print(divmod(-7, 2), divmod(7, -2))
### divmod float
print(divmod(-7.5, 2))
### round halves
print(round(-2.5), round(2.5), round(-3.5))
### round ndigits
print(round(2.675, 2), round(1.005, 2))
### int base
print(int('0x1f', 16), int('11', 2), int('ff', 16))
### int forms
print(int('-12'), int('  7  '), int(3.9))
### sum start
print(sum([1, 2, 3], 10), sum([1.5, 2.5]))
### enumerate start
print(list(enumerate('ab', 5)))
### zip unequal
print(list(zip([1, 2, 3], 'ab')))
### reversed
print(list(reversed([1, 2, 3])))
### iter next
print(next(iter([1, 2])))
### isinstance tuple
print(isinstance(1, (int, str)), isinstance('a', (int, str)))
### min max multi
print(min(3, 1, 2), max([3, 1, 2]))
### abs
print(abs(-2.5), abs(-2))
### repr
print(repr(1.0), repr('a'))
### str containers
print(str([1, 'a', (2,)]), str({'a': 1}))
### sorted reverse
print(sorted([3, 1, 2], reverse=True))
### len forms
print(len({'a': 1}), len((1, 2)), len('ab'))
### bool forms
print(bool(0), bool(''), bool([]), bool([0]))
### map filter
print(list(map(abs, [-1, 2])), list(filter(None, [0, 1, '', 'a'])))
### bases
print(hex(255), oct(8), bin(5))
### chr ord
print(chr(65), ord('A'))
### any all
print(any([0, 1]), all([1, 0]))
### conversions
print(int(True), float(False), str(None))
### floordiv mod
print(3 // 2, -3 // 2, 3 % -2, -3 % 2)
### division
print(10 / 4, 10 // 4, divmod(10, 4))
### sorted key tuple
print(sorted([(2, 'b'), (1, 'a')], key=lambda p: p[0]))
