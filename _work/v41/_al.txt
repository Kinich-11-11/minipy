### a1 return container after append
def build2(out):
    out.append(1)
    return out
x = []
r = build2(x)
print(x, r)
### a2 return new list built by appends
def build3():
    out = []
    out.append(1)
    return out
print(build3())
### a3 return param unchanged
def ident(out):
    return out
y = [5]
print(ident(y))
### a4 nested function appends then returns
def outer():
    out = []
    def inner():
        out.append(1)
    inner()
    return out
print(outer())
### a5 return dict alias
def addk(d):
    d['a'] = 1
    return d
dd = {}
r5 = addk(dd)
print(dd, r5)
### a6 build2([]) inline
def build2b(out):
    out.append(1)
    return out
