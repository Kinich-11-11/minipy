class Queue:
    def __init__(self):
        self.q = []
    def push(self, v):
        self.q.append(v)
    def pop(self):
        return self.q.pop(0)
    def __len__(self):
        return len(self.q)
qu = Queue()
for x in [10, 20, 30]:
    qu.push(x)
print(qu.pop(), qu.pop(), len(qu), qu.q)
### bst insert search inorder
class BT:
    def __init__(self, v):
        self.v = v
        self.l = None
        self.r = None
    def ins(self, v):
        if v < self.v:
            if self.l is None:
                self.l = BT(v)
            else:
                self.l.ins(v)
        else:
            if self.r is None:
                self.r = BT(v)
            else:
                self.r.ins(v)
    def has(self, v):
        if v == self.v:
            return True
        if v < self.v:
            if self.l is None:
                return False
            return self.l.has(v)
        if self.r is None:
            return False
        return self.r.has(v)
    def inorder(self):
        out = []
        if self.l is not None:
            out.extend(self.l.inorder())
        out.append(self.v)
        if self.r is not None:
            out.extend(self.r.inorder())
        return out
root = BT(5)
for x in [3, 8, 1, 4, 7, 9]:
    root.ins(x)
print(root.inorder(), root.has(4), root.has(6))
### custom str method
class Temp:
    def __init__(self, c):
        self.c = c
    def __str__(self):
        return str(self.c) + 'C'
t = Temp(23)
print(t)

### custom repr in container
class Pt:
    def __init__(self, x):
        self.x = x
    def __repr__(self):
        return 'Pt(' + str(self.x) + ')'
print([Pt(1), Pt(2)])
print(Pt(3))
print((Pt(4),))
### operator overload add eq
class V:
    def __init__(self, x):
        self.x = x
    def __add__(self, o):
        return V(self.x + o.x)
    def __eq__(self, o):
        return self.x == o.x
    def __repr__(self):
        return 'V' + str(self.x)
a = V(1)
b = V(2)
c = a + b
print(c, a == b, V(3) == c, a != b)
### builder chaining
class Str:
    def __init__(self):
        self.parts = []
    def add(self, s):
        self.parts.append(s)
        return self
    def text(self):
        return '-'.join(self.parts)
print(Str().add('a').add('b').add('c').text())
### method dispatch self
class Shape:
    def __init__(self, sides):
        self.sides = sides
    def kind(self):
        names = {3: 'triangle', 4: 'quad', 5: 'pentagon'}
        if self.sides in names:
            return names[self.sides]
        return 'polygon'
    def describe(self):
        return 'a ' + self.kind() + ' with ' + str(self.sides) + ' sides'
print(Shape(3).describe(), Shape(4).describe(), Shape(7).describe())
