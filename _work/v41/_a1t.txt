### dict literal both ways
row = [1, 2]
d = {"r": row}
row.append(3)
d["r"].append(4)
print(d, row)
### indexset both ways
row = [1]
d = {}
d["r"] = row
row.append(2)
print(d, row)
### rebind breaks link
row = [1]
d = {}
d["r"] = row
row = [9]
row.append(1)
print(d, row)
### unrelated key not synced
row = [1]
d = {}
d["x"] = row
d["y"] = 5
row.append(2)
print(d, row)
