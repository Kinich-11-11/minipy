### merge sort
def msort(a):
    if len(a) <= 1:
        return a
    mid = len(a) // 2
    l = msort(a[:mid])
    r = msort(a[mid:])
    out = []
    i = j = 0
    while i < len(l) and j < len(r):
        if l[i] <= r[j]:
            out.append(l[i])
            i += 1
        else:
            out.append(r[j])
            j += 1
    while i < len(l):
        out.append(l[i])
        i += 1
    while j < len(r):
        out.append(r[j])
        j += 1
    return out
print(msort([5, 2, 8, 1, 9, 3, 3, 7]), msort([]), msort([1]))
### heap sort
def sift(a, n, i):
    while True:
        big = i
        l = 2 * i + 1
        r = 2 * i + 2
        if l < n and a[l] > a[big]:
            big = l
        if r < n and a[r] > a[big]:
            big = r
        if big == i:
            return
        a[i], a[big] = a[big], a[i]
        i = big
def hsort(a):
    h = a[:]
    n = len(h)
    for i in range(n // 2 - 1, -1, -1):
        sift(h, n, i)
    for end in range(n - 1, 0, -1):
        h[0], h[end] = h[end], h[0]
        sift(h, end, 0)
    return h
print(hsort([9, 4, 7, 1, 8, 2, 6, 3, 5]), hsort([1]), hsort([]))
### recursive binary search count
def bsearch(a, t, lo, hi, cnt):
    if lo > hi:
        return -1, cnt
    mid = (lo + hi) // 2
    if a[mid] == t:
        return mid, cnt + 1
    if a[mid] < t:
        return bsearch(a, t, mid + 1, hi, cnt + 1)
    return bsearch(a, t, lo, mid - 1, cnt + 1)
data = [1, 3, 5, 7, 9, 11, 13]
print(bsearch(data, 7, 0, len(data) - 1, 0))
print(bsearch(data, 4, 0, len(data) - 1, 0))
print(bsearch(data, 1, 0, len(data) - 1, 0))
### four queens
def nqueens(n):
    cols = []
    sols = []
    def place(r):
        if r == n:
            sols.append(cols[:])
            return
        for c in range(n):
            ok = True
            for pr in range(r):
                if cols[pr] == c or abs(cols[pr] - c) == r - pr:
                    ok = False
            if ok:
                cols.append(c)
                place(r + 1)
                cols.pop()
    place(0)
    return sols
sols = nqueens(4)
print(len(sols))
print(sols[0])
print(sols[1])
### matrix multiply and determinant
def matmul(a, b):
    n = len(a)
    m = len(b[0])
    k = len(b)
    out = [[0] * m for _ in range(n)]
    for i in range(n):
        for j in range(m):
            s = 0
            for t in range(k):
                s += a[i][t] * b[t][j]
            out[i][j] = s
    return out
def det3(m):
    return (m[0][0] * (m[1][1] * m[2][2] - m[1][2] * m[2][1])
            - m[0][1] * (m[1][0] * m[2][2] - m[1][2] * m[2][0])
            + m[0][2] * (m[1][0] * m[2][1] - m[1][1] * m[2][0]))
a = [[1, 2, 3], [4, 5, 6]]
b = [[7, 8], [9, 10], [11, 12]]
print(matmul(a, b))

### bfs shortest path
graph = {1: [2, 3], 2: [4], 3: [4, 5], 4: [6], 5: [6], 6: []}
def bfs(g, s, e):
    q = [[s]]
    seen = [s]
    while q:
        path = q.pop(0)
        node = path[-1]
        if node == e:
            return path
        for nxt in g[node]:
            if nxt not in seen:
                seen.append(nxt)
                q.append(path + [nxt])
    return None
print(bfs(graph, 1, 6))
print(bfs(graph, 2, 5))

### sieve primes
def sieve(n):
    flags = [True] * (n + 1)
    flags[0] = False
    if n >= 1:
        flags[1] = False
    p = 2
    while p * p <= n:
        if flags[p]:
            m = p * p
            while m <= n:
                flags[m] = False
                m += p
        p += 1
    return [i for i in range(n + 1) if flags[i]]
print(sieve(50))
### josephus order
def josephus(n, k):
    people = list(range(1, n + 1))
    idx = 0
    order = []
    while people:
        idx = (idx + k - 1) % len(people)
        order.append(people.pop(idx))
    return order
print(josephus(7, 3))
print(josephus(5, 2)[-1])
### gray code reflect
def gray(n):
    if n == 0:
        return ['']
    prev = gray(n - 1)
    return ['0' + s for s in prev] + ['1' + s for s in prev[::-1]]
print(gray(3))
### balanced brackets
def bal(s):
    pairs = {')': '(', ']': '[', '}': '{'}
    st = []
    for ch in s:
        if ch in '([{':
            st.append(ch)
        elif ch in ')]}':
            if not st or st[-1] != pairs[ch]:
                return False
            st.pop()
    return len(st) == 0
print(bal('({[]})'), bal('([)]'), bal('((()))'), bal(''), bal('('))
### longest common subsequence
def lcs(a, b):
    m = len(a)
    n = len(b)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if a[i - 1] == b[j - 1]:
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    return dp[m][n]
print(lcs('ABCBDAB', 'BDCABA'), lcs('', 'abc'), lcs('xyz', 'xyz'))
### knapsack dp
def knap(ws, vs, cap):
    n = len(ws)
    dp = [[0] * (cap + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for c in range(cap + 1):
            dp[i][c] = dp[i - 1][c]
            if ws[i - 1] <= c:
                take = dp[i - 1][c - ws[i - 1]] + vs[i - 1]
                if take > dp[i][c]:
                    dp[i][c] = take
    return dp[n][cap]
print(knap([2, 3, 4, 5], [3, 4, 5, 6], 8))
print(knap([1, 1, 1], [10, 20, 30], 2))
### power set bitmask
def pset(items):
    n = len(items)
    out = []
    for mask in range(2 ** n):
        cur = []
        for i in range(n):
            if mask % (2 ** (i + 1)) >= 2 ** i:
                cur.append(items[i])
        out.append(cur)
    return out
print(pset(['x', 'y', 'z']))
### permutations recursive
def perms(xs):
    if len(xs) <= 1:
        return [xs]
    out = []
    for i in range(len(xs)):
        rest = xs[:i] + xs[i + 1:]
        for p in perms(rest):
            out.append([xs[i]] + p)
    return out
print(sorted([''.join(p) for p in perms(['a', 'b', 'c'])]))
### roman numerals both ways
vals = [[1000, 'M'], [900, 'CM'], [500, 'D'], [400, 'CD'], [100, 'C'], [90, 'XC'], [50, 'L'], [40, 'XL'], [10, 'X'], [9, 'IX'], [5, 'V'], [4, 'IV'], [1, 'I']]
def to_roman(n):
    s = ''
    for v, sym in vals:
        while n >= v:
            s += sym
            n -= v
    return s
romans = {'I': 1, 'V': 5, 'X': 10, 'L': 50, 'C': 100, 'D': 500, 'M': 1000}
def from_roman(s):
    total = 0
    for i in range(len(s)):
        v = romans[s[i]]
        if i + 1 < len(s) and romans[s[i + 1]] > v:
            total -= v
        else:
            total += v
    return total
print(to_roman(1994), to_roman(3888), from_roman('MCMXCIV'), from_roman('MMMCMXCIX'))
### rle encode decode
def rle_enc(s):
    if s == '':
        return ''
    out = ''
    run = 1
    for i in range(1, len(s)):
        if s[i] == s[i - 1]:
            run += 1
        else:
            out += str(run) + s[i - 1]
            run = 1
    return out + str(run) + s[-1]
def rle_dec(s):
    out = ''
    i = 0
    while i < len(s):
        n = ''
        while s[i].isdigit():
            n += s[i]
            i += 1
        out += s[i] * int(n)
        i += 1
    return out
print(rle_enc('aaabbbccccd'), rle_dec('3a4b2c'), rle_dec(rle_enc('zzzxy')))
