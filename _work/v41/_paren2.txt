### two paren tuples
p = (1, 2), (3, 4)
print(p)
### paren tuple then int
q = (1, 2), 3
print(q)
### int then paren tuple
r = 1, (2, 3)
print(r)
### two list literals
s = [1, 2], [3, 4]
print(s)
### paren tuple then list
t = (1, 2), [3, 4]
print(t)
### singletons
u = (1,), (2,)
print(u)
### doubly wrapped
v = ((1, 2)), (3, 4)
print(v)
### three elements
w = (1, 2), (3, 4), 5
print(w)
### calls
def f():
    return 1, 2
def g():
    return 3, 4
x = f(), g()
print(x)
### paren ints
y = (1), (2)
print(y)
### paren tuple then var
q = 7
z = (1, 2), q
print(z)
### inside call args
print((1, 2), (3, 4))
### chained assign
aa = bb = (1, 2), (3, 4)
print(aa, bb)
### return two paren tuples
def h():
    return (1, 2), (3, 4)
print(h())
### list of paren tuples
cc = [(1, 2), (3, 4)]
print(cc)
