### binary search iterative
def bsearch(a, t):
    lo = 0
    hi = len(a) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if a[mid] == t:
            return mid
        if a[mid] < t:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1
print(bsearch([1, 3, 5, 7, 9, 11], 7), bsearch([1, 3, 5, 7, 9, 11], 4), bsearch([], 1))
### bubble sort and insertion sort
def bsort(a):
    b = a[:]
    n = len(b)
    for i in range(n):
        for j in range(n - 1 - i):
            if b[j] > b[j + 1]:
                b[j], b[j + 1] = b[j + 1], b[j]
    return b
def isort(a):
    b = a[:]
    for i in range(1, len(b)):
        key = b[i]
        j = i - 1
        while j >= 0 and b[j] > key:
            b[j + 1] = b[j]
            j -= 1
        b[j + 1] = key
    return b
data = [5, 2, 9, 1, 7, 3]
print(bsort(data), isort(data), data)
### quicksort recursive
def qsort(a):
    if len(a) <= 1:
        return a
    p = a[0]
    left = []
    right = []
    for x in a[1:]:
        if x < p:
            left.append(x)
        else:
            right.append(x)
    return qsort(left) + [p] + qsort(right)
print(qsort([8, 3, 9, 1, 5, 5, 2, 7]))
### fibonacci memoized
memo = {0: 0, 1: 1}
def fib(n):
    if n in memo:
        return memo[n]
    r = fib(n - 1) + fib(n - 2)
    memo[n] = r
    return r
print([fib(i) for i in range(12)], fib(25))
### gcd lcm euclid
def gcd(a, b):
    while b:
        a, b = b, a % b
    return a
def lcm(a, b):
    return a * b // gcd(a, b)
print(gcd(48, 36), gcd(17, 5), lcm(4, 6), lcm(21, 6), gcd(0, 7))
### tower of hanoi
moves = []
def hanoi(n, src, dst, aux):
    if n == 1:
        moves.append(src + dst)
        return
    hanoi(n - 1, src, aux, dst)
    moves.append(src + dst)
    hanoi(n - 1, aux, dst, src)
hanoi(3, 'A', 'C', 'B')
print(len(moves), moves)
### stack class
class Stack:
    def __init__(self):
        self.items = []
    def push(self, x):
        self.items.append(x)
    def pop(self):
        if not self.items:
            raise IndexError('empty')
        return self.items.pop()
    def peek(self):
        return self.items[-1]
    def size(self):
        return len(self.items)
s = Stack()
s.push(1)
s.push(2)
s.push(3)
print(s.pop(), s.peek(), s.size())
s.pop()
s.pop()
try:
    s.pop()
except IndexError as e:
    print('caught', e)

### matrix multiply
def matmul(A, B):
    rows = len(A)
    cols = len(B[0])
    inner = len(B)
    C = []
    for i in range(rows):
        row = []
        for j in range(cols):
            s = 0
            for k in range(inner):
                s += A[i][k] * B[k][j]
            row.append(s)
        C.append(row)
    return C
A = [[1, 2], [3, 4]]
B = [[5, 6], [7, 8]]
print(matmul(A, B))

### two pointer pair sum
def pairsum(a, t):
    i = 0
    j = len(a) - 1
    while i < j:
        s = a[i] + a[j]
        if s == t:
            return [i, j]
        if s < t:
            i += 1
        else:
            j -= 1
    return None
print(pairsum([1, 2, 4, 7, 11, 15], 15), pairsum([1, 2, 3], 10))
### climbing stairs dp
def stairs(n):
    if n <= 2:
        return n
    dp = [0] * (n + 1)
    dp[1] = 1
    dp[2] = 2
    for i in range(3, n + 1):
        dp[i] = dp[i - 1] + dp[i - 2]
    return dp[n]
print([stairs(i) for i in range(1, 8)])
### coin change dp
def coinmin(coins, amount):
    INF = 10 ** 9
    dp = [INF] * (amount + 1)
    dp[0] = 0
    for a in range(1, amount + 1):
        for c in coins:
            if c <= a and dp[a - c] + 1 < dp[a]:
                dp[a] = dp[a - c] + 1
    return dp[amount] if dp[amount] < INF else -1
print(coinmin([1, 5, 12], 17), coinmin([2], 3), coinmin([1, 2, 5], 11))
### prime sieve
def sieve(n):
    is_p = [True] * (n + 1)
    is_p[0] = False
    if n >= 1:
        is_p[1] = False
    i = 2
    while i * i <= n:
        if is_p[i]:
            j = i * i
            while j <= n:
                is_p[j] = False
                j += i
        i += 1
    return [k for k in range(n + 1) if is_p[k]]
print(sieve(30), len(sieve(100)))
### reverse polish notation
def rpn(tokens):
    st = []
    for t in tokens:
        if t == '+':
            b = st.pop()
            a = st.pop()
            st.append(a + b)
        elif t == '-':
            b = st.pop()
            a = st.pop()
            st.append(a - b)
        elif t == '*':
            b = st.pop()
            a = st.pop()
            st.append(a * b)
        elif t == '/':
            b = st.pop()
            a = st.pop()
            st.append(int(a / b))
        else:
            st.append(int(t))
    return st[0]
print(rpn(['2', '1', '+', '3', '*']))
print(rpn(['4', '13', '5', '/', '+']))
print(rpn(['10', '6', '9', '3', '+', '-11', '*', '/', '*', '17', '+', '5', '+']))
### longest increasing subsequence
def lis(a):
    if not a:
        return 0
    dp = [1] * len(a)
    for i in range(1, len(a)):
        for j in range(i):
            if a[j] < a[i] and dp[j] + 1 > dp[i]:
                dp[i] = dp[j] + 1
    return max(dp)
print(lis([10, 9, 2, 5, 3, 7, 101, 18]), lis([3, 1, 2]), lis([5]))
### josephus problem
def josephus(n, k):
    people = list(range(1, n + 1))
    idx = 0
    order = []
    while people:
        idx = (idx + k - 1) % len(people)
        order.append(people.pop(idx))
    return order
print(josephus(5, 2), josephus(7, 3))
### rotate matrix layers
def rotate(a):
    n = len(a)
    b = [[0] * n for _ in range(n)]
    for i in range(n):
        for j in range(n):
            b[j][n - 1 - i] = a[i][j]
    return b
m = [[1, 2], [3, 4]]
print(rotate(m))
m3 = [[1, 2, 3], [4, 5, 6], [7, 8, 9]]
print(rotate(rotate(m3)))
