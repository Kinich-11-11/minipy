### global keyword battery
count = 0
items = []
def record(v):
    global count
    count += 1
    items.append(v)
    return count
record('a')
record('b')
print(count, items)
### decorator basic
def double_result(f):
    def wrapper(n):
        return f(n) * 2
    return wrapper
@double_result
def val(n):
    return n + 1
print(val(5), val(0))
