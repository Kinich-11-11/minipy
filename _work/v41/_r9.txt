### p1 dict object
class Thing:
    def __init__(self, n): self.n = n
    def __repr__(self): return 'Thing!' + str(self.n)
print({'k': Thing(3)})
### p2 list object
class Thing:
    def __init__(self, n): self.n = n
    def __repr__(self): return 'Thing!' + str(self.n)
print([Thing(3)])
