### r1 dict repr custom
class Thing:
    def __init__(self, n): self.n = n
    def __repr__(self): return 'Thing!' + str(self.n)
print({'k': Thing(3)})
### r2 list repr custom
print([Thing(3)])
### r3 dict str value
print({'k': 'v'})
### r4 dict nested list
print({'k': [1, 2]})
### r5 dict inside list
print([{'a': 1}])
### r6 str of dict
print(str({'k': Thing(3)}))
