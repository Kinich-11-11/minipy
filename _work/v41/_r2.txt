### k1 max key only
print(max([1, -5], key=abs))
### k2 max default only
print(max([1, 2], default=0))
### k3 min default only
print(min([], default=-1))
### k4 max two kwargs
print(max([1, 2], default=0, key=abs))
### k5 sorted reverse
print(sorted([1, 2], reverse=True))
### k6 sorted key reverse
print(sorted([1, 2], key=abs, reverse=True))
### k7 user fn kw
def f(a, b=2):
    return a + b
print(f(1, b=5))
### k8 user fn unknown kw
def g(a):
    return a
print(g(1, zz=2))
