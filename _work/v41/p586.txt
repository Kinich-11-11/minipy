// ---- P586: map 回调结果过 lesc（P585 的 divmod 元组暴露的既有缺口）----
//
// `list(map(divmod, xs, ys))` 得到 `[(), 0, 1, …]`：P262/P567 的 map 收集器
// `mpout = enc(mpout, evtype, ev)` **没有过 lesc**。回调结果（eval 产物）是
// 未转义文本，元组/列表/字典的编码自带 lsep —— 直接拼进 mpout 后，结果列表
// 自身的 lsep 结构被内层 lsep 撑爆（实测每个元组拆成 `()` `0` `1` 三段）。
// filter 不受影响：它回填的是 listitem 的产物（存储段本来就是转义形态）。
// 修法：两处 map enc 前先 `lesc(ev)`（lesc 就地改 ev，enc 拼接的正是转义后的
// 文本；标量结果 lesc 幂等无副作用）。
patch('P586 map collector escapes callback results', () => {
  // VALUE 引用链里的变量名（描述符形态有 [12,name,id] 内联与块引用两种，统一展开）
  const varNames = (root) => {
    const out = new Set();
    const walk = (id, d) => {
      if (!id || d > 14) return;
      const q = BL()[id];
      if (!q) return;
      if (q.opcode === 'data_variable' && q.fields.VARIABLE) out.add(q.fields.VARIABLE[0]);
      for (const v of Object.values(q.inputs || {})) {
        const x = v[1];
        if (typeof x === 'string') walk(x, d + 1);
        else if (Array.isArray(x)) {
          if (x[0] === 12) out.add(x[1]);
          else if (typeof x[1] === 'string') walk(x[1], d + 1);
        }
      }
    };
    walk(root, 0);
    return out;
  };
  const hits = [];
  for (const id of allIds()) {
    const b = BL()[id];
    if (b.opcode !== 'data_setvariableto' || !b.fields.VARIABLE || b.fields.VARIABLE[0] !== 'mpout') continue;
    const val = b.inputs.VALUE;
    if (!Array.isArray(val) || typeof val[1] !== 'string') continue;
    const names = varNames(val[1]);
    if (names.has('evtype') && names.has('ev') && names.has('lsep')) hits.push(id);
  }
  if (hits.length !== 2) throw new Error('[P586] map enc sites = ' + hits.length);
  for (const site of hits) {
    insertBeforeStmt(site, [callProc('lesc %s', [VAR('ev')])]);
  }
  console.log('[P586] lesc installed before ' + hits.length + ' map collector(s)');
});
