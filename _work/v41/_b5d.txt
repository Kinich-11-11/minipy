### C1 compose selfcontained
def dbl(x):
    return x * 2
def inc(x):
    return x + 1
def compose(f, g):
    return lambda x: f(g(x))
h = compose(dbl, inc)
print(h(5))
### C2 lambda fn plus var selfcontained
def ten(x):
    return 10
def ap2(f, v):
    return lambda x: f(x) + v
h2 = ap2(ten, 5)
print(h2(1))
### C3 lambda nested direct
def dbl2(x):
    return x * 2
def inc2(x):
    return x + 1
lam3 = lambda x: dbl2(inc2(x))
print(lam3(5))
### C4 single fn capture plus var
def ten3(x):
    return 10
def ap4(f, v):
    return lambda x: f(x) + v
h4 = ap4(ten3, 5)
print(h4(1))
