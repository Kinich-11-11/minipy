### f1 different class rhs
class B:
    def __init__(self, x): self.x = x
    def __add__(self, o): return B(self.x + o.x)
class A:
    def __init__(self, x): self.x = x
print((B(1) + A(2)).x)
### f2 same class rhs
class B:
    def __init__(self, x): self.x = x
    def __add__(self, o): return B(self.x + o.x)
print((B(1) + B(2)).x)
### f3 same class, const return
class B:
    def __init__(self, x): self.x = x
    def __add__(self, o): return 7
print(B(1) + B(2))
### f4 same class, return self
class B:
    def __init__(self, x): self.x = x
    def __add__(self, o): return self
print((B(1) + B(2)).x)
### f5 same class, no new object
class B:
    def __init__(self, x): self.x = x
    def __add__(self, o): return self.x + o.x
