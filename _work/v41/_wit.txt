### witness leading bracket
augw = ''
(a, b), c = (1, 2), 3
print('W1', augw)
print(a, b, c)
### witness flat
augw = ''
a, b = 1, 2
print('W2', augw)
print(a, b)
### witness nested second
augw = ''
a, (b, c) = 1, (2, 3)
print('W3', augw)
print(a, b, c)
### witness for stmt
augw = ''
for (a, b), c in [((1, 2), 3)]:
    print('W4', augw)
    print(a, b, c)
