### unhashable key probe
try:
    d = {}
    d[[1]] = 2
    print('no error', d)
except TypeError as e:
    print('caught', str(e)[:20])
try:
    s = {[1]}
    print('no error2', s)
except TypeError as e:
    print('caught2', str(e)[:20])
try:
    d2 = {{1: 2}: 3}
    print('no error3', d2)
except TypeError as e:
    print('caught3', str(e)[:20])
