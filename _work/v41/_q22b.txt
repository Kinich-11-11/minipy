### RL1 recursion with loop
def rl(n):
    print('enter', n)
    acc = []
    for i in range(n):
        acc.append(i)
        if n > 1:
            rl(n - 1)
    print('exit', n, acc)
    return acc
print(rl(3))
### RL2 nested call in loop no recursion
def g(v):
    return v * 2
out = []
for i in [1, 2, 3]:
    out.append(g(i))
print(out)
### RL3 loop at end of function body
def r3(n):
    acc = []
    for i in range(2):
        acc.append(n * 10 + i)
        if n > 1:
            r3(n - 1)
    return acc
print(r3(2))
### RL4 loop last stmt inner
def r4(n):
    acc = []
    if n > 1:
        for j in range(2):
            acc.append(j)
            r4(n - 1)
    return acc
print(r4(2))
