### rec extend
def g(n):
    out = [n]
    if n > 0:
        out.extend(g(n - 1))
    return out
print(g(4))
### rec append
def g(n):
    out = [n]
    if n > 0:
        out.append(n)
        out.extend(g(n - 1))
    return out
print(g(4))
### rec rebind
def g(n):
    out = [n]
    if n > 0:
        out = out + g(n - 1)
    return out
print(g(4))
### rec nested call arg
def h(x):
    return [x]
def g(n):
    out = [n]
    if n > 0:
        out.extend(h(n) + g(n - 1))
    return out
print(g(4))
