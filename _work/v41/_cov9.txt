### parenthesized continuation in call
print(1 +
      2)
### parenthesized continuation list literal
a = [1,
     2,
     3]
print(a, len(a))
### parenthesized continuation dict literal
d = {
    'a': 1,
    'b': 2,
}
print(d)
### nested parenthesized calls
print(max(1,
          2),
      min(3,
          4))
### multi line function call after definition
def add(a, b, c):
    return a + b + c
total = add(1,
            2,
            3)
print(total)
### multi line condition
x = 5
if (x > 1 and
        x < 10):
    print('in range')
### multi line boolean chain
a = True
b = False
c = True
print(a and
      b or
      c)
### backslash continuation
x = 1 + \
    2
print(x)
### backslash in string is not continuation
print('a\
b')
### trailing comma in call and literal
print(1, 2,)
print([1, 2,])
print((1, 2,))
print({'a': 1,})
def f(a,):
    return a
print(f(1,))
### implicit string concatenation
print('a' 'b')
print('a' 'b' 'c')
### implicit concatenation across lines
s = ('a'
     'b')
print(s)
### blank lines inside blocks
def f():
    x = 1

    y = 2

    return x + y
print(f())
### blank lines inside if and for
for i in range(2):

    if i >= 0:

        print('i', i)

    # comment
    x = 1
    # another
    return x

      for i in range(1):
          print('deep', i)

1}')

### dict literal spanning lines with nested values
cfg = {
    'name': 'x',
    'opts': {
        'a': 1,
        'b': 2
    },
    'list': [1, 2, 3]
}
print(cfg['name'], cfg['opts']['a'], cfg['list'][1])
### list literal spanning lines with calls
vals = [
    len('ab'),
    max(1, 2),
    min([3, 4])
]
print(vals)
### conditional expression spanning lines
x = 5
y = (1
     if x > 3
     else 2)
print(y)
### string with parentheses and quotes
print('(a)', "b'c", 'd"e')
### return spanning lines
def r(n):
    return (n * 2
            + 1)
print(r(3))
### while condition spanning lines
i = 0
while (i < 3 and
       i >= 0):
    i += 1
print(i)
### deeply nested parentheses

    return a + b

        b=5))
