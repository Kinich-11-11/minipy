// PZ36（临时调试）：给 eval 里所有 callfn 调用点编号，并在 callfn 入口打印编号 + (fn,lo,hi)
patch('PZ36 callfn site tags', () => {
  newVar('cfsite', '');
  newVar('dbg', '');
  const nameOf = (d) => {
    if (!Array.isArray(d)) return null;
    if (d[0] === 12) return d[1];
    if ((d[0] === 1 || d[0] === 2 || d[0] === 3) && Array.isArray(d[1])) return nameOf(d[1]);
    const id = typeof d === 'string' ? d : (typeof d[1] === 'string' ? d[1] : null);
    const b = id ? BL()[id] : null;
    if (!b) return null;
    if (b.opcode === 'data_variable') return b.fields.VARIABLE[0];
    if (b.opcode === 'argument_reporter_string_number') return b.fields.VALUE[0];
    return null;
  };
  const isStatement = (id) => {
    const par = BL()[id].parent;
    if (!par) return false;
    const pb = BL()[par];
    return !Object.values(pb.inputs || {}).some((v) => Array.isArray(v) && v[1] === id);
  };
  const ev = reach(defIdOf('eval %s %s'));
  const sites = find(ev, (b) => b.opcode === 'procedures_call' && b.mutation
    && ['callfn %s %s %s', 'callfnW %s %s %s'].includes(b.mutation.proccode));
  console.log('[PZ36] callfn sites in eval = ' + sites.length);
  let i = 0;
  const map = [];
  for (const s of sites) {
    i++;
    const args = Object.keys(BL()[s].inputs).map((k) => {
      const v = BL()[s].inputs[k];
      const t = Array.isArray(v) && typeof v[1] === 'string' ? BL()[v[1]] : null;
      return t ? (t.opcode === 'operator_add' || t.opcode === 'operator_subtract' ? t.opcode.replace('operator_', '') : t.opcode) : JSON.stringify(v);
    });
    const tag = 'CF' + i;
    map.push(tag + '[' + BL()[s].mutation.proccode.split(' ')[0] + ' ' + args.join(',') + (isStatement(s) ? '' : ' REPORTER') + ']');
    if (!isStatement(s)) continue;
    try { insertBeforeStmt(s, [setV('cfsite', txt(tag))]); } catch (e) { console.log('[PZ36] tag failed ' + tag + ': ' + e.message); }
  }
  console.log('[PZ36] map: ' + map.join(' '));
  // callfn 入口打印
  let defId = null;
  for (const id of allIds()) {
    const b = BL()[id];
    if (b.opcode === 'procedures_definition') {
      const pr = BL()[b.inputs.custom_block[1]];
      if (pr.mutation.proccode === 'callfn %s %s %s') defId = id;
    }
  }
  const head = BL()[defId].next;
  insertBeforeStmt(head, [
    setV('dbg', txt('CF@')),
    setV('dbg', JOIN(VAR('dbg'), VAR('cfsite'))),
    setV('dbg', JOIN(VAR('dbg'), txt(' fn='))),
    setV('dbg', JOIN(VAR('dbg'), argOf('fn'))),
    setV('dbg', JOIN(VAR('dbg'), txt(' lo='))),
    setV('dbg', JOIN(VAR('dbg'), argOf('lo'))),
    setV('dbg', JOIN(VAR('dbg'), txt(' hi='))),
    setV('dbg', JOIN(VAR('dbg'), argOf('hi'))),
    setV('dbg', JOIN(VAR('dbg'), txt(' tok='))),
    setV('dbg', JOIN(VAR('dbg'), item('Tok', argOf('lo')))),
    callProc('fail %s', [VAR('dbg')])
  ]);
  console.log('[PZ36] callfn entry probe installed');
});
