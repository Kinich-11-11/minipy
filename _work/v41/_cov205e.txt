### nested try rethrow
def level2():
    try:
        raise ValueError('deep')
    except ValueError:
        raise
def level1():
    try:
        level2()
    except ValueError as e:
        print('caught in level1', e)
        raise RuntimeError('wrapped')
try:
    level1()
except RuntimeError as e:
    print('caught at top', e)
### finally ordering
def f(n):
    try:
        if n == 0:
            raise KeyError('zero')
        print('body', n)
    except KeyError as e:
        print('except', e)
    finally:
        print('finally', n)
f(0)
f(1)
try:
    try:
        raise IndexError('inner')
    finally:
        print('fin1')
except IndexError as e:
    print('outer got', e)
### exception with attributes
class MyErr(Exception):
    def __init__(self, code, msg):
        Exception.__init__(self, msg)
        self.code = code
        self.msg = msg
def check(v):
    if v < 0:
        raise MyErr(400, 'negative ' + str(v))
    if v > 100:
        raise MyErr(413, 'too big ' + str(v))
    return v * 2
for v in [5, -3, 500]:
    try:
        print(check(v))
    except MyErr as e:
        print(e.code, e.msg, str(e))
### exception in loop continue
vals = [1, 'a', 3, None, 5]
total = 0
caught = 0
for v in vals:
    try:
        total += int(v) * 2
    except (TypeError, ValueError):
        caught += 1
        continue
    finally:
        print('step', v)
print('total', total, 'caught', caught)

### global keyword in nested funcs
count = 0
def outer():
    global count
    def inner():
        global count
        count += 5
    inner()
    inner()
    count += 1
outer()
print(count)
acc = []
def rec(n):
    if n == 0:
        return
    acc.append(n)
    rec(n - 1)
    acc.append(n)
rec(3)
print(acc)
