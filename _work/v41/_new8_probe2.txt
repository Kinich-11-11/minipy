    return len(dp), len(dp[0])

### probe knapsack inner write
def k2(ws, cap):
    n = len(ws)
    dp = [[0] * (cap + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for c in range(cap + 1):
            dp[i][c] = dp[i - 1][c]
    return dp
print(k2([2, 3], 3))
### probe knapsack read below
def k3(ws, cap):
    n = len(ws)
    dp = [[0] * (cap + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for c in range(cap + 1):
            dp[i][c] = dp[i - 1][c]
            if ws[i - 1] <= c:
                dp[i][c] = dp[i - 1][c - ws[i - 1]] + 1
    return dp
print(k3([2, 3], 4))

### probe bst inorder alone
class BT:
    def __init__(self, v):
        self.v = v
        self.l = None
        self.r = None
    def ins(self, v):
        if v < self.v:
            if self.l is None:
                self.l = BT(v)
            else:
                self.l.ins(v)
        else:
            if self.r is None:
                self.r = BT(v)
            else:
                self.r.ins(v)
    def inorder(self):
        out = []
        if self.l is not None:
            out.extend(self.l.inorder())
        out.append(self.v)
        if self.r is not None:
            out.extend(self.r.inorder())
        return out
root = BT(5)
for x in [3, 8, 1]:
    root.ins(x)
print(root.inorder())
### probe bst without extend
class BT2:
    def __init__(self, v):
        self.v = v
        self.l = None
        self.r = None
    def ins(self, v):
        if v < self.v:
            if self.l is None:
                self.l = BT2(v)
            else:
                self.l.ins(v)
        else:
            if self.r is None:
                self.r = BT2(v)
            else:
                self.r.ins(v)
    def collect(self, out):
        if self.l is not None:
            self.l.collect(out)
        out.append(self.v)
        if self.r is not None:
            self.r.collect(out)
        return out
root2 = BT2(5)
for x in [3, 8, 1]:
    root2.ins(x)
print(root2.collect([]))
