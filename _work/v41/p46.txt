// ---- P46: make listlit reentrant -- nested container literals (A13)
//
// Root cause (proved with the P44 probe, see A13_context.md): a nested list literal
// calls listlit again, and the inner call clears the *shared* scratch lists
// LComm/LOut and overwrites lpi/lplo -- so the outer call loses its element
// boundaries and its already-evaluated elements.  It then reads an empty LComm,
// computes lplo=1 and evaluates the tail as eval(1, hi) -> 'SyntaxError: parentheses'.
//
// Fix: save exactly the state the outer call still needs (lpi, lplo + LComm + LOut)
// across each recursive eval, on VStack, in the same LIFO discipline tuplelit uses,
// and restore it unconditionally right after eval returns (ev/evtype/error are not
// touched by the save/restore, so the element value survives).
//
// NB the three earlier attempts (P42/P43/P45) all pushed the list length BEFORE the
// elements and then read it back with `item last of VStack`, which returns the last
// ELEMENT, not the length; the mismatched `del` count wrecked VStack.  Here the
// elements are pushed in reverse and the count last, so the count really is on top.
// NB a `control_repeat_until` body runs while its condition is FALSE, so the
// count-down push loop must test NOT(i > 0), not i > 0.
patch('P46 listlit reentrant', () => {
  for (const v of ['l6n', 'l6i']) newVar(v, 0);
  const delLast = (list) => I('data_deleteoflist', { fields: B.LISTF(list), inputs: { INDEX: txt('last') } });
  const pushV = (n) => addTo('VStack', VAR(n));
  const popV = (n) => [setV(n, item('VStack', llen('VStack'))), delLast('VStack')];
  // push the list contents in REVERSE order, then the count on top
  const saveL = (X) => [
    setV('l6n', llen(X)),
    setV('l6i', VAR('l6n')),
    REPEATU(NOT(GT(VAR('l6i'), num(0))), [addTo('VStack', item(X, VAR('l6i'))), chgV('l6i', num(-1))]),
    addTo('VStack', VAR('l6n'))
  ];
  // pop the count, then pop the elements back in forward order
  const restoreL = (X) => [
    setV('l6n', item('VStack', llen('VStack'))),
    delLast('VStack'),
    clearL(X),
    setV('l6i', num(1)),
    REPEATU(GT(VAR('l6i'), VAR('l6n')), [addTo(X, item('VStack', llen('VStack'))), delLast('VStack'), chgV('l6i', num(1))])
  ];
  const saveState = () => [pushV('lpi'), pushV('lplo'), ...saveL('LComm'), ...saveL('LOut')];
  const restoreState = () => [...restoreL('LOut'), ...restoreL('LComm'), ...popV('lplo'), ...popV('lpi')];
  // the error guard spells the error variable either as a data_variable block or as
  // an inline descriptor [3,[12,'error',id],[10,'']] -- accept both shapes
  const mentionsError = (desc) => {
    const seen = new Set(); const stack = [desc];
    while (stack.length) {
      const cur = stack.pop();
      if (cur === null || cur === undefined) continue;
      if (Array.isArray(cur)) {
        if (cur[0] === 12 && cur[1] === 'error') return true;
        for (const e of cur) stack.push(e);
        continue;
      }
      if (typeof cur !== 'string' || seen.has(cur)) continue;
      seen.add(cur);
      const b = BL()[cur]; if (!b) continue;
      if (b.opcode === 'data_variable' && b.fields.VARIABLE[0] === 'error') return true;
      for (const v of Object.values(b.inputs || {})) stack.push(v);
    }
    return false;
  };
  const ll = inDef('listlit %s %s');
  const evals = find(ll, (b) => b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === 'eval %s %s');
  if (evals.length !== 2) throw new Error('[P46] expected 2 eval anchors in listlit, got ' + evals.length);
  let n = 0;
  for (const callId of evals) {
    // sanity: each anchor must be followed by the error guard, the lesc call and the LOut append
    let cur = BL()[callId].next, guard = 0, ok = false;
    while (cur && guard++ < 4) {
      const b = BL()[cur];
      if (b.opcode === 'control_if' && b.inputs.CONDITION && mentionsError(b.inputs.CONDITION[1])) ok = true;
      cur = b.next;
    }
    if (!ok) throw new Error('[P46] eval anchor ' + callId + ' has no error guard');
    insertBeforeStmt(callId, saveState());
    spliceAfter(callId, chain(restoreState()));
    n++;
  }
  console.log('[P46] listlit recursive evals protected = ' + n);
});
