### d1 attr on assigned result
class V:
    def __init__(self, x): self.x = x
    def __add__(self, o): return V(self.x + o.x)
r = V(1) + V(2)
print(r.x)
### d2 attr directly on parens
class V:
    def __init__(self, x): self.x = x
    def __add__(self, o): return V(self.x + o.x)
print((V(1) + V(2)).x)
