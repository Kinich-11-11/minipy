### print sep only empty
print('x', 'y', sep='')
### print end only
print('x', 'y', end='')
print('z')
### print sep and end
print('x', 'y', sep='', end='\n')
### print sep dash end bang
print('x', 'y', sep='-', end='!')
### print sep dash only
print('x', 'y', sep='-')
### print end newline explicit
print('x', 'y', end='\n')
### print empty sep nonempty end
print('a', 'b', sep='', end=';')
### print with end only then next
print('a', end='-')
print('b', end='-')
print('c')
### print sep empty three args
print(1, 2, 3, sep='')
### print no kwargs
print('x', 'y')
### print sep with spaces
print('x', 'y', sep=' ')
### print end empty then newline
print('p', end='')
print('q')
