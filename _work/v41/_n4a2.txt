class Node:
    def __init__(self, v):
        self.v = v
        self.nxt = None
class LinkedList:
    def __init__(self):
        self.head = None
        self.n = 0
    def append(self, v):
        node = Node(v)
        if self.head is None:
            self.head = node
        else:
            cur = self.head
            while cur.nxt is not None:
                cur = cur.nxt
            cur.nxt = node
        self.n += 1
    def to_list(self):
        out = []
        cur = self.head
        while cur is not None:
            out.append(cur.v)
            cur = cur.nxt
        return out
    def has(self, v):
        cur = self.head
        while cur is not None:
            if cur.v == v:
                return True
            cur = cur.nxt
        return False
ll = LinkedList()
for x in [3, 1, 4, 1, 5]:
    ll.append(x)
print(ll.to_list(), ll.n, ll.has(4), ll.has(9))

class Stack:
    def __init__(self):
        self.items = []
    def push(self, v):
        self.items.append(v)
    def pop(self):
        if not self.items:
            raise IndexError('pop from empty stack')
        return self.items.pop()
s = Stack()
s.push(1)
s.push(2)
print(s.pop(), s.pop())
try:
    s.pop()
except IndexError as e:
    print('caught:', e)
