### k1 list builtin key
print(sorted(['bb', 'a', 'ccc'], key=len))
### k2 dict builtin key
d = {'bb': 1, 'a': 2, 'ccc': 3}
print(sorted(d, key=len))
### k3 list lambda key
print(sorted(['bb', 'a', 'ccc'], key=lambda k: len(k)))
### k4 dict builtin key str
d2 = {'b': 1, 'a': 2}
print(sorted(d2, key=str))
