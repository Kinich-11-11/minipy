### list slice step zero
try:
    x = [1, 2, 3][::0]
except ValueError:
    print('list caught')
### string slice step zero
try:
    x = 'abc'[::0]
except ValueError:
    print('str caught')
### string slice step zero explicit range
try:
    x = 'abc'[0:3:0]
except ValueError:
    print('str range caught')
### variable slice step zero
s = 'abc'
i = 0
try:
    x = s[::i]
except ValueError:
    print('var caught')
### slice step zero on list variable
a = [1, 2, 3]
try:
    x = a[::0]
except ValueError:
    print('listvar caught')
### slice step zero expression step
try:
    x = 'abcdef'[1:5:0]
except ValueError:
    print('expr caught')
### slice step zero via function
def sl(s):
    return s[::0]
try:
    sl('abc')
except ValueError:
    print('fn caught')
