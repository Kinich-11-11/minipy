### attr set after ctor
class N:
    def __init__(self, v):
        self.v = v
        self.kid = None
    def vals(self):
        return [self.v]
l1 = N(1)
top = N(5)
top.kid = l1
print(top.kid.vals())
print(top.kid.vals())
### attr set in ctor from param
class M:
    def __init__(self, v, kid):
        self.v = v
        self.kid = kid
    def vals(self):
        return [self.v]
m2 = M(2, None)
m1 = M(1, m2)
print(m1.kid.vals())
