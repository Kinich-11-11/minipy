for i, w in enumerate(['a', 'b', 'c']):
    print(i, w)
print(list(enumerate(['x', 'y'], start=1)))
print(list(reversed([1, 2, 3])), list(reversed('abc')))
### decorator basic
def double_result(f):
    def wrapper(n):
        return f(n) * 2
    return wrapper
@double_result
def val(n):
    return n + 1
print(val(5), val(0))
