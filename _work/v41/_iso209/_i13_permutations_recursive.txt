### permutations recursive
def perms(xs):
    if len(xs) <= 1:
        return [xs]
    out = []
    for i in range(len(xs)):
        rest = xs[:i] + xs[i + 1:]
        for p in perms(rest):
            out.append([xs[i]] + p)
    return out
print(sorted([''.join(p) for p in perms(['a', 'b', 'c'])]))
