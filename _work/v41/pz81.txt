// PZ81（临时调试）：listlit 收到的 lo/hi 与两端 token
patch('PZ81 listlit range probe', () => {
  const taken = new Set(Object.values(T(IT).variables).map((x) => x[0]));
  const names = ['z1', 'z2', 'z3', 'z4', 'z5', 'dbg'];
  for (const n of names) if (taken.has(n)) throw new Error('[PZ81] var taken: ' + n);
  for (const n of names) newVar(n, '');
  const def = defIdOf('listlit %s %s');
  const head = BL()[def].next;
  if (!head) throw new Error('[PZ81] listlit body not found');
  const parts = [
    setV('dbg', txt('PZ81 lo=')),
    setV('dbg', JOIN(VAR('dbg'), VAR('z1'))),
    setV('dbg', JOIN(VAR('dbg'), txt(' tok='))),
    setV('dbg', JOIN(VAR('dbg'), VAR('z3'))),
    setV('dbg', JOIN(VAR('dbg'), txt(' hi='))),
    setV('dbg', JOIN(VAR('dbg'), VAR('z2'))),
    setV('dbg', JOIN(VAR('dbg'), txt(' tok='))),
    setV('dbg', JOIN(VAR('dbg'), VAR('z4'))),
    setV('dbg', JOIN(VAR('dbg'), txt(' next='))),
    setV('dbg', JOIN(VAR('dbg'), VAR('z5')))
  ];
  insertBeforeStmt(head, [
    setV('z1', argOf('lo')),
    setV('z2', argOf('hi')),
    setV('z3', item('Tok', argOf('lo'))),
    setV('z4', item('Tok', argOf('hi'))),
    setV('z5', item('Tok', ADD(argOf('hi'), num(1))))
  ].concat(parts).concat([FAIL(VAR('dbg'))]));
  console.log('[PZ81] probe installed');
});
