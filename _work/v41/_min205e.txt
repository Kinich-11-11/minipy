### R1 init no classattr three instances
class R1:
    n = 0
    def __init__(self, t):
        self.t = t
    def bump(self):
        R1.n += 10
        return R1.n
a = R1('a')
b = R1('b')
c = R1('c')
print(R1.n, a.bump(), R1.n)
### R2 init bumps classattr one instance
