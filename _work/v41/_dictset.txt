### dict overwrite in function
def setk(d, k, v):
    d[k] = v
    return d
d = {'a': 1, 'b': 2}
print(setk(d, 'a', 9))
print(d)
### dict overwrite inside loop in function
def bump(d, k):
    for i in range(2):
        d[k] = d.get(k) + 1
    return d
print(bump({'a': 1, 'b': 2}, 'a'))

### dict word count
def count(words):
    counts = {}
    for w in words:
        counts[w] = counts.get(w, 0) + 1
    return counts
print(count(['a', 'b', 'a', 'a']))
### dict of lists accumulate
def group(pairs):
    g = {}
    for k, v in pairs:
        if g.get(k) == None:
            g[k] = []
        g[k].append(v)
    return g
print(group([('a', 1), ('b', 2), ('a', 3)]))
### dict built in loop then overwritten

### inventory update loop
stock = {'apple': 10, 'pear': 5}
moves = [('apple', -3), ('pear', 2), ('apple', 1)]
for name, delta in moves:
    if stock.get(name) == None:
        stock[name] = 0
    stock[name] = stock.get(name) + delta
print(sorted(stock.items()))
### word frequency report
text = 'the cat and the hat and the bat'
counts = {}
for w in text.split():
    counts[w] = counts.get(w, 0) + 1
for w in sorted(counts):
    print(w, counts[w])
### groups and totals
rows = [('a', 1), ('b', 2), ('a', 3), ('b', 4), ('c', 5)]
totals = {}
for k, v in rows:
    if totals.get(k) == None:
        totals[k] = 0
    totals[k] = totals.get(k) + v
print(sorted(totals.items()))
