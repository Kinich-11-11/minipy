### two-level factory no decorator
def addn(k):
    def deco(f):
        def w(x):
            return f(x) + k
        return w
    return deco
def inc(x):
    return x * 1
w = addn(10)(inc)
print(w(5))
### simple factory on 210
def mk(k):
    def ap(x):
        return x * k
    return ap
g = mk(3)
print(g(4))
