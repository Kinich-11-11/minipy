// PZ96（临时调试）：给 eval 内部每一个"调用 eval"的站点编号，并打印当时的 lo/hi
// 目的：找出 c2 里那个 `eval(5,8)`（区间 `5 ) ( 1`）到底是哪个站点发出的
patch('PZ96 eval-in-eval site markers', () => {
  const BB = T(IT).blocks;
  const ownerOf = (id) => {
    let cur = id;
    for (let i = 0; i < 400 && cur; i++) {
      const q = BB[cur];
      if (!q) return '?';
      if (q.opcode === 'procedures_definition') {
        const pr = BB[q.inputs.custom_block[1]];
        return pr && pr.mutation ? String(pr.mutation.proccode) : '?';
      }
      cur = q.parent;
    }
    return '(toplevel)';
  };
  const sites = [];
  for (const id of Object.keys(BB)) {
    const b = BB[id];
    if (b.opcode !== 'procedures_call' || !b.mutation) continue;
    if (String(b.mutation.proccode) !== 'eval %s %s') continue;
    const ow = ownerOf(id);
    if (ow === 'eval %s %s') continue;
    if (!/atom|callfn|calluser|cdo|methodcall|indexget|listlit|builtin/.test(ow)) continue;
    sites.push(id);
  }
  if (!sites.length) throw new Error('[PZ96] no eval-in-eval sites');
  let k = 0;
  for (const id of sites) {
    k++;
    const tag = 'EV' + k;
    try {
    insertBeforeStmt(id, [
      callProc('log %s', [JOIN(txt(tag + ' lo='), argOf('lo'))]),
      callProc('log %s', [JOIN(txt(tag + ' hi='), argOf('hi'))])
    ]);
    } catch (e) { console.log('[PZ96] skip ' + id + ' : ' + e.message); }
  }
  console.log('[PZ96] marked ' + k + ' eval-in-eval site(s): ' + sites.join(','));
});
