### a max default
print(max([1, 2], default=0))
### b min empty default
print(min([], default=-1))
### c max plain
print(max([1, 2]))
### d min plain empty
print(min([]))
### e max key
print(max([1, -5], key=abs))
### f pow one arg
print(pow(2))
### g pow two
print(pow(2, 3))
### h pow three
print(pow(2, 3, 5))
### i sorted key
print(sorted([1, -5], key=abs))
### j min default with key
print(min([], default=-1, key=abs))
