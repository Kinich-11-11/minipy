### len of empties
print(len([]), len({}), len(()), len(set()), len(''))
### len of one element
print(len([1]), len({1: 2}), len((1,)), len({1}), len('a'))
### len of two elements
print(len([1, 2]), len({1: 2, 3: 4}), len((1, 2)), len({1, 2}), len('ab'))
### empty container printed
print([], {}, (), set())
### empty list in list
print([[]], len([[]]))
### empty dict nested
print({'a': {}}, len({'a': {}}))
### comparison of empty container to empty string
print([] == '', {} == '', () == '', set() == '')
### nested empty truthiness via len
print(len([]) == 0, len({}) == 0, len(set()) == 0, len(()) == 0)
### list append then len
a = []
a.append(1)
print(len(a))
### dict keys of empty
d = {}
print(len(d.keys()), list(d.keys()))
### range len
print(len(range(0)), len(range(3)))
