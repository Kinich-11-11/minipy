### try except else finally
def f(x):
    try:
        r = 10 // x
    except ZeroDivisionError:
        return 'zero'
    else:
        return 'ok:' + str(r)
print(f(2), f(0))
### finally runs and overrides return
def g():
    try:
        return 1
    finally:
        return 2
def h():
    out = []
    try:
        out.append('t')
        return 'ret'
    finally:
        out.append('f')
print(g())
print(h())
### raise and catch with message
def conv(s):
    try:
        return int(s)
    except ValueError as e:
        return 'bad:' + str(e)
print(conv('42'), conv('x1'))
### assert caught with message
def check(n):
    try:
        assert n > 0, 'bad value'
        return 'ok'
    except AssertionError as e:
        return 'err:' + str(e)
print(check(3), check(-1))
### exception hierarchy order
def p(v):
    try:
        x = 1 / v
        return 'inf?'
    except ZeroDivisionError:
        return 'zde'
    except Exception:
        return 'exc'
print(p(0), p(2))
### chained comparisons
print(1 < 2 < 3, 3 > 2 == 2, 1 < 2 > 3, 5 == 5 == 5, 1 <= 1 <= 1, 4 > 3 > 2 > 1)
### short circuit side effects
calls = []
def t(v):
    calls.append(v)
    return v
r1 = t(False) and t('never')
r2 = t(True) or t('never')
r3 = t(0) or t('fallback')
r4 = t(1) and t('kept')
print(r1, r2, r3, r4, calls)
### nested ternary
def grade(x):
    return 'hi' if x >= 90 else 'mid' if x >= 60 else 'low'
print(grade(95), grade(70), grade(10), grade(90), grade(60))
