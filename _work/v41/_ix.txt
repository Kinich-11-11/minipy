### p1 one indexed target
a = [5, 2]
a[0] = 9
print(a)
### p2 two indexed targets
a = [5, 2]
a[0], a[1] = 9, 8
print(a)
### p3 swap
a = [5, 2]
a[0], a[1] = a[1], a[0]
print(a)
### p4 plain tuple target
a = [5, 2]
x, y = a[0], a[1]
print(x, y)
### p5 two indexed no parens rhs
a = [5, 2]
a[1], a[0] = 8, 9
print(a)
### p6 dict indexed targets
d = {'x': 1, 'y': 2}
d['x'], d['y'] = 9, 8
print(d)
