### k1 closure basic
def adder(n):
    def add(x):
        return x + n
    return add
add5 = adder(5)
print(add5(1))
### k2 closure no arg
def mk():
    v = 9
    def g():
        return v
    return g
print(mk()())
### k3 global read in nested
n = 7
def f(x):
    return x + n
print(f(1))
### k4 local in same fn
def h(n):
    def inner(x):
        return x + n
    return inner(1)
print(h(5))
### k5 nested def called immediately
def o(n):
    def i(x):
        return x + n
    return i(2)
print(o(3))
