### dict comprehension
print({k: v for k, v in [('a', 1), ('b', 2)]})

### dict accumulate in function
def count(words):
    counts = {}
    for w in words:
        counts[w] = counts.get(w, 0) + 1
    return counts
print(count(['a', 'b', 'a', 'a']))
### dict of lists accumulated
def group(pairs):
    g = {}
    for k, v in pairs:
        if g.get(k) == None:
            g[k] = []
        g[k].append(v)
    return g
print(group([('a', 1), ('b', 2), ('a', 3)]))

### set basics
