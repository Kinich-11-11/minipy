### mutable default identity and alias
def acc(v, buf=[]):
    buf.append(v)
    return buf
a = acc(1)
b = acc(2)
a.append(9)
print(a, b)
print(acc(3))
### default as membership seen list
def f(x, seen=[]):
    if x in seen:
        return 'dup'
    seen.append(x)
    return 'new'
print(f(1), f(2), f(1), f(3))
