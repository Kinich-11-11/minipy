### nested comprehension as iterable
print([x for x in [i for i in range(3)]])
### comprehension over call with inner for
print([x for x in sorted([3, 1, 2])])
### comprehension in dict value
print({'a': [i for i in range(2)]})
### for over list comprehension
for x in [i * 2 for i in range(3)]:
    print(x)
### comprehension with tuple arg
print([x for x in [(1, 2), (3, 4)]])
### comprehension containing in keyword
d = {'a': 1}
print([k for k in d if 'a' in k])
