### T1 append str method no concat
out1 = []
out1.append('abc'.upper())
print(out1)
### T2 append concat with str method
out2 = []
out2.append('a' + 'bc'.upper())
print(out2)
### T3 join arg with str method
print(' '.join(['ab'.upper()]))
print('-'.join(['x', 'y'.upper()]))
### T4 append builtin call in arg
out4 = []
out4.append(len('abc') + int('2'))
print(out4)
### T5 append method call on var string
s5 = 'abc'
out5 = []
out5.append(s5.upper())
print(out5)
### T6 nested method call arg on dict method
d6 = {}
d6.update({'a': 1})
print(d6)
### T7 pop as arg unrolled context
lst7 = [1, 2, 3]
print(lst7.pop(1), lst7)
### T8 append pop as arg
lst8 = [1, 2, 3]
acc8 = []
acc8.append(lst8.pop(1))
print(acc8, lst8)
### T9 str method in return from method
class T9:
    def __init__(self):
        self.parts = []
    def add(self, w):
        self.parts.append(w[0].upper() + w[1:])
        return self
t9 = T9()
t9.add('hi')
print(t9.parts)
