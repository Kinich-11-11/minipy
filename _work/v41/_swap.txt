### simple assign from index
a = [5, 2]
t = a[0]
print(t)
### swap with temp
a = [5, 2]
t = a[0]
a[0] = a[1]
a[1] = t
print(a)
### swap two temps
a = [3, 1, 2]
i = 0
j = 1
t = a[i]
a[i] = a[j]
a[j] = t
print(a)
### read then overwrite
a = [7, 8]
t = a[0]
a[0] = 9
print(t, a)
### assign index to two vars
a = [1, 2]
x = a[0]
y = a[0]
a[0] = 5
print(x, y)
### bubble sort
a = [5, 1, 4, 2, 8]
n = len(a)
for i in range(n):
    for j in range(n - 1 - i):
        if a[j] > a[j + 1]:
            t = a[j]
            a[j] = a[j + 1]
            a[j + 1] = t
print(a)
### selection style swap
a = [9, 4, 6]
m = 0
for i in range(1, 3):
    if a[i] < a[m]:
        m = i
t = a[0]
a[0] = a[m]
a[m] = t
print(a)
### list of lists swap
a = [[1, 2], [3, 4]]
t = a[0]
a[0] = a[1]
a[1] = t
print(a)
### dict value assign
d = {'a': 1, 'b': 2}
t = d['a']
d['a'] = d['b']
d['b'] = t
print(d)
### string swap
a = ['x', 'y']
t = a[0]
a[0] = a[1]
a[1] = t
print(a)
### tuple index read
a = (5, 2)
t = a[0]
print(t, a[1])
### slice copy then modify
a = [1, 2, 3]
b = a[0]
a[0] = 99
print(b, a)
### nested index swap
a = [[1, 2], [3, 4]]
t = a[0][1]
a[0][1] = a[1][0]
a[1][0] = t
print(a)
### augmented then read
a = [1, 2]
t = a[0]
a[0] += 10
print(t, a)
### in loop accumulate
a = [1, 2, 3]
t = a[0]
for x in a:
    if x > t:
        t = x
print(t)
### swap variables via tuple
a = 1
b = 2
a, b = b, a
print(a, b)
### reverse via temp
a = [1, 2, 3, 4]
n = len(a)
for i in range(n // 2):
    t = a[i]
    a[i] = a[n - 1 - i]
    a[n - 1 - i] = t
print(a)
