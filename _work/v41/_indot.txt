### type of dict int value
print(type({'a': 1}))
### type of dict float value
print(type({'a': 1.5}))
### type of dict string value
print(type({'a': 'x'}))
### type of dict dot string value
print(type({'a': '1.5'}))
### type of dict tuple value
print(type({'a': (1, 2)}))
### type of dict list value
print(type({'a': [1]}))
### type of empty dict
print(type({}))
### type of dict via variable
d = {'a': 1.5}
print(type(d), isinstance(d, dict))
### len of float dict
print(len({'a': 1.5}))
### list of float dict
print(list({'a': 1.5}))
### get item of float dict
d = {'a': 1.5}
print(d['a'])
### in with int value dict
print('a' in {'a': 1})
### in with float value dict
print('a' in {'a': 1.5})
### in with dot string dict
print('a' in {'a': '1.5'})
### in with float key
print(1.5 in {1.5: 'x'})
### in with float value and extra key
print('a' in {'a': 1.5, 'b': 2})
### not in with float value
print('z' not in {'a': 1.5})
### keys of float dict
print(sorted({'a': 1.5, 'b': 2}))
### values of float dict
d = {'a': 1.5}
for k, v in d.items():
    print(k, v)
### dict equality with float value
print({'a': 1.5} == {'a': 1.5})
### dict print with float
print({'a': 1.5, 'b': 2})
### float as dict value then type of item
d = {'a': 1.5}
print(type(d['a']))
### nested dict with float leaf
d = {'a': {'b': 1.5}}
print(type(d), type(d['a']))
### set with float
print(1.5 in {1.5})
### list with float
print(1.5 in [1.5])
### tuple with float
print(1.5 in (1.5,))
### string with dot
print('1.5' in '1.5')
### dict with negative value
print('a' in {'a': -1})
### dict with zero value
print('a' in {'a': 0})
### dict with scientific value
print('a' in {'a': 1e3})
