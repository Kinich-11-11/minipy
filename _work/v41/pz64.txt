// PZ64（临时调试）：逐个 "requires a list" 站点插探针，看哪个被执行、当时的 evtype/ev
patch('PZ64 sorted site probes', () => {
  const hasText = (v, needle, d) => {
    if (!v || d > 20) return false;
    if (Array.isArray(v)) {
      if ((v[0] === 10 || v[0] === 4) && typeof v[1] === 'string') return v[1].indexOf(needle) >= 0;
      return v.some((x) => hasText(x, needle, d + 1));
    }
    return false;
  };
  const deepText = (id, needle, d) => {
    if (!id || d > 25 || typeof id !== 'string') return false;
    const b = BL()[id];
    if (!b) return false;
    if (b.opcode === 'text' && String(b.fields.TEXT[0]).indexOf(needle) >= 0) return true;
    if (Object.values(b.inputs || {}).some((v) => hasText(v, needle, 0))) return true;
    return deepText(b.next, needle, d + 1);
  };
  const taken = new Set(Object.values(T(IT).variables).map((x) => x[0]));
  if (taken.has('dbg')) throw new Error('[PZ64] dbg taken');
  newVar('dbg', '');
  let n = 0;
  for (const id of allIds()) {
    const b = BL()[id];
    if (b.opcode !== 'procedures_call' || !b.mutation || b.mutation.proccode !== 'fail %s') continue;
    if (!deepText(id, 'requires a list', 0)) continue;
    n++;
    const mark = 'PZ64 #' + n + ' ';
    insertBeforeStmt(id, [
      setV('dbg', txt(mark)),
      setV('dbg', JOIN(VAR('dbg'), VAR('evtype'))),
      setV('dbg', JOIN(VAR('dbg'), txt('/'))),
      setV('dbg', JOIN(VAR('dbg'), VAR('ev'))),
      callProc('fail %s', [VAR('dbg')])
    ]);
    console.log('[PZ64] probe at site ' + n + ' = ' + id);
  }
  console.log('[PZ64] probes = ' + n);
});
