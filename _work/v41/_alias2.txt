### false alias equal texts
c = [1]
d = [1]
d.append(2)
print(c, d)
### rebind breaks alias
a = [1]
b = a
b = [9]
print(a, b)
### rebind source
a = [1]
b = a
a = [2]
print(a, b)
### two aliases mutate one
a = [1]
b = a
c = a
b.append(2)
print(a, b, c)
### alias then mutate via other
a = [1]
b = a
a.append(2)
print(a, b)
### alias of alias chain
a = [1]
b = a
c = b
c.append(3)
print(a, b, c)
### alias in function scope
def f(x):
    y = x
    y.append(1)
a = []
f(a)
print(a)
### equal dicts not aliased
p = {'k': 1}
q = {'k': 1}
q['j'] = 2
print(p, q)
### nested alias
a = [[1]]
b = a[0]
c = b
c.append(2)
print(a, b, c)
### delete element via alias
a = [1, 2]
b = a
del b[0]
print(a, b)
### string is immutable
s = 'x'
t = s
t = t + 'y'
print(s, t)
