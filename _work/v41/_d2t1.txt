class BT:
    def __init__(self, v):
        self.v = v
        self.l = None
        self.r = None
    def ins(self, v):
        if v < self.v:
            if self.l is None:
                self.l = BT(v)
            else:
                self.l.ins(v)
        else:
            if self.r is None:
                self.r = BT(v)
            else:
                self.r.ins(v)
    def inorder(self):
        out = []
        out.append(self.v)
        return out
root = BT(5)
for x in [3, 8, 1, 4, 7, 9]:
    root.ins(x)
print(root.inorder())
