// ---- P194: the bitwise operators accept negative integers (v4.121) ----
//
//   -1 | 0   -> ValueError: bitwise needs non-negative integers   (CPython: -1)
//   -2 & 3   -> same                                              (CPython: 2)
//   -1 ^ 1   -> same                                              (CPython: -2)
//   -1 << 2  -> same                                              (CPython: -4)
//   -8 >> 1  -> same                                              (CPython: -4)
//
// P25's `intCheck()` deliberately rejects a negative operand -- `bitop` walks the two operands bit
// by bit with `mod 2` / `floor(/2)`, which only describes non-negative integers.  The two guards
// (`if 任一侧 < 0 then fail`) are what the probes hit.
//
//   * `<<` / `>>` need no fix at all: the shift arm already computes `bt_r = base * 2` repeated and
//     `floor(base / 2)` repeated, which matches CPython for a negative base (`-1 << 2 = -4`,
//     `-8 >> 1 = -4`, `-3 >> 1 = -2`).  Its guard is lifted and replaced by the check that actually
//     belongs there: a negative **shift count** is CPython's `ValueError: negative shift count`.
//   * `& | ^` need two's complement.  The result of these three is the same as computing them on the
//     W-bit two's-complement representations and then reading the result back as signed, for any
//     W with `2^(W-1) > max(|a|, |b|)`.  So the arm first picks the smallest such power of two
//     (starting at 2, doubling until it exceeds `2 * max(|a|, |b|)`, capped at 2^53 so everything
//     stays exact in a double), maps a negative operand `x` to `x + 2^W`, runs the existing `bitop`
//     on the mapped pair, and maps a result `>= 2^(W-1)` back with `result - 2^W`.
//
// The arm still runs the **old** path unchanged when neither operand is negative (`bt_neg`):
// the copies `bt_a`/`bt_b` are taken either way, but the width scan and the sign fix only happen
// for the negative case, so a large non-negative result such as `(1 << 40) | 1` is untouched.
//
// Exactness: the mapping is exact while `max(|a|, |b|) < 2^52` (the sum `x + 2^W < 2^53`).
// Beyond that MiniPy's doubles have already lost the low bits, the same limit the `2**53 + 1`
// probe documents.
patch('P194 negative bitwise', () => {
  const F = (v) => VAR(v);
  const rs = inDef('rightside %s %s %s %s %s');
  const inChain = (stackId, target) => {
    let cur = stackId, n = 0;
    while (typeof cur === 'string' && cur && n++ < 5000) {
      if (cur === target) return true;
      cur = BL()[cur] && BL()[cur].next;
    }
    return false;
  };

  // ---------- (a) the two "non-negative" guards in intCheck() are lifted
  const negFail = find(rs, (b) => b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === 'fail %s'
    && JSON.stringify(b.inputs).includes('bitwise needs non-negative'));
  if (negFail.length !== 2) throw new Error('[P194] negative bitwise fails = ' + negFail.length);
  const guards = find(rs, (b) => b.opcode === 'control_if'
    && b.inputs.SUBSTACK && typeof b.inputs.SUBSTACK[1] === 'string'
    && negFail.some((f) => inChain(b.inputs.SUBSTACK[1], f)));
  if (guards.length !== 2) throw new Error('[P194] negative bitwise guards = ' + guards.length);
  for (const g of guards) replaceExpr(BL()[g].inputs.CONDITION[1], () => EQ(num(0), num(1)));
  console.log('[P194] the non-negative guard is lifted in both arms');

  // ---------- (b) a negative shift count keeps CPython's wording
  const shiftSet = find(rs, (b) => b.opcode === 'data_setvariableto' && b.fields.VARIABLE
    && b.fields.VARIABLE[0] === 'bt_r'
    && Object.values(b.inputs).some((v) => typeof v[1] === 'string' && BL()[v[1]]
      && BL()[v[1]].opcode === 'argument_reporter_string_number' && BL()[v[1]].fields.VALUE[0] === 'leftval'));
  if (shiftSet.length !== 1) throw new Error('[P194] shift base set = ' + shiftSet.length);
  insertBeforeStmt(shiftSet[0], [
    IF(LT(F('ev'), num(0)), [FAIL(txt('ValueError: negative shift count'))])
  ]);
  console.log('[P194] a negative shift count is CPython\'s ValueError');

  // ---------- (c) two's complement for `& | ^` when a side is negative
  newVar('bt_m', 0); newVar('bt_n', 0); newVar('bt_w', 2); newVar('bt_neg', 0);
  const bitCall = find(rs, (b) => b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === 'bitop %s %s %s');
  if (bitCall.length !== 1) throw new Error('[P194] bitop call = ' + bitCall.length);
  const argIds = JSON.parse(BL()[bitCall[0]].mutation.argumentids);
  insertBeforeStmt(bitCall[0], [
    setV('bt_a', argOf('leftval')), setV('bt_b', F('ev')), setV('bt_neg', num(0)),
    IF(OR(LT(F('bt_a'), num(0)), LT(F('bt_b'), num(0))), [
      setV('bt_neg', num(1)),
      // the smallest width with 2^(W-1) > max(|a|, |b|)
      setV('bt_m', F('bt_a')),
      IF(LT(F('bt_m'), num(0)), [setV('bt_m', SUB(num(0), F('bt_m')))]),
      setV('bt_n', F('bt_b')),
      IF(LT(F('bt_n'), num(0)), [setV('bt_n', SUB(num(0), F('bt_n')))]),
      IF(GT(F('bt_n'), F('bt_m')), [setV('bt_m', F('bt_n'))]),
      setV('bt_w', num(2)),
      REPEATU(OR(GT(F('bt_w'), MUL(F('bt_m'), num(2))), NOT(LT(F('bt_w'), num(9007199254740992)))), [
        setV('bt_w', MUL(F('bt_w'), num(2)))
      ]),
      IF(LT(F('bt_a'), num(0)), [setV('bt_a', ADD(F('bt_a'), F('bt_w')))]),
      IF(LT(F('bt_b'), num(0)), [setV('bt_b', ADD(F('bt_b'), F('bt_w')))])
    ])
  ]);
  BL()[bitCall[0]].inputs[argIds[1]] = B.REP(F('bt_a'), [4, '0']);
  BL()[bitCall[0]].inputs[argIds[2]] = B.REP(F('bt_b'), [4, '0']);
  spliceAfter(bitCall[0], chain([
    IF(EQ(F('bt_neg'), num(1)), [
      IF(NOT(LT(F('ev'), DIV(F('bt_w'), num(2)))), [setV('ev', SUB(F('ev'), F('bt_w')))])
    ])
  ]));
  console.log('[P194] negative & | ^ go through two\'s complement');
});
