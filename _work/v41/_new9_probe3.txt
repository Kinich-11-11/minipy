    def __init__(self):
        self.p = []
    def add(self, s):
        self.p.append(s)
        return self
    def t(self):
        return '-'.join(self.p)

### probe bst 6 nodes single print
class BT:
    def __init__(self, v):
        self.v = v
        self.l = None
        self.r = None
    def ins(self, v):
        if v < self.v:
            if self.l is None:
                self.l = BT(v)
            else:
                self.l.ins(v)
        else:
            if self.r is None:
                self.r = BT(v)
            else:
                self.r.ins(v)
    def inorder(self):
        out = []
        if self.l is not None:
            out.extend(self.l.inorder())
        out.append(self.v)
        if self.r is not None:
            out.extend(self.r.inorder())
        return out
root = BT(5)
for x in [3, 8, 1, 4, 7, 9]:
    root.ins(x)
print(root.inorder())

class BT:
    def __init__(self, v):
        self.v = v
        self.l = None
        self.r = None
    def ins(self, v):
        if v < self.v:
            if self.l is None:
                self.l = BT(v)
            else:
                self.l.ins(v)
        else:
            if self.r is None:
                self.r = BT(v)
            else:
                self.r.ins(v)
    def vals(self):
        out = [self.v]
        if self.l is not None:
            out.extend(self.l.vals())
        if self.r is not None:
            out.extend(self.r.vals())
        return out
root = BT(5)
for x in [3, 8, 1]:
    root.ins(x)
print(root.vals())
