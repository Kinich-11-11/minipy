### if True and False
if True:
    print('t')
else:
    print('f')
if False:
    print('t')
else:
    print('f')
### not on bools
print(not True, not False)
### bool of bools
print(bool(True), bool(False))
### and or with bools
print(True and 1, False or 2, False and 1, True or 2)
### type of function and generator
f = lambda: 1
print(type(f), type(lambda: 2))
g = (x for x in [])
print(type(g))
### truthiness of function
def fn():
    pass
if fn:
    print('fn truthy')
else:
    print('fn falsy')
### truthiness of generator expr empty
e = (x for x in [])
if e:
    print('gen truthy')
else:
    print('gen falsy')
### truthiness of class
class C:
    pass
if C:
    print('cls truthy')
else:
    print('cls falsy')
### truthiness of instance
c = C()
if c:
    print('inst truthy')
else:
    print('inst falsy')
