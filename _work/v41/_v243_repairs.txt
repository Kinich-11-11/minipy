### eg_custom_new
class Mine(ExceptionGroup):
    def __new__(cls,errors,code):
        self=super().__new__(cls,'custom',errors)
        self.code=code
        return self
    def __init__(self,errors,code):
        pass
    def derive(self,errors):
        return Mine(errors,self.code)
g=Mine([ValueError('v'),TypeError('t')],123)

print(type(a).__name__,a.code,type(b).__name__,b.code)
