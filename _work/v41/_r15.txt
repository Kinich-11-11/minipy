### e1 original mutated
class V:
    def __init__(self, x): self.x = x
    def __add__(self, o): return V(self.x + o.x)
a = V(1)
b = V(2)
r = a + b
print(a.x, b.x)
### e2 method assigns then returns self
class Q:
    def __init__(self, x): self.x = x
    def __add__(self, o):
        self.y = 9
        return self
q = Q(1)
r = q + Q(2)
print(r.y, r.x)
### e3 plain method new object
class P:
    def __init__(self, x): self.x = x
    def mk(self, n): return P(n)
print(P(1).mk(7).x)
