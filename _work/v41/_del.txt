### del slice step even
a = [1, 2, 3, 4, 5]
del a[::2]
print(a)
### del slice step offset
a = [1, 2, 3, 4, 5, 6, 7]
del a[1::3]
print(a)
### del slice negative step
a = [1, 2, 3, 4, 5]
del a[::-1]
print(a)
### del slice negative step 2
a = [1, 2, 3, 4, 5]
del a[::-2]
print(a)
### del slice plain
a = [1, 2, 3, 4, 5]
del a[1:3]
print(a)
### del single index
a = [1, 2, 3]
del a[1]
print(a)
### del slice empty result
a = [1, 2, 3]
del a[:]
print(a)
### del slice step zero caught
a = [1, 2, 3]
try:
    del a[::0]
except ValueError:
    print('caught')
### del slice all with step
a = [1, 2, 3, 4]
del a[0:4:1]
print(a)
### del slice step from end
a = [1, 2, 3, 4, 5, 6]
del a[4::2]
print(a)
### del dict key still works
d = {'a': 1, 'b': 2}
del d['a']
print(d)
### del nested list element
a = [[1, 2], [3, 4], [5, 6]]
del a[::2]
print(a)
### del slice then reuse
a = [1, 2, 3, 4]
del a[::2]
print(a, len(a), a[0])
