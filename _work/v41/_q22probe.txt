def fl2(x, tag):
    out = []
    print('enter', tag, x)
    for item in x:
        if isinstance(item, list):
            inner = fl2(item, tag + 1)
            print('gotinner', tag, inner)
            for sub in inner:
                out.append(sub)
        else:
            out.append(item)
        print('outnow', tag, out)
    return out
print(fl2([1, [2, [3, 4]], 5], 0))
