### generator basic
def gen():
    yield 1
    yield 2
    yield 3
g = gen()
print(list(g))
### generator next and exhaustion
def gen2():
    yield 'a'
    yield 'b'
g = gen2()
print(next(g), next(g))
try:
    print(next(g))
except StopIteration:
    print('stop')
### generator in for loop
def counter(n):
    i = 0
    while i < n:
        yield i
        i += 1
print([x for x in counter(4)])
for v in counter(3):
    print('v', v)
### generator reused after exhaustion
def g3():
    yield 1
g = g3()
print(list(g), list(g))

### generator inside sum any all

### generator short circuit
def side(x):
    print('s', x)
    return x
print(any(side(x) > 1 for x in [0, 1, 2, 3]))
print(all(side(x) < 2 for x in [0, 1, 2, 3]))
### generator with tuple yield
def pairs():
    yield 1, 2
    yield 3, 4
print(list(pairs()))
for a, b in pairs():
    print(a + b)
### generator with argument and closure
def repeat(v, n):
    i = 0
    while i < n:
        yield v
        i += 1
print(list(repeat('x', 3)))
### generator that consumes another
def counter(n):
    i = 0
    while i < n:
        yield i
        i += 1
def double(src):
    for v in src:
        yield v * 2
print(list(double(counter(4))))
### generator return value stops iteration
def gr():
    yield 1
    return
    yield 2
print(list(gr()))
### yield inside condition and loop
def evens(n):
    for i in range(n):
        if i % 2 == 0:
            yield i
print(list(evens(7)))
### nested generator functions
def outer():
    def inner():
        yield 1
        yield 2
    for v in inner():
        yield v * 10
print(list(outer()))
### generator exception propagation
def bad():
    yield 1
    raise ValueError('gen')
g = bad()
print(next(g))
try:
    next(g)
except ValueError as e:
    print('gen err', e)
### yield from if supported
def inner2():
    yield 1
    yield 2
def outer2():
    yield from inner2()
    yield 3
try:
    print(list(outer2()))
except Exception as e:
    print('yield from unsupported')

### iterator protocol on custom class
class It:
    def __init__(self):
        self.n = 0
    def __iter__(self):
        return self
    def __next__(self):
        self.n += 1
        if self.n > 3:
            raise StopIteration
        return self.n
print(list(It()))
