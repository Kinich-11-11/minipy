### matrix trap
m = [[0] * 3] * 3
m[0][1] = 5
print(m)
### list mult append
a = [[0]] * 3
a[0].append(1)
print(a)
### comp rows distinct (no group)
m = [[0] * 3 for _ in range(3)]
m[0][1] = 5
print(m)
### rebind drops group
m = [[0]] * 2
m = [1, 2]
m[0] = 9
print(m)
### plus no false group
x = [1, 2]
y = x + [[0]] * 1
print(y)
