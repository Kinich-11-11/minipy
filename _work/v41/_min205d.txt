### Q16 add dunder temp var workaround
class W6:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    def __add__(self, o):
        nx = self.x + o.x

### Q22 isinstance nested
def fl2(x):
    out = []
    for item in x:
        if isinstance(item, list):
            inner = fl2(item)
            for sub in inner:
                out.append(sub)
        else:
            out.append(item)
    return out
print(fl2([1, [2, [3, 4]], 5]))
