### shape a bare map
r = map(lambda x: x + 1, [1, 2])
print(r)
### shape b list of map str
print(list(map(str, [1, 2])))
### shape c lambda as second arg
print(list(map(str, [1, 2])) , (lambda x: x)(5))
### shape d assign lambda then map
f = lambda x: x + 1
print(list(map(f, [1, 2])))
### shape e nested call lambda
print(sum(map(lambda x: x * 2, [1, 2])))
