### chained comparison evaluated once
def f(x):
    print('eval', x)
    return x
print(f(1) < f(2) < f(3))
print(f(3) < f(2) < f(1))
### set of tuples and strings
print(sorted({(1, 2), (0, 3)}), sorted(set('hello')), len({(1, 2), (1, 2)}))
### bool in arithmetic
print(True + 1, True * 3, int(True))
### float of string with sign
print(float('-1.5'), float('+1.5'))
### int of bool and float
print(int(True), int(-0.9), int(2.9))
### append in recursion post order
def g(n, out):
    if n == 0:
        return
    g(n - 1, out)
    out.append(n)
out = []
g(3, out)
print(out)
