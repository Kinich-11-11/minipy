### generator yield squares
def squares(n):
    i = 0
    while i < n:
        yield i * i
        i += 1
print(list(squares(6)))
g = squares(3)
print(next(g), next(g), next(g))
### generator fibonacci
def fibgen():
    a = 0
    b = 1
    while True:
        yield a
        a, b = b, a + b
out = []
fg = fibgen()
for _ in range(10):
    out.append(next(fg))
print(out)
