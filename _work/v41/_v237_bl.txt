class Count:
    def __init__(self,n):
        self.i = 0
        self.n = n
    def __iter__(self):
        return self
    def __next__(self):
        if self.i >= self.n:
            raise StopIteration
        self.i += 1
        return self.i
print(sum(Count(4)))

class Student:
    def __init__(self,name,scores):
        self.name = name
        self.scores = scores
    def average(self):
        return sum(self.scores)/len(self.scores)
a = [Student("Ann",[90,80]),Student("Bob",[75,99])]
a.sort(key=lambda s:s.average(),reverse=True)
print(a[0].name,round(a[0].average(),1))

### R2_E01
class Shop:
    def __init__(self):
        self.stock = {"a":6,"b":4}
        self.price = {"a":20,"b":35}
        self.total = 0
    def buy(self,name,count):
        if count>self.stock[name]:
            raise ValueError("stock")
        self.stock[name] -= count
        cost = count*self.price[name]
        if cost>=100:
            cost -= 10
        self.total += cost
shop = Shop()
for name,n in [("a",5),("b",2),("a",3)]:
    try:
        shop.buy(name,n)
    except ValueError:
        print("rejected")
print(shop.total)
print(shop.stock["a"],shop.stock["b"])

### R3_P03
def matmul(a,b):
    out = []
    for i in range(len(a)):
        row = []
        for j in range(len(b[0])):
            s=0
            for k in range(len(b)):
                s += a[i][k]*b[k][j]
            row.append(s)
        out.append(row)
    return out
def transpose(a):
    return [[a[i][j] for i in range(len(a))] for j in range(len(a[0]))]
a = [[1,2,3],[4,5,6]]
b = transpose(a)
c = matmul(a,b)
print(c[0])
print(c[1])

### R3_P07
class School:
    def __init__(self):
        self.students = {}
    def add(self,name,score):
        self.students[name] = score
    def remove(self,name):
        del self.students[name]
    def top(self):
        return sorted(self.students,key=lambda n:self.students[n],reverse=True)
s = School()
s.add("Ann",90)
s.add("Bob",80)
s.add("Cat",95)
s.add("Bob",99)
s.remove("Ann")
print(s.top())
print(len(s.students))
