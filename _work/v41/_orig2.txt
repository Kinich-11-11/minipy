### docstring one line
def f():
    """doc"""
    return 1

print(f())
### docstring multi line
def g():
    """line one
    line two
    """
    return 2

print(g())
### plain multi line string
s = """a
b"""
print(s)
### comment only
# a comment
x = 1
print(x)
### chained subscript assign
d = {}
d["a"] = {}
d["a"]["b"] = 5
print(d)
### chained subscript read
d = {"a": {"b": 7}}
print(d["a"]["b"])
### list in dict assign
m = {"k": [1, 2]}
m["k"][0] = 9
print(m)
### nested dict literal then assign
c = {"x": {"y": 1}}
c["x"]["y"] = c["x"]["y"] + 10
print(c)
