// ---- P87: `import math as m` and the module constants math.pi / math.e (E52/E53) ----
//
// `import math as m` was accepted (Tok[2] is still "math") but the alias was thrown away, so
// `m.floor(2.5)` raised NameError.  And `math.pi` never even reached the method dispatch:
// eval's attribute branch demands an instance and `math` is not a variable.
//
// The module name now lives in `mthname` (default "math"): the import branch records the
// alias, methodcall's math dispatch accepts either spelling, and eval answers the constants
// before the attribute lookup.
patch('P87 math alias and constants', () => {
  const litOf = (d) => {
    if (!d) return null;
    if (Array.isArray(d) && (d[0] === 10 || d[0] === 4) && typeof d[1] !== 'object') return String(d[1]);
    if (Array.isArray(d[1]) && (d[1][0] === 10 || d[1][0] === 4)) return String(d[1][1]);
    return null;
  };
  const blk = (d) => (d && typeof d[1] === 'string') ? BL()[d[1]] : null;
  const argName = (d) => { const b = blk(d); return b && b.opcode === 'argument_reporter_string_number' ? b.fields.VALUE[0] : null; };
  const varName = (d) => (Array.isArray(d) && Array.isArray(d[1]) && d[1][0] === 12) ? d[1][1] : null;
  const STOPTHIS = () => I('control_stop', { fields: { STOP_OPTION: ['this script', null] }, mutation: { tagName: 'mutation', children: [], hasnext: 'false' } });
  const taken = new Set(Object.values(T(IT).variables).map((x) => x[0]));
  if (taken.has('mthname')) throw new Error('[P87] variable name taken: mthname');
  newVar('mthname', 'math');

  // (a) the import branch keeps the alias
  const ex = inDef('execute');
  const hasmath = find(ex, (b) => b.opcode === 'data_setvariableto' && b.fields.VARIABLE && b.fields.VARIABLE[0] === 'hasmath' && litOf(b.inputs.VALUE) === '1');
  if (hasmath.length !== 1) throw new Error('[P87] hasmath set = ' + hasmath.length);
  spliceAfter(hasmath[0], chain([
    setV('mthname', txt('math')),
    IF(EQ(item('Tok', num(3)), txt('as')), [setV('mthname', item('Tok', num(4)))])
  ]));
  console.log('[P87] import math as NAME recorded');

  // (b) methodcall's math dispatch accepts the alias
  const mc = inDef('methodcall %s %s');
  const mathEq = find(mc, (b) => b.opcode === 'operator_equals'
    && ((blk(b.inputs.OPERAND1) && blk(b.inputs.OPERAND1).opcode === 'data_itemoflist' && blk(b.inputs.OPERAND1).fields.LIST && blk(b.inputs.OPERAND1).fields.LIST[0] === 'Tok' && argName(blk(b.inputs.OPERAND1).inputs.INDEX) === 'lo' && litOf(b.inputs.OPERAND2) === 'math')));
  if (mathEq.length !== 1) throw new Error('[P87] math dispatch test = ' + mathEq.length);
  replaceExpr(mathEq[0], () => OR(mathEq[0], EQ(item('Tok', argOf('lo')), VAR('mthname'))));
  console.log('[P87] math dispatch also answers the alias');

  // (c) eval: math.pi / math.e
  const ev = inDef('eval %s %s');
  const tokAt = (off, lit) => I('operator_equals', { inputs: { OPERAND1: B.REP(item('Tok', ADD(argOf('lo'), num(off))), [10, '']), OPERAND2: B.REP(txt(lit), [10, '']) } });
  const attrIf = find(ev, (b) => {
    if (b.opcode !== 'control_if' && b.opcode !== 'control_if_else') return false;
    const c = b.inputs.CONDITION && b.inputs.CONDITION[1];
    if (typeof c !== 'string') return false;
    const a = BL()[c];
    if (!a || a.opcode !== 'operator_and') return false;
    const eqAt = (d, off, lit) => {
      const e = blk(d);
      if (!e || e.opcode !== 'operator_equals') return false;
      const t = (x) => {
        const b = blk(x);
        if (!b || b.opcode !== 'data_itemoflist') return false;
        if (!b.fields.LIST || b.fields.LIST[0] !== 'Tok') return false;
        const ix = blk(b.inputs.INDEX);
        return !!ix && ix.opcode === 'operator_add' && argName(ix.inputs.NUM1) === 'lo' && litOf(ix.inputs.NUM2) === String(off);
      };
      return (t(e.inputs.OPERAND1) && litOf(e.inputs.OPERAND2) === lit);
    };
    const eqHi = (d) => { const e = blk(d); return !!e && e.opcode === 'operator_equals' && argName(e.inputs.OPERAND1) === 'hi'; };
    return (eqAt(a.inputs.OPERAND1, 1, '.') && eqHi(a.inputs.OPERAND2)) || (eqAt(a.inputs.OPERAND2, 1, '.') && eqHi(a.inputs.OPERAND1));
  });
  if (attrIf.length !== 1) throw new Error('[P87] eval attribute branch = ' + attrIf.length);
  const constSet = (v) => [setV('ev', num(v)), setV('evtype', txt('N')), chgV('fnlimit', num(-1)), STOPTHIS()];
  const guard = AND(AND(tokAt(1, '.'), EQ(argOf('hi'), ADD(argOf('lo'), num(2)))), AND(EQ(item('Tok', argOf('lo')), VAR('mthname')), EQ(VAR('hasmath'), num(1))));
  insertBeforeStmt(attrIf[0], [
    IF(guard, [
      IFELSE(EQ(item('Tok', ADD(argOf('lo'), num(2))), txt('pi')), constSet('3.141592653589793'), [
        IF(EQ(item('Tok', ADD(argOf('lo'), num(2))), txt('e')), constSet('2.718281828459045'))
      ])
    ])
  ]);
  console.log('[P87] math.pi / math.e answered');
});
