### Z1 plain twoarg ctor
class A:
    def __init__(self, p, q):
        self.p = p
        self.q = q
z = A(5, 6)
print(z.p, z.q)
### Z2 ctor in method
class B:
    def __init__(self, p, q):
        self.p = p
        self.q = q
    def mk(self, k):
        return B(self.p + k, self.q + k)
print(B(1, 2).mk(10).p)
### Z3 ctor one arg in method
class C:
    def __init__(self, p):
        self.p = p
    def mk(self, k):
        return C(self.p + k)
print(C(1).mk(10).p)
