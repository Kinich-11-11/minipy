### w1 pow one arg
try:
    print(pow(2))
except TypeError as e:
    print('TypeError')
### w2 pow two args
print(pow(2, 3))
### w3 pow float
print(pow(2.0, 0.5))
### w4 pow three args
print(pow(2, 3, 5))
### w5 pow negative exp
print(pow(2, -1))
### w6 pow mod zero
try:
    print(pow(2, 3, 0))
except ValueError as e:
    print('ValueError')
### w7 pow no args
try:
    print(pow())
except TypeError as e:
    print('TypeError2')
### w8 pow kwargs
try:
    print(pow(2, 3, mod=5))
except TypeError as e:
    print('TypeError3')
