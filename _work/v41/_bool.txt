### bool arithmetic
print(1 + True, True + 1, True + True, True * 3)
### bool subtraction
print(3 - True, True - 1)
### bool sum
print(sum([True, True]))
### bool sum mixed
print(sum([True, 1]))
### bool abs
print(abs(True), abs(False))
### bool int conversion
print(int(True), int(False), float(True))
### bool comparison
print(True == 1, False == 0)
### bool in arithmetic chain
x = True
print(x + x)
