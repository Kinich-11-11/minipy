// ---- P415: 对象显示回退到 `__repr__`（文档 #2；保守版，不用 stop this script）----
//
// 根因（第 155 轮）：`strval %s %s`（形参 vv, vt）在 `vt == "O"` 时直接产出
//   `<` + 类名 + ` object>`，**缺少 CPython 的"无 `__str__` 就回退 `__repr__`"这一步**。
//   ⇒ 顶层 `print(Thing(3))` 与 `{'k': Thing(3)}` 都错（定义 `__str__` 的对象已正确，不受影响）。
//
// 上一版失败原因：在 `strval` 入口插入并用 `stop this script` 收尾 ⇒ **所有打印都空**（stop 影响面过大）；
//   另 `callfn(vv,1,0)` 也无效（它的后两参是 token 区间）。
//
// 本版（保守）：
//   * 定位 `strval` 里那条"产出 `<X object>`"的赋值（`data_setvariableto ev`，其值 JOIN 含 ` object>`）；
//   * **在它之后**追加一段：
//         IF(vt == "O" AND listcontainsitem(FnName, JOIN(InstCls[vv], ".__repr__"))):
//             callmeth(vv, "__repr__", 1, 0)      ← 用方法返回值覆盖 ev
//     不 stop、不改入口 ⇒ 只覆盖"兜底文本"，影响面最小。
patch('P415 __repr__ fallback after strval fallback', () => {
  const BB = T(IT).blocks;
  const defId = defIdOf('strval %s %s');
  let p0 = 'vv', p1 = 'vt';
  for (const id of Object.keys(BB)) {
    const b = BB[id];
    if (b.opcode !== 'procedures_prototype' || !b.mutation) continue;
    if (String(b.mutation.proccode) !== 'strval %s %s') continue;
    const names = JSON.parse(b.mutation.argumentnames);
    p0 = names[0]; p1 = names[1];
    break;
  }
  // 收集子树文字，用于识别"含 object> 的赋值"
  const texts = (q) => {
    const out = [];
    if (q.mutation && q.mutation.proccode) out.push(String(q.mutation.proccode));
    for (const k of Object.keys(q.fields || {})) { const f = q.fields[k]; if (Array.isArray(f) && typeof f[0] === 'string') out.push(f[0]); }
    const rec = (v) => {
      if (!Array.isArray(v)) return;
      if ((v[0] === 10 || v[0] === 4) && typeof v[1] !== 'object') out.push(String(v[1]));
      else if (typeof v[1] === 'string' && BB[v[1]]) { const w = BB[v[1]]; if (w.inputs) for (const x of Object.values(w.inputs)) rec(x); }
      else if (Array.isArray(v[1])) rec(v[1]);
    };
    for (const x of Object.values(q.inputs || {})) rec(x);
    return out.join('|');
  };
  let fallback = null;
  const walk = (id) => {
    let c = id;
    while (c) {
      const q = BB[c];
      if (!q) break;
      if (q.opcode === 'data_setvariableto' && q.fields && q.fields.VARIABLE && q.fields.VARIABLE[0] === 'ev'
          && texts(q).indexOf(' object>') >= 0) { fallback = c; }
      for (const [k, v] of Object.entries(q.inputs || {})) {
        if ((k === 'SUBSTACK' || k === 'SUBSTACK2') && typeof v[1] === 'string') walk(v[1]);
      }
      c = q.next;
    }
  };
  walk(BB[defId].next);
  if (!fallback) throw new Error('[P415] `<X object>` fallback assignment not found in strval');
  console.log('[P415] fallback node=' + fallback + ' params=' + p0 + ',' + p1);
  const hasRepr = I('data_listcontainsitem', {
    fields: B.LISTF('FnName'),
    inputs: { ITEM: B.REP(JOIN(item('InstCls', argOf(p0)), txt('.__repr__')), [10, '']) }
  });
  spliceAfter(fallback, chain([
    IF(EQ(argOf(p1), txt('O')), [
      IF(hasRepr, [
        callProc('callmeth %s %s %s %s', [argOf(p0), txt('__repr__'), num(1), num(0)])
      ])
    ])
  ]));
  console.log('[P415] __repr__ overwrite installed after the fallback');
});
