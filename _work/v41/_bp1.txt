### sorted key battery
words = ['pear', 'fig', 'apple', 'kiwi']
print(sorted(words), sorted(words, key=len), sorted(words, key=len, reverse=True))
pairs = [(1, 'b'), (2, 'a'), (1, 'a')]
print(sorted(pairs))
print(sorted(words, key=lambda w: w[-1]))
### decorator basic
def double_result(f):
    def wrapper(n):
        return f(n) * 2
    return wrapper
@double_result
def val(n):
    return n + 1
print(val(5), val(0))
