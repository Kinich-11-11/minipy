### dispatch table lambdas
ops = {'add': lambda a, b: a + b, 'sub': lambda a, b: a - b, 'mul': lambda a, b: a * b}
def calc(op, a, b):
    if op in ops:
        return ops[op](a, b)
    return None
print(calc('add', 3, 4), calc('mul', 3, 4), calc('sub', 3, 4), calc('div', 3, 4))
### decorator basic
def double_result(f):
    def wrapper(n):
        return f(n) * 2
    return wrapper
@double_result
def val(n):
    return n + 1
print(val(5), val(0))
