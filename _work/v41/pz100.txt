// PZ100（临时调试）：给**每一个** `eval %s %s` 调用点编号并打标记
// （第 95 轮的教训：不要只打外层 lo/hi，也不要用实参名字挑站点；全打，然后看"进入坏 eval 之前最后一个标记"）
patch('PZ100 all eval call-site markers', () => {
  const BB = T(IT).blocks;
  const ownerOf = (id) => {
    let cur = id;
    for (let i = 0; i < 400 && cur; i++) {
      const q = BB[cur];
      if (!q) return '?';
      if (q.opcode === 'procedures_definition') {
        const pr = BB[q.inputs.custom_block[1]];
        return pr && pr.mutation ? String(pr.mutation.proccode) : '?';
      }
      cur = q.parent;
    }
    return '(toplevel)';
  };
  let k = 0;
  const rows = [];
  for (const id of Object.keys(BB)) {
    const b = BB[id];
    if (b.opcode !== 'procedures_call' || !b.mutation) continue;
    if (String(b.mutation.proccode) !== 'eval %s %s') continue;
    k++;
    const tag = 'E' + k;
    try {
      insertBeforeStmt(id, [
        callProc('log %s', [JOIN(txt('EID ' + id + ' in ' + ownerOf(id) + ' '), JOIN(VAR('head'), JOIN(txt('/'), VAR('count'))))])
      ]);
      rows.push(tag);
    } catch (e) {
      rows.push(tag + '(SKIP)');
    }
  }
  if (!k) throw new Error('[PZ100] no eval call sites');
  console.log('[PZ100] marked ' + k + ' sites');
});
