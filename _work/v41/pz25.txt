// PZ25（临时调试）：在 tuplelit 递归 eval(tqlo, tqhi) 之前打印它切出的元素区间
patch('PZ25 tuplelit ranges', () => {
  newVar('dbg', '');
  const nameOf = (d) => {
    if (!Array.isArray(d)) return null;
    if (d[0] === 12) return d[1];
    if ((d[0] === 1 || d[0] === 2 || d[0] === 3) && Array.isArray(d[1])) return nameOf(d[1]);
    const id = typeof d === 'string' ? d : (typeof d[1] === 'string' ? d[1] : null);
    const b = id ? BL()[id] : null;
    if (!b) return null;
    if (b.opcode === 'data_variable') return b.fields.VARIABLE[0];
    if (b.opcode === 'argument_reporter_string_number') return b.fields.VALUE[0];
    return null;
  };
  const tl = reach(defIdOf('tuplelit %s %s'));
  const calls = find(tl, (b) => b.opcode === 'procedures_call' && b.mutation.proccode === 'eval %s %s'
    && Object.values(b.inputs).some((v) => nameOf(v) === 'tqlo')
    && Object.values(b.inputs).some((v) => nameOf(v) === 'tqhi'));
  if (!calls.length) throw new Error('[PZ25] no tuplelit element eval call');
  for (const c of calls) {
    insertBeforeStmt(c, [
      setV('dbg', txt('TQ ')),
      setV('dbg', JOIN(VAR('dbg'), VAR('tqlo'))),
      setV('dbg', JOIN(VAR('dbg'), txt('..'))),
      setV('dbg', JOIN(VAR('dbg'), VAR('tqhi'))),
      setV('dbg', JOIN(VAR('dbg'), txt(' cut='))),
      setV('dbg', JOIN(VAR('dbg'), VAR('tqek'))),
      setV('dbg', JOIN(VAR('dbg'), txt(' dep='))),
      setV('dbg', JOIN(VAR('dbg'), VAR('tqed'))),
      setV('dbg', JOIN(VAR('dbg'), txt(' tok='))),
      setV('dbg', JOIN(VAR('dbg'), VAR('tqeop'))),
      setV('dbg', JOIN(VAR('dbg'), txt(' kind='))),
      setV('dbg', JOIN(VAR('dbg'), VAR('tqeopK'))),
      callProc('fail %s', [VAR('dbg')])
    ]);
  }
  console.log('[PZ25] instrumented ' + calls.length + ' tuplelit element eval calls');
});
