### set pop single
s = {1, 2}
print(s.pop())
print(sorted(s))
### set pop from generator
s = {1, 2, 3}
print(sorted(s.pop() for _ in range(2)))
print(sorted(s))
### set pop empty caught
s = set()
try:
    s.pop()
except KeyError as e:
    print('ke', e)
### set pop all
s = {5}
print(s.pop())
print(len(s))
### set methods full
s = {1, 2, 3}
s.add(4)
s.discard(1)
s.discard(99)
print(sorted(s))
s.remove(2)
print(sorted(s))
try:
    s.remove(99)
except KeyError as e:
    print('ke')
s.update([7, 8])
print(sorted(s))
print(sorted(s.pop() for _ in range(2)))
### set pop via variable receiver
s = {'a', 'b'}
x = s.pop()
print(x in ['a', 'b'])
print(len(s))
### float inf

### float neg inf
print(float('-inf'))

### float nan
print(float('nan'))
### float nan signed
print(float('-nan'), float('+NaN'))
### float normal still works
print(float('2.5'), float('3'), float('-0.5'))

    int('inf')
except ValueError as e:
    print('ve')

    float('abc')
