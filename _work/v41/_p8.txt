### t1 implicit concat str
print('a' 'b')
### t2 trailing comma list
print([1, 2,])
### t3 trailing comma dict
print({'a': 1,})
### t4 trailing comma call
def f(x):
    return x
print(f(1,))
### t5 trailing comma tuple
print((1,))
### t6 implicit concat with var
x = 'b'
print('a' 'b')
### t7 pow two args
print(pow(2, 3))
### t8 max default list
print(max([1, 2], default=0))
### t9 min empty default
print(min([], default=-1))
### t10 str as value
print(str)
