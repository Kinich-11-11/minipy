    def add():

    add()

### nonlocal keyword
def o():
    n = 0
    def i():
        nonlocal n
        n += 1
        return n
    return i

### nested generator yields
def inner():
    yield 1
    yield 2
def outer():
    for v in inner():
        yield v * 10
print(list(outer()))
