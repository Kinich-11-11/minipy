### cr length and index
s = 'a\rb'
print(len(s), s[0], s[1] == '\r', s[2])
### cr find b
s = 'a\rb'
print(s.find('b'), s.find('a'))
### cr find cr
s = 'a\rb'
print(s.find('\r'))
### cr count b
s = 'a\rb'
print(s.count('b'), s.count('a'))
### cr count cr
s = 'a\rb'
print(s.count('\r'))
### cr in
s = 'a\rb'
print('\r' in s, 'x' in s)
### cr equality
print('\r' == '\r', '\r' == 'r')
### cr join
print(repr('a' + '\r' + 'b'))
### cr split on cr
print('a\rb'.split('\r'))
### cr strip
print(repr('\ra\r'.strip()))
### cr upper
print(repr('a\rb'.upper()))
### cr replace
print(repr('a\rb'.replace('\r', '-')))
### tab find
print('a\tb'.find('\t'), 'a\tb'.count('\t'))
### tab split
print('a\tb'.split('\t'))
### tab index
s = 'a\tb'
print(len(s), s[1] == '\t')
### cr slicing
s = 'a\rb'
print(repr(s[1:2]), repr(s[0:3]))
### cr startswith
print('a\rb'.startswith('a'), 'a\rb'.endswith('b'))
