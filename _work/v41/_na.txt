### b1 nested append returns out
def build2(out):
    def add(v):
        out.append(v)
        return out
    return add(1)
print(build2([]))
### b2 nested append no return
def build2b(out):
    def add(v):
        out.append(v)
    add(1)
    return out
print(build2b([]))
### b3 nested appends then outer returns
def build2c():
    out = []
    def add(v):
        out.append(v)
    add(1)
    add(2)
    return out
print(build2c())
### b4 closure counter then return
def make():
    out = []
    def add(v):
        out.append(v)
        return out
    add(9)
    return out
print(make())
