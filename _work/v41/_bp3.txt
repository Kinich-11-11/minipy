### function factory compose
def multiplier(k):
    def apply(x):
        return x * k
    return apply
def compose(f, g):
    return lambda x: f(g(x))
double = multiplier(2)
triple = multiplier(3)
both = compose(double, triple)
print(double(5), triple(5), both(5), multiplier(10)(4))
### decorator basic
def double_result(f):
    def wrapper(n):
        return f(n) * 2
    return wrapper
@double_result
def val(n):
    return n + 1
print(val(5), val(0))
