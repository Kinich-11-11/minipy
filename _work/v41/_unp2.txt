### leading bracket nested target
(a, b), c = (1, 2), 3
print(a, b, c)
### leading bracket both sides
(a, b), (c, d) = (1, 2), (3, 4)
print(a, b, c, d)
### paren tuple simple target
(a, b) = (1, 2)
print(a, b)
### bracket list target
[a, b] = (1, 2)
print(a, b)
### parenthesized expression statement
(1 + 2)
print('ok')
### paren expression with comma not assignment
(a, b)
print('ok2')
### leading bracket in for
for (a, b), c in [((1, 2), 3)]:
    print(a, b, c)
### flat multi target still works
a, b = 1, 2
print(a, b)
### starred flat target still works
a, *rest = [1, 2, 3]
print(a, rest)
### nested assignment via index
d = {}
(d['a'], d['b']) = (1, 2)
print(d)
### paren target with function values
def f():
    return 7, 8
(x, y) = f()
print(x, y)
### swap still works
a, b = 1, 2
a, b = b, a
print(a, b)
### nested unpack with list literal
[(a, b), c] = [(1, 2), 3]
print(a, b, c)
