### q1 print object top
class Thing:
    def __init__(self, n): self.n = n
    def __repr__(self): return 'Thing!' + str(self.n)
print(Thing(3))
### q2 print object with str
class Other:
    def __init__(self, n): self.n = n
    def __str__(self): return 'S' + str(self.n)
    def __repr__(self): return 'R' + str(self.n)
print(Other(1))
