// ---- P78: min/max over a single container, and int(x, base) (E22/E25) ----
//
// `min(a)` / `max(a)` only existed in the multi-argument form (minmax), so the one-argument
// call fell through callfn to "NameError: function min".  `int('101', 2)` was rejected by
// eval's two-argument whitelist (max/min/pow/round only) before it ever reached builtin2b.
patch('P78 minmax container and int base', () => {
  const litOf = (d) => {
    if (!d) return null;
    if (Array.isArray(d) && (d[0] === 10 || d[0] === 4) && typeof d[1] !== 'object') return String(d[1]);
    if (Array.isArray(d[1]) && (d[1][0] === 10 || d[1][0] === 4)) return String(d[1][1]);
    return null;
  };
  const blk = (d) => (d && typeof d[1] === 'string') ? BL()[d[1]] : null;
  const argName = (d) => { const b = blk(d); return b && b.opcode === 'argument_reporter_string_number' ? b.fields.VALUE[0] : null; };
  const varName = (d) => (Array.isArray(d) && Array.isArray(d[1]) && d[1][0] === 12) ? d[1][1] : null;
  const isArgEq = (d, name, lit) => { const b = blk(d); return !!b && b.opcode === 'operator_equals' && ((argName(b.inputs.OPERAND1) === name && litOf(b.inputs.OPERAND2) === lit) || (argName(b.inputs.OPERAND2) === name && litOf(b.inputs.OPERAND1) === lit)); };
  const isTokLo = (x) => { const b = blk(x); return !!b && b.opcode === 'data_itemoflist' && b.fields.LIST && b.fields.LIST[0] === 'Tok'; };
  const taken = new Set(Object.values(T(IT).variables).map((x) => x[0]));
  for (const n of ['mmv', 'mmt', 'mmi']) if (taken.has(n)) throw new Error('[P78] variable name taken: ' + n);
  for (const v of ['mmv', 'mmt', 'mmi']) newVar(v, 0);

  // ---------- (a) callfn: min/max over one iterable argument
  const cf = inDef('callfn %s %s %s');
  const sortedIf = find(cf, (b) => b.opcode === 'control_if' && b.inputs.CONDITION && isArgEq(b.inputs.CONDITION, 'fn', 'sorted'));
  if (sortedIf.length !== 1) throw new Error('[P78] sorted branch = ' + sortedIf.length);
  const isIter = OR(EQ(VAR('evtype'), txt('L')), OR(EQ(VAR('evtype'), txt('T')), EQ(VAR('evtype'), txt('G'))));
  const minmaxOne = IF(OR(EQ(argOf('fn'), txt('min')), EQ(argOf('fn'), txt('max'))), [
    IF(NOT(isIter), [FAILB(JOIN(txt("TypeError: '"), JOIN(argOf('fn'), txt("' object is not iterable"))))]),
    setV('ltmp', VAR('ev')),
    callProc('listlen %s', [VAR('ltmp')]),
    setV('ln', VAR('ev')),
    IF(EQ(VAR('ln'), num(0)), [FAILB(JOIN(argOf('fn'), txt('() arg is an empty sequence')))]),
    callProc('listitem %s %s', [VAR('ltmp'), num(1)]),
    setV('mmv', VAR('ev')),
    setV('mmt', VAR('evtype')),
    setV('mmi', num(2)),
    REPEATU(GT(VAR('mmi'), VAR('ln')), [
      callProc('listitem %s %s', [VAR('ltmp'), VAR('mmi')]),
      IFELSE(AND(EQ(VAR('mmt'), txt('N')), EQ(VAR('evtype'), txt('N'))), [
        IFELSE(EQ(argOf('fn'), txt('max')), [
          IF(GT(VAR('ev'), VAR('mmv')), [setV('mmv', VAR('ev'))])
        ], [
          IF(LT(VAR('ev'), VAR('mmv')), [setV('mmv', VAR('ev'))])
        ])
      ], [
        IF(AND(EQ(VAR('mmt'), txt('S')), EQ(VAR('evtype'), txt('S'))), [
          callProc('compareText %s %s', [VAR('ev'), VAR('mmv')]),
          IFELSE(EQ(argOf('fn'), txt('max')), [
            IF(GT(VAR('cmp'), num(0)), [setV('mmv', VAR('ev'))])
          ], [
            IF(LT(VAR('cmp'), num(0)), [setV('mmv', VAR('ev'))])
          ])
        ])
      ]),
      chgV('mmi', num(1))
    ]),
    setV('ev', VAR('mmv')),
    setV('evtype', VAR('mmt')),
    setV('pending', num(1))
  ]);
  insertBeforeStmt(sortedIf[0], [minmaxOne]);
  console.log('[P78] min(iterable) / max(iterable) installed');

  // ---------- (b) eval: allow int(x, base) through the two-argument path
  const ev = inDef('eval %s %s');
  const roundEq = find(ev, (b) => b.opcode === 'operator_equals'
    && ((isTokLo(b.inputs.OPERAND1) && litOf(b.inputs.OPERAND2) === 'round') || (isTokLo(b.inputs.OPERAND2) && litOf(b.inputs.OPERAND1) === 'round')));
  if (roundEq.length !== 1) throw new Error('[P78] eval round test = ' + roundEq.length);
  const roundOr = BL()[roundEq[0]].parent;
  if (!roundOr || BL()[roundOr].opcode !== 'operator_or') throw new Error('[P78] eval round or not found');
  replaceExpr(roundOr, () => OR(roundOr, EQ(item('Tok', argOf('lo')), txt('int'))));
  console.log('[P78] eval routes int(x, base) to builtin2');

  // ---------- (c) builtin2b: parse the string in the requested base
  const bb = inDef('builtin2b %s %s %s %s %s');
  const pendIf = find(bb, (b) => b.opcode === 'control_if' && b.inputs.CONDITION && typeof b.inputs.CONDITION[1] === 'string'
    && BL()[b.inputs.CONDITION[1]] && BL()[b.inputs.CONDITION[1]].opcode === 'operator_equals'
    && varName(BL()[b.inputs.CONDITION[1]].inputs.OPERAND1) === 'pending');
  if (pendIf.length !== 1) throw new Error('[P78] builtin2b pending guard = ' + pendIf.length);
  insertBeforeStmt(pendIf[0], [
    IF(EQ(argOf('fn'), txt('int')), [
      IF(NOT(EQ(argOf('a1t'), txt('S'))), [FAIL("TypeError: int() can't convert non-string with explicit base")]),
      IF(NOT(EQ(VAR('evtype'), txt('N'))), [FAIL('TypeError: int() base must be an integer')]),
      callProc('basenum %s %s', [argOf('a1v'), VAR('ev')]),
      setV('evtype', txt('N')),
      setV('pending', num(1))
    ])
  ]);
  console.log('[P78] builtin2b handles int(x, base)');
});
