### slice step assignment even
a = [1, 2, 3, 4, 5]
a[::2] = [0, 0, 0]
print(a)
### slice step assign len2
a = [1, 2, 3, 4]
a[::2] = [9, 9]
print(a)
### slice step assign offset
a = [1, 2, 3, 4, 5, 6]
a[1::3] = [0, 0]
print(a)
### slice step assign negative step
a = [1, 2, 3, 4, 5]
a[::-2] = [7, 8, 9]
print(a)
### slice step assign string
s = 'abcdef'
s2 = list(s)
s2[::2] = ['X', 'Y', 'Z']
print(''.join(s2))
### slice step assign length mismatch
a = [1, 2, 3, 4]
try:
    a[::2] = [1]
except ValueError as e:
    print('ValueError', e)
### slice step zero caught
try:
    print('abc'[::0])
except ValueError:
    print('caught')
### slice step zero assign
a = [1, 2, 3]
try:
    a[::0] = [1, 2, 3]
except ValueError:
    print('caught')
### slice read negative step
a = [1, 2, 3, 4, 5]
print(a[::-1], a[::-2], a[4:1:-1])
### slice assign plain
a = [1, 2, 3, 4]
a[1:3] = [9]
print(a)
### slice assign grow
a = [1, 2, 3]
a[1:2] = [8, 9, 10]
print(a)
### del slice step
a = [1, 2, 3, 4, 5]
del a[::2]
print(a)
### startswith tuple
print('abc'.startswith(('z', 'a')), 'abc'.endswith(('z', 'c')))
### startswith tuple no match
print('abc'.startswith(('x', 'y')), 'abc'.endswith(('x', 'y')))
### startswith list
print('abc'.startswith(['z', 'ab']))
### startswith with start arg
print('abc'.startswith(('b', 'x'), 1))
### startswith single still ok
print('abc'.startswith('ab'), 'abc'.startswith('b', 1), 'abc'.endswith('bc'))
### repr control chars
print(repr('\x0b'), repr('\x0c'), repr('\x07'), repr('\x08'))
### repr nbsp and high char
print(repr('\x1f'), repr('\x7f'))
### repr escapes basic
print(repr('\n'), repr('\t'), repr('\r'), repr('\\'))
