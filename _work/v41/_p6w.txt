### operator overloading add and eq
class V:
    def __init__(self, x):
        self.x = x
    def __add__(self, other):
        return V(self.x + other.x)
    def __eq__(self, other):
        return self.x == other.x
    def __str__(self):
        return 'V' + str(self.x)
print(V(1) + V(2), V(1) == V(1), V(1) == V(2), V(1) != V(2))
