### b1 add returns const
class V:
    def __add__(self, o): return 42
print(V(1) + V(2))
### b2 add returns str
class Z:
    def __add__(self, o): return 'joined'
print(Z(1) + Z(2))
### b3 add uses self only
class Y:
    def __init__(self, x): self.x = x
    def __add__(self, o): return self.x
print(Y(7) + Y(9))
### b4 add uses both attrs
class X:
    def __init__(self, x): self.x = x
    def __add__(self, o): return self.x + o.x
print(X(1) + X(2))
