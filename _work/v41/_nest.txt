### assign dict literal into dict
d = {'a': 1, 'b': 2}
d['a'] = {'x': 1}
print(d)
### assign dict literal into dict second key
d = {'a': 1, 'b': 2}
d['b'] = {'x': 1}
print(d)
### assign call result into dict
def f():
    return {'x': 1}
d = {'a': 1, 'b': 2}
d['a'] = f()
print(d)
### assign list literal into dict
d = {'a': 1, 'b': 2}
d['a'] = [1, 2]
print(d)
### assign list literal into dict second key
d = {'a': 1, 'b': 2}
d['b'] = [1, 2]
print(d)
### key from a call
def k():
    return 'a'
d = {'a': 1, 'b': 2}
d[k()] = 9
print(d)
### key from a subscript
a = ['a']
d = {'a': 1, 'b': 2}
d[a[0]] = 9
print(d)
### key from a subscript plus dict value
keys = ['a']
d = {'a': 1, 'b': 2}
d[keys[0]] = {'w': 1}
print(d)
### index from a call into a list
def i():
    return 1
a = [1, 2, 3]
a[i()] = 9
print(a)
### index from a subscript into a list
idx = [1]
a = [1, 2, 3]
a[idx[0]] = 9
print(a)
### value from a call into a list
def v():
    return [7]
a = [1, 2, 3]
a[0] = v()
print(a)
### value dict literal into list
a = [1, 2, 3]
a[0] = {'x': 1}
print(a)
### nested dict value from a call
def g():
    return {'y': 2}
d = {'a': {'x': 1}, 'b': {'z': 3}}
d['a'] = g()
print(d)
### dict value from arithmetic call
def h(n):
    return n * 2
d = {'a': 1, 'b': 2}
d['a'] = h(5)
print(d)
### key call plus dict value literal
def k2():
    return 'b'
d = {'a': 1, 'b': 2}
d[k2()] = {'q': 7}
print(d)
### dict value list literal then append
d = {'a': 1, 'b': 2}
d['a'] = [1]
d['a'].append(2)
print(d)
### two dict literal assignments
d = {'a': 1, 'b': 2}
d['a'] = {'x': 1}
d['b'] = {'y': 2}
print(d)
### dict value tuple literal
d = {'a': 1, 'b': 2}
d['a'] = (1, 2)
print(d)
### dict value set literal
d = {'a': 1, 'b': 2}
d['a'] = {1, 2}
print(d)
### dict value nested list of dict
d = {'a': 1, 'b': 2}
d['a'] = [{'x': 1}]
print(d)
### list value dict literal at index 1
a = [0, 0, 0]
a[1] = {'x': 1}
print(a)
### list value dict literal in a loop
a = [0, 0]
for i in range(2):
    a[i] = {'i': i}
print(a)
### dict value dict literal in a loop
d = {'a': 1, 'b': 2}
for k in ['a', 'b']:
    d[k] = {'k': k}
print(d)
### dict value from function returning dict twice
def mk(n):
    return {'n': n}
d = {'a': 0, 'b': 0}
d['a'] = mk(1)
d['b'] = mk(2)
print(d)
