### dict lambda via var
ops = {'add': lambda a, b: a + b}
f = ops['add']
print(type(f), f(6, 3))
### dict lambda chain
ops = {'add': lambda a, b: a + b}
print(ops['add'](6, 3))
### compose chain
def compose(f, g):
    return lambda x: f(g(x))
inc = lambda x: x + 1
dbl = lambda x: x * 2
print(compose(dbl, inc)(5))
### call of call result
print((lambda x: x * 2)(5))
