function chain(ids) { return B.chain(IT, ids); }
function blockOf(id) { return BL()[id]; }
function allIds() { return Object.keys(BL()); }

function defIdOf(proccode) {
  for (const id of allIds()) {
    const b = BL()[id];
    if (b.opcode === 'procedures_definition' && b.inputs && b.inputs.custom_block) {
      const proto = BL()[b.inputs.custom_block[1]];
      if (proto && proto.mutation && proto.mutation.proccode === proccode) return id;
    }
  }
  throw new Error('definition not found: ' + proccode);
}
function reach(roots) {
  const seen = new Set();
  const stack = (Array.isArray(roots) ? roots : [roots]).slice();
  while (stack.length) {
    const id = stack.pop();
    if (!id || seen.has(id)) continue;
    const b = BL()[id];
    if (!b) continue;
    seen.add(id);
    for (const v of Object.values(b.inputs || {})) { if (v && typeof v[1] === 'string') stack.push(v[1]); }
    if (b.next) stack.push(b.next);
  }
  return seen;
}
function inDef(proccode) { return reach(defIdOf(proccode)); }
function find(set, pred) { const out = []; for (const id of set) if (pred(BL()[id], id)) out.push(id); return out; }
function refOf(id) {
  for (const pid of allIds()) {
    const p = BL()[pid];
    for (const [slot, v] of Object.entries(p.inputs || {})) {
      if (v && v[1] === id) return { pid, slot, v };
    }
  }
  return null;
}
function cloneExpr(id) {
  if (typeof id !== 'string') return id;
  const src = BL()[id];
  if (!src) return id;
  const copy = JSON.parse(JSON.stringify(src));
  copy.next = null; copy.parent = null; copy.inputs = {};
  const newId = B.nid('cl');
  for (const [k, v] of Object.entries(src.inputs || {})) {
    const nv = JSON.parse(JSON.stringify(v));
    if (typeof nv[1] === 'string') nv[1] = cloneExpr(nv[1]);
    copy.inputs[k] = nv;
  }
  BL()[newId] = copy;
  return newId;
}
// replace the expression block oldId (which sits in some input slot) with a fresh builder result
function replaceExpr(oldId, build) {
  const ref = refOf(oldId);
  if (!ref) throw new Error('cannot locate parent of ' + oldId);
  const fresh = build();
  const desc = ref.v.length > 2 ? [ref.v[0], fresh, ref.v[2]] : [ref.v[0], fresh];
  BL()[ref.pid].inputs[ref.slot] = desc;
  if (typeof fresh === 'string') BL()[fresh].parent = ref.pid;
  return fresh;
}
function spliceAfter(afterId, firstId) {
  const a = BL()[afterId];
  const nx = a.next;
  a.next = firstId;
  BL()[firstId].parent = afterId;
  let t = firstId;
  while (BL()[t].next) t = BL()[t].next;
  BL()[t].next = nx;
  if (nx) BL()[nx].parent = t;
  return firstId;
}
function detachTail(predId) {
  const p = BL()[predId];
  const nx = p.next;
  p.next = null;
  if (nx) BL()[nx].parent = null;
  return nx;
}
function callProc(proccode, args, fallbackWarp) {
  const defId = defIdOf(proccode);
  const t = T(IT);
  const proto = t.blocks[t.blocks[defId].inputs.custom_block[1]];
  const argIds = JSON.parse(proto.mutation.argumentids);
  const inputs = {};
  for (let i = 0; i < argIds.length; i++) inputs[argIds[i]] = B.REP(args[i], [4, '0']);
  return I('procedures_call', {
    inputs,
    mutation: { tagName: 'mutation', children: [], proccode, argumentids: proto.mutation.argumentids, warp: fallbackWarp || proto.mutation.warp || 'true' }
  });
}
const FAIL = (msg) => callProc('fail %s', [typeof msg === 'string' ? txt(msg) : msg]);
const FAILB = (expr) => callProc('fail %s', [expr]);   // message built by a block
function newProc(proccode, argNames) {
  const d = B.makeProcedure(IT, proccode, argNames, true);
  return d;
}
function attachBody(defId, ids) {
  const first = chain(ids);
  BL()[defId].next = first;
  if (first) BL()[first].parent = defId;
  return first;
}
function pruneUnreachable() {
  const keep = new Set();
  const stack = allIds().filter((id) => BL()[id].topLevel);
  while (stack.length) {
    const id = stack.pop();
    if (!id || keep.has(id)) continue;
    const b = BL()[id];
    if (!b) continue;
    keep.add(id);
    for (const v of Object.values(b.inputs || {})) { if (v && typeof v[1] === 'string') stack.push(v[1]); }
    if (b.next) stack.push(b.next);
  }
  let dropped = 0;
  for (const id of allIds()) if (!keep.has(id)) { delete BL()[id]; dropped++; }
  return dropped;
}
function relinkParents() {
  const blocks = BL();
  for (const id of Object.keys(blocks)) blocks[id].parent = null;
  for (const id of Object.keys(blocks)) {
    const b = blocks[id];
    if (b.next && blocks[b.next]) blocks[b.next].parent = id;
    for (const v of Object.values(b.inputs || {})) if (v && typeof v[1] === 'string' && blocks[v[1]]) blocks[v[1]].parent = id;
  }
}

function ancestor(id, pred) {
  let cur = id, guard = 0;
  while (cur && guard++ < 200) {
    const p = BL()[cur].parent;
    if (!p) return null;
    cur = p;
    if (pred(BL()[cur])) return cur;
  }
  return null;
}
function predecessorOf(id) { for (const pid of allIds()) if (BL()[pid].next === id) return pid; return null; }
function setStack(parentId, slot, ids) {
  const first = ids && ids.length ? chain(ids.slice()) : null;
  BL()[parentId].inputs[slot] = [2, first];
  if (first) BL()[first].parent = parentId;
  return first;
}
function prependStack(parentId, slot, blockId) {
  const cur = BL()[parentId].inputs[slot] ? BL()[parentId].inputs[slot][1] : null;
  let tail = blockId;
  while (BL()[tail].next) tail = BL()[tail].next;
  if (typeof cur === 'string') { BL()[tail].next = cur; BL()[cur].parent = tail; }
  BL()[parentId].inputs[slot] = [2, blockId];
  BL()[blockId].parent = parentId;
  return blockId;
}
function replaceStmt(oldId, newId) {
  const pred = predecessorOf(oldId);
  const nx = BL()[oldId].next;
  if (pred) { BL()[pred].next = newId; BL()[newId].parent = pred; }
  else {
    const ref = refOf(oldId);
    if (!ref) throw new Error('cannot place statement ' + oldId);
    BL()[ref.pid].inputs[ref.slot] = ref.v.length > 2 ? [ref.v[0], newId, ref.v[2]] : [ref.v[0], newId];
    BL()[newId].parent = ref.pid;
  }
  BL()[newId].next = nx; if (nx) BL()[nx].parent = newId;
  return newId;
}
function insertBeforeStmt(targetId, ids) {
  const first = chain(ids.slice());
  const pred = predecessorOf(targetId);
  if (pred) spliceAfter(pred, first);
  else {
    const ref = refOf(targetId);
    if (!ref) throw new Error('cannot insert before ' + targetId);
    BL()[ref.pid].inputs[ref.slot] = ref.v.length > 2 ? [ref.v[0], first, ref.v[2]] : [ref.v[0], first];
    BL()[first].parent = ref.pid;
    let tail = first;
    while (BL()[tail].next) tail = BL()[tail].next;
    BL()[tail].next = targetId;
    BL()[targetId].parent = tail;
  }
  return first;
}

// ---- P2: int()/float() accept surrounding whitespace (Python semantics)
patch('P2 int trim', () => {
  newVar('ta', 0); newVar('tb', 0);
  const d = newProc('trimsp %s', ['v']);
  attachBody(d.defId, [
    setV('ta', num(1)),
    REPEATU(OR(GT(VAR('ta'), lenOf(argOf('v'))), NOT(EQ(letterOf(VAR('ta'), argOf('v')), txt(' ')))), [chgV('ta', num(1))]),
    setV('tb', lenOf(argOf('v'))),
    REPEATU(OR(LT(VAR('tb'), VAR('ta')), NOT(EQ(letterOf(VAR('tb'), argOf('v')), txt(' ')))), [chgV('tb', num(-1))]),
    IFELSE(LT(VAR('tb'), VAR('ta')),
      [setV('ev', txt('')), setV('evtype', txt('S'))],
      [callProc('substr %s %s %s', [argOf('v'), VAR('ta'), VAR('tb')])])
  ]);
  const set = inDef('callfn %s %s %s');
  // the OUTER "fn == int or fn == float" test is the one whose comparison feeds an OR
  const eqInt = find(set, (b) => b.opcode === 'operator_equals' && b.inputs.OPERAND2 && b.inputs.OPERAND2[1] && b.inputs.OPERAND2[1][1] === 'int'
    && b.parent && BL()[b.parent] && BL()[b.parent].opcode === 'operator_or');
  if (!eqInt.length) throw new Error('callfn int/float test not found');
  const target = ancestor(eqInt[0], (b) => b.opcode === 'control_if');
  if (!target) throw new Error('callfn int control_if not found');
  const call = callProc('trimsp %s', [VAR('ev')]);
  prependStack(target, 'SUBSTACK', call);
  console.log('[P2] int()/float() trim');
});

// ---- P3: round() uses banker's rounding (half to even)
patch('P3 round half even', () => {
  newVar('lf', 0); newVar('fr', 0); newVar('mrt', 0);
  const d = newProc('roundpy %s', ['v']);
  attachBody(d.defId, [
    setV('lf', FLOOR(argOf('v'))),
    setV('fr', SUB(argOf('v'), FLOOR(argOf('v')))),
    IFELSE(EQ(VAR('fr'), num('0.5')),
      [IFELSE(EQ(MOD(VAR('lf'), num(2)), num(1)), [setV('ev', ADD(VAR('lf'), num(1)))], [setV('ev', VAR('lf'))])],
      [setV('ev', I('operator_round', { inputs: { NUM: B.REP(argOf('v'), [4, '0']) } }))]),
    setV('evtype', txt('N'))
  ]);
  // callfn: ev = round(ev)
  const cf = inDef('callfn %s %s %s');
  const setRounds = find(cf, (b) => b.opcode === 'data_setvariableto' && b.fields.VARIABLE[0] === 'ev' && b.inputs.VALUE && typeof b.inputs.VALUE[1] === 'string' && BL()[b.inputs.VALUE[1]].opcode === 'operator_round');
  if (setRounds.length !== 1) throw new Error('callfn round stmt not found (' + setRounds.length + ')');
  replaceStmt(setRounds[0], callProc('roundpy %s', [VAR('ev')]));
  // builtin2b: round(x, n)
  const b2 = inDef('builtin2b %s %s %s %s %s');
  const ife = find(b2, (b) => b.opcode === 'control_if_else' && reach(b.inputs.SUBSTACK[1]).size && [...reach(b.inputs.SUBSTACK[1])].some((id) => BL()[id].opcode === 'operator_round'));
  if (ife.length !== 1) throw new Error('builtin2b round if not found (' + ife.length + ')');
  const a1 = () => argOf('a1v');
  const qp = () => VAR('qpow');
  setStack(ife[0], 'SUBSTACK', [setV('mrt', DIV(a1(), qp())), callProc('roundpy %s', [VAR('mrt')]), setV('ev', MUL(VAR('ev'), qp()))]);
  const ife2 = find(reach(ife[0]), (b) => b.opcode === 'control_if_else' && b.inputs.SUBSTACK2 && typeof b.inputs.SUBSTACK2[1] === 'string' && [...reach(b.inputs.SUBSTACK2[1])].some((id) => BL()[id].opcode === 'operator_round'));
  if (ife2.length !== 1) throw new Error('builtin2b else round not found');
  setStack(ife2[0], 'SUBSTACK2', [setV('mrt', MUL(a1(), qp())), callProc('roundpy %s', [VAR('mrt')]), setV('ev', DIV(VAR('ev'), qp()))]);
  console.log('[P3] round = half-to-even');
});

// ---- P11: **= and //= augmented assignment
patch('P11 pow/floor assign', () => {
  const ex = inDef('execute');
  const eqPlus = find(ex, (b) => b.opcode === 'operator_equals' && b.inputs.OPERAND2 && b.inputs.OPERAND2[1] && b.inputs.OPERAND2[1][1] === '+=');
  if (!eqPlus.length) throw new Error('+= branch not found');
  const augIf = ancestor(eqPlus[0], (b) => b.opcode === 'control_if');
  if (!augIf) throw new Error('aug assign if not found');
  const tok2 = () => item('Tok', num(2));
  const cond = AND(EQ(item('Kind', num(1)), txt('I')), AND(OR(EQ(tok2(), txt('//')), EQ(tok2(), txt('**'))), EQ(item('Tok', num(3)), txt('='))));
  const body = [
    setV('name', item('Tok', num(1))),
    callProc('lookup %s', [VAR('name')]),
    IFELSE(EQ(VAR('found'), num(0)),
      [FAIL(JOIN(txt('NameError: '), VAR('name')))],
      [callProc('rightside %s %s %s %s %s', [item('Values', VAR('found')), item('Types', VAR('found')), item('Tok', num(2)), txt('4'), VAR('count')]),
       IF(EQ(VAR('error'), txt('')), [callProc('assign %s %s %s', [VAR('name'), VAR('ev'), VAR('evtype')])])]),
    setV('pending', num(1))
  ];
  insertBeforeStmt(augIf, [IF(cond, body)]);
  console.log('[P11] **= //= installed');
});

// ---- P12: is / is not
patch('P12 is operator', () => {
  // rank: is / isnot sit with the comparison operators
  const rk = defIdOf('rank %s');
  const tail = (() => { let id = BL()[rk].next; while (BL()[id].next) id = BL()[id].next; return id; })();
  spliceAfter(tail, chain([
    IF(EQ(argOf('op'), txt('is')), [setV('erank', num(3))]),
    IF(EQ(argOf('op'), txt('isnot')), [setV('erank', num(3))])
  ]));
  // eval: turn "is not" into one operator token
  const ev = inDef('eval %s %s');
  const ifNotIn = find(ev, (b) => b.opcode === 'control_if_else' && b.inputs.SUBSTACK && typeof b.inputs.SUBSTACK[1] === 'string' && [...reach(b.inputs.SUBSTACK[1])].some((id) => BL()[id].opcode === 'data_setvariableto' && BL()[id].fields.VARIABLE[0] === 'eop' && BL()[id].inputs.VALUE[1] && BL()[id].inputs.VALUE[1][1] === 'notin'));
  if (ifNotIn.length !== 1) throw new Error('eval notin dispatch not found (' + ifNotIn.length + ')');
  spliceAfter(ifNotIn[0], chain([
    IF(AND(EQ(VAR('eop'), txt('is')), EQ(item('Tok', ADD(VAR('ek'), num(1))), txt('not'))), [setV('eop', txt('isnot'))]),
    IF(AND(EQ(VAR('eop'), txt('not')), EQ(item('Tok', SUB(VAR('ek'), num(1))), txt('is'))), [setV('eop', txt('skip'))])
  ]));
  // pair: isnot consumes two tokens
  const pr = inDef('pair %s %s %s %s');
  const pairIf = find(pr, (b) => b.opcode === 'control_if_else' && b.inputs.SUBSTACK && typeof b.inputs.SUBSTACK[1] === 'string' && [...reach(b.inputs.SUBSTACK[1])].some((id) => BL()[id].opcode === 'procedures_call' && BL()[id].mutation && BL()[id].mutation.proccode === 'rightside %s %s %s %s %s' && JSON.stringify(BL()[id].inputs).includes('notin')));
  if (pairIf.length !== 1) throw new Error('pair notin branch not found (' + pairIf.length + ')');
  const oldElse = BL()[pairIf[0]].inputs.SUBSTACK2 ? BL()[pairIf[0]].inputs.SUBSTACK2[1] : null;
  const newElse = IFELSE(EQ(argOf('op'), txt('isnot')),
    [callProc('rightside %s %s %s %s %s', [VAR('ev'), VAR('evtype'), txt('isnot'), ADD(argOf('cut'), num(2)), argOf('hi')])],
    oldElse ? [oldElse] : []);
  setStack(pairIf[0], 'SUBSTACK2', [newElse]);
  // rightside: identity comparison
  const rs = inDef('rightside %s %s %s %s %s');
  const errIf = find(rs, (b) => b.opcode === 'control_if' && b.inputs.CONDITION && typeof b.inputs.CONDITION[1] === 'string' && BL()[b.inputs.CONDITION[1]].opcode === 'operator_equals' && BL()[b.inputs.CONDITION[1]].inputs.OPERAND1 && BL()[b.inputs.CONDITION[1]].inputs.OPERAND2 && BL()[b.inputs.CONDITION[1]].inputs.OPERAND2[1] && BL()[b.inputs.CONDITION[1]].inputs.OPERAND2[1][1] === '');
  if (errIf.length !== 1) throw new Error('rightside body if not found (' + errIf.length + ')');
  newVar('isv', 0);
  const idChain = [
    IF(OR(EQ(argOf('op'), txt('is')), EQ(argOf('op'), txt('isnot'))), [
      setV('isv', num(0)),
      IF(AND(EQ(argOf('leftval'), VAR('ev')), EQ(argOf('lefttype'), VAR('evtype'))), [setV('isv', num(1))]),
      setV('ev', VAR('isv')),
      IF(EQ(argOf('op'), txt('isnot')), [setV('ev', SUB(num(1), VAR('ev')))]),
      setV('evtype', txt('B'))
    ])
  ];
  prependStack(errIf[0], 'SUBSTACK', chain(idChain));
  console.log('[P12] is / is not installed');
});

// ---- P4: conditional expression  a if c else b
patch('P4 ternary', () => {

       setV('pmbuf', txt('')), setV('pendnl', num(1)), setV('pendtxt', txt(''))],
      [setV('pmbuf', JOIN(VAR('pmbuf'), VAR('printbuf')))])
  ];
  const head = chain([...scanKw, ...commaScan, setV('printbuf', txt(''))]);
  insertBeforeStmt(firstStmt, [head]);
  // drop the original "clear(Comma) ... append(Comma, count)" preamble: the new
  // scan above already built Comma, stopping before any sep=/end= keyword
  const oldSentinel = find(pa, (b) => b.opcode === 'data_addtolist' && b.fields.LIST && b.fields.LIST[0] === 'Comma'
    && b.inputs.ITEM && Array.isArray(b.inputs.ITEM[1]) && b.inputs.ITEM[1][1] === 'count');
  if (oldSentinel.length !== 1) throw new Error('printargs old sentinel = ' + oldSentinel.length);
  {
    const pred = predecessorOf(firstStmt);
    const after = BL()[oldSentinel[0]].next;
    if (!pred) throw new Error('printargs head not spliced');
    BL()[pred].next = after;
    if (after) BL()[after].parent = pred;
    BL()[firstStmt].parent = null;
    BL()[oldSentinel[0]].next = null;
  }
  // rebuild the argument loop: empty segments (trailing comma before sep=) produce
  // no text, and the join uses psep instead of a hard-coded space
  const loop = find(pa, (b) => b.opcode === 'control_repeat' && b.inputs.TIMES && typeof b.inputs.TIMES[1] === 'string'
    && BL()[b.inputs.TIMES[1]].opcode === 'data_lengthoflist' && BL()[b.inputs.TIMES[1]].fields.LIST[0] === 'Comma');
  if (loop.length !== 1) throw new Error('printargs loop = ' + loop.length);
  setStack(loop[0], 'SUBSTACK', [
    setV('phi', SUB(item('Comma', VAR('pi')), num(1))),
    IF(NOT(GT(VAR('plo'), VAR('phi'))), [
      callProc('eval %s %s', [VAR('plo'), VAR('phi')]),
      IF(GT(VAR('pi'), num(1)), [setV('printbuf', JOIN(VAR('printbuf'), VAR('psep')))]),
      callProc('strval %s %s', [VAR('ev'), VAR('evtype')]),
      setV('printbuf', JOIN(VAR('printbuf'), VAR('ev')))
    ]),
    setV('plo', ADD(item('Comma', VAR('pi')), num(1))),
    chgV('pi', num(1))
  ]);
  // final log -> emit (honours end=), preceded by the sep=/end= evaluation
  const emitFirst = chain(emit);
  replaceStmt(logCall, emitFirst);
  // sep=/end= are evaluated before the arguments (they apply to the whole call)
  insertBeforeStmt(loop[0], [...sepEval, ...endEval]);
  console.log('[P7] print sep/end installed');
});

// ---- P8/P9: list slicing -- a[1:3], a[i:j] = [...], del a[i:j]
patch('P8/P9 list slices', () => {
  newVar('lsacc', ''); newVar('lsi', 0);
  newVar('slc', 0); newVar('sd', 0); newVar('sk', 0);
  newVar('sls', 0); newVar('sle', 0); newVar('sln', 0); newVar('slr', ''); newVar('slrec', '');
  newVar('swa', ''); newVar('swn', 0); newVar('sws', 0); newVar('swe', 0); newVar('swi', 0); newVar('swacc', '');
  const tokAt = (x) => item('Tok', x);

  // shared writer: name, recv(encoded list), xs, xe (0-based, half-open), repl(encoded list or "")
  const sw = newProc('slicewrite %s %s %s %s %s', ['name', 'recv', 'xs', 'xe', 'repl']);
  attachBody(sw.defId, [
    setV('swa', argOf('recv')),
    callProc('listlen %s', [VAR('swa')]),
    setV('swn', VAR('ev')),
    IFELSE(LT(argOf('xs'), num(0)), [setV('sws', ADD(argOf('xs'), VAR('swn')))], [setV('sws', argOf('xs'))]),
    IF(LT(VAR('sws'), num(0)), [setV('sws', num(0))]),
    IF(GT(VAR('sws'), VAR('swn')), [setV('sws', VAR('swn'))]),
    IFELSE(LT(argOf('xe'), num(0)), [setV('swe', ADD(argOf('xe'), VAR('swn')))], [setV('swe', argOf('xe'))]),
    IF(LT(VAR('swe'), num(0)), [setV('swe', num(0))]),
    IF(GT(VAR('swe'), VAR('swn')), [setV('swe', VAR('swn'))]),
    IF(LT(VAR('swe'), VAR('sws')), [setV('swe', VAR('sws'))]),
    setV('swacc', txt('')), setV('swi', num(1)),
    REPEATU(GT(VAR('swi'), VAR('sws')), [
      callProc('listitem %s %s', [VAR('swa'), VAR('swi')]),
      setV('swacc', JOIN(JOIN(JOIN(VAR('swacc'), VAR('lsep')), VAR('evtype')), VAR('ev'))),
      chgV('swi', num(1))
    ]),
    setV('swacc', JOIN(VAR('swacc'), argOf('repl'))),
    setV('swi', ADD(VAR('swe'), num(1))),
    REPEATU(GT(VAR('swi'), VAR('swn')), [
      callProc('listitem %s %s', [VAR('swa'), VAR('swi')]),
      setV('swacc', JOIN(JOIN(JOIN(VAR('swacc'), VAR('lsep')), VAR('evtype')), VAR('ev'))),
      chgV('swi', num(1))
    ]),
    callProc('assign %s %s %s', [argOf('name'), VAR('swacc'), txt('L')]),
    setV('ev', txt('')), setV('evtype', txt('S'))
  ]);

  // find the top-level ':' inside an index region (returns the token index in slc)
  const colonScan = (startExpr, endExpr) => [
    setV('slc', num(0)), setV('sd', num(0)), setV('sk', startExpr),
    REPEATU(GT(VAR('sk'), endExpr), [
      IF(OR(EQ(tokAt(VAR('sk')), txt('(')), EQ(tokAt(VAR('sk')), txt('['))), [chgV('sd', num(1))]),
      IF(OR(EQ(tokAt(VAR('sk')), txt(')')), EQ(tokAt(VAR('sk')), txt(']'))), [chgV('sd', num(-1))]),
      IF(AND(EQ(VAR('sd'), num(0)), AND(EQ(tokAt(VAR('sk')), txt(':')), EQ(VAR('slc'), num(0)))), [setV('slc', VAR('sk'))]),
      chgV('sk', num(1))
    ])
  ];
  const bounds = (startTok, endTok) => [
    setV('sls', num(0)),
    setV('sle', VAR('sln')),
    IF(GT(VAR('slc'), startTok), [callProc('eval %s %s', [startTok, SUB(VAR('slc'), num(1))]), setV('sls', VAR('ev'))]),
    IF(LT(VAR('slc'), endTok), [callProc('eval %s %s', [ADD(VAR('slc'), num(1)), endTok]), setV('sle', VAR('ev'))])
  ];

  // ---- read: a[1:3]
  const ig = inDef('indexget %s %s');
  const xlenSet = find(ig, (b) => b.opcode === 'data_setvariableto' && b.fields.VARIABLE[0] === 'xlen' && b.inputs.VALUE && typeof b.inputs.VALUE[1] === 'string' && BL()[b.inputs.VALUE[1]].opcode === 'operator_length');
  if (xlenSet.length !== 1) throw new Error('indexget xlen = ' + xlenSet.length);
  spliceAfter(xlenSet[0], chain([
    IF(EQ(VAR('xb'), txt('L')), [callProc('listlen %s', [VAR('xa')]), setV('xlen', VAR('ev'))])
  ]));
  const subFinal = find(ig, (b) => b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === 'substr %s %s %s'
    && Object.values(b.inputs).some((v) => Array.isArray(v[1]) && v[1][1] === 'xhi'));
  if (subFinal.length !== 1) throw new Error('indexget slice tail = ' + subFinal.length);
  replaceStmt(subFinal[0], IFELSE(EQ(VAR('xb'), txt('L')), [
    setV('lsacc', txt('')),
    setV('lsi', ADD(VAR('xlo'), num(1))),
    REPEATU(GT(VAR('lsi'), VAR('xhi')), [
      callProc('listitem %s %s', [VAR('xa'), VAR('lsi')]),
      setV('lsacc', JOIN(JOIN(JOIN(VAR('lsacc'), VAR('lsep')), VAR('evtype')), VAR('ev'))),
      chgV('lsi', num(1))
    ]),
    setV('ev', VAR('lsacc')),
    setV('evtype', txt('L'))
  ], [callProc('substr %s %s %s', [VAR('xa'), ADD(VAR('xlo'), num(1)), VAR('xhi')])]));

  // ---- assign: a[i:j] = [...]
  const iset = inDef('indexset %s %s');
  const setEqFail = find(iset, (b) => b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === 'fail %s' && JSON.stringify(b.inputs).includes('expected = after'));
  if (setEqFail.length !== 1) throw new Error('indexset = check not found');
  const setIf = ancestor(setEqFail[0], (b) => b.opcode === 'control_if');
  if (!setIf) throw new Error('indexset = check if not found');
  spliceAfter(setIf, chain([
    ...colonScan(ADD(argOf('lo'), num(2)), SUB(VAR('lsp'), num(1))),
    IF(GT(VAR('slc'), num(0)), [
      callProc('eval %s %s', [argOf('lo'), argOf('lo')]),
      IF(NOT(EQ(VAR('evtype'), txt('L'))), [FAIL(txt('TypeError: slice assignment needs a list'))]),
      setV('slrec', VAR('ev')),
      callProc('listlen %s', [VAR('slrec')]),
      setV('sln', VAR('ev')),
      setV('sls', num(0)),
      setV('sle', VAR('sln')),
      IF(GT(VAR('slc'), ADD(argOf('lo'), num(2))), [
        callProc('eval %s %s', [ADD(argOf('lo'), num(2)), SUB(VAR('slc'), num(1))]),
        setV('sls', VAR('ev'))
      ]),
      IF(LT(VAR('slc'), SUB(VAR('lsp'), num(1))), [
        callProc('eval %s %s', [ADD(VAR('slc'), num(1)), SUB(VAR('lsp'), num(1))]),
        setV('sle', VAR('ev'))
      ]),
      callProc('eval %s %s', [ADD(VAR('lsp'), num(2)), argOf('hi')]),
      IF(NOT(EQ(VAR('evtype'), txt('L'))), [FAIL(txt('TypeError: slice assignment needs a list'))]),
      setV('slr', VAR('ev')),
      callProc('slicewrite %s %s %s %s %s', [tokAt(argOf('lo')), VAR('slrec'), VAR('sls'), VAR('sle'), VAR('slr')]),
      I('control_stop', { fields: { STOP_OPTION: ['this script', null] }, mutation: { tagName: 'mutation', children: [], hasnext: 'false' } })
    ])
  ]));

  // ---- del a[i:j]
  const idel = inDef('indexdel %s %s');
  const delBracket = find(idel, (b) => b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === 'fail %s' && JSON.stringify(b.inputs).includes('expected ] after del'));
  if (delBracket.length !== 1) throw new Error('indexdel bracket check not found');
  const delIf = ancestor(delBracket[0], (b) => b.opcode === 'control_if');
  if (!delIf) throw new Error('indexdel bracket if not found');
  spliceAfter(delIf, chain([
    ...colonScan(ADD(argOf('lo'), num(3)), SUB(VAR('lsp'), num(1))),
    IF(GT(VAR('slc'), num(0)), [
      callProc('eval %s %s', [ADD(argOf('lo'), num(1)), ADD(argOf('lo'), num(1))]),
      IF(NOT(EQ(VAR('evtype'), txt('L'))), [FAIL(txt('TypeError: del needs a list'))]),
      setV('slrec', VAR('ev')),
      callProc('listlen %s', [VAR('slrec')]),
      setV('sln', VAR('ev')),
      setV('sls', num(0)),
      setV('sle', VAR('sln')),
      IF(GT(VAR('slc'), ADD(argOf('lo'), num(3))), [
        callProc('eval %s %s', [ADD(argOf('lo'), num(3)), SUB(VAR('slc'), num(1))]),
        setV('sls', VAR('ev'))
      ]),
      IF(LT(VAR('slc'), SUB(VAR('lsp'), num(1))), [
        callProc('eval %s %s', [ADD(VAR('slc'), num(1)), SUB(VAR('lsp'), num(1))]),
        setV('sle', VAR('ev'))
      ]),
      callProc('slicewrite %s %s %s %s %s', [tokAt(ADD(argOf('lo'), num(1))), VAR('slrec'), VAR('sls'), VAR('sle'), txt('')]),
      I('control_stop', { fields: { STOP_OPTION: ['this script', null] }, mutation: { tagName: 'mutation', children: [], hasnext: 'false' } })
    ])
  ]));
  // del slice needs the bounds too: recompute inside the branch
  console.log('[P8/P9] list slicing installed');
});
