### class basics
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    def total(self):
        return self.x + self.y
p = Point(1, 2)
print(p.x, p.y, p.total())
### class str and repr
class Thing:
    def __init__(self, n):
        self.n = n
    def __str__(self):
        return 'Thing(' + str(self.n) + ')'
    def __repr__(self):
        return 'Thing!' + str(self.n)
t = Thing(3)
print(t, str(t), repr(t))
print([t], {'k': t}, (t,))

### method mutating self list attribute
class Bag:
    def __init__(self):
        self.items = []
    def add(self, v):
        self.items.append(v)
        return len(self.items)
b = Bag()
print(b.add(1), b.add(2), b.items)
### method reassigning self attribute
class Counter:
    def __init__(self):
        self.n = 0
    def bump(self):
        self.n = self.n + 1
        return self.n
c = Counter()
print(c.bump(), c.bump(), c.n)

### inheritance override and super
class Animal:
    def __init__(self, name):
        self.name = name
    def speak(self):
        return '...'
    def describe(self):
        return self.name + ' says ' + self.speak()
class Dog(Animal):
    def speak(self):
        return 'woof'
d = Dog('rex')
print(d.name, d.speak(), d.describe(), isinstance(d, Animal), isinstance(d, Dog))
### super init chain
class A:
    def __init__(self):
        self.a = 1
class B(A):
    def __init__(self):
        super().__init__()
        self.b = 2
b = B()
print(b.a, b.b)
### super method call

print(B2().hi())

### multiple inheritance mro basic

### diamond inheritance
class Base:
    def __init__(self):
        self.tag = 'base'
    def who(self):
        return 'Base'
class Mid1(Base):
    def who(self):
        return 'Mid1'
class Mid2(Base):
    def who(self):
        return 'Mid2'
class Leaf(Mid1, Mid2):
    pass
l = Leaf()
print(l.who(), l.tag, isinstance(l, Base), isinstance(l, Mid1), isinstance(l, Mid2))
### diamond with super chain
class B4:
    def name(self):
        return 'B4'
class M1(B4):
    def name(self):
        return 'M1>' + super().name()
class M2(B4):
    def name(self):
        return 'M2>' + super().name()
class L4(M1, M2):
    def name(self):
        return 'L4>' + super().name()
print(L4().name())
### operator overloading add and eq
class V:
    def __init__(self, x):
        self.x = x
    def __add__(self, other):
        return V(self.x + other.x)
    def __eq__(self, other):
        return self.x == other.x
    def __str__(self):
        return 'V' + str(self.x)
print(V(1) + V(2), V(1) == V(1), V(1) == V(2), V(1) != V(2))
### operator overloading len getitem
class Seq:
    def __init__(self, items):
        self.items = items
    def __len__(self):
        return len(self.items)
    def __getitem__(self, i):
        return self.items[i]
s = Seq([10, 20, 30])
print(len(s), s[0], s[-1], s[1:])
for v in s:
    print('v', v)
### operator overloading lt for sort
class P:
    def __init__(self, n):
        self.n = n
    def __lt__(self, other):
        return self.n < other.n
    def __str__(self):
        return str(self.n)
ps = [P(3), P(1), P(2)]
print([str(p) for p in sorted(ps)])

### class with list of objects and methods
class Stack:
    def __init__(self):
        self.data = []
    def push(self, v):
        self.data.append(v)
    def pop(self):
        return self.data.pop()
    def size(self):
        return len(self.data)
s = Stack()
s.push(1)
s.push(2)
print(s.size(), s.pop(), s.size())

### except else finally order
def run(flag):
    try:
        if flag:
            raise ValueError('boom')
        print('no error')
    except ValueError as e:
        print('caught', e)
    else:
        print('else')
    finally:
        print('finally')
run(False)
run(True)

### custom exception with attributes
class AppError(Exception):
    def __init__(self, code, msg):
        self.code = code
        self.msg = msg
    def __str__(self):
        return str(self.code) + ':' + self.msg
try:
    raise AppError(7, 'bad')
except AppError as e:
    print(e.code, e.msg, e)
