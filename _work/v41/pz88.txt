// PZ88（临时调试）：execute 里 unpackto 之前，bnames 的实际内容/长度是什么？
patch('PZ88 bnames probe', () => {
  const taken = new Set(Object.values(T(IT).variables).map((x) => x[0]));
  const names = ['y1', 'y2', 'dbg'];
  for (const n of names) if (taken.has(n)) throw new Error('[PZ88] var taken: ' + n);
  newVar('y1', ''); newVar('y2', 0); newVar('dbg', '');
  const ownerOf = (id) => {
    let cur = id;
    for (let i = 0; i < 200 && cur; i++) {
      const q = BL()[cur];
      if (!q) return '?';
      if (q.opcode === 'procedures_definition') {
        const pr = BL()[q.inputs.custom_block[1]];
        return pr && pr.mutation ? String(pr.mutation.proccode) : '?';
      }
      cur = q.parent;
    }
    return '(toplevel)';
  };
  let site = null;
  for (const id of new Set([...allIds(), ...Object.keys(T(IT).blocks)])) {
    const b = BL()[id];
    if (!b || b.opcode !== 'procedures_call' || !b.mutation) continue;
    if (String(b.mutation.proccode).indexOf('unpackto') < 0) continue;
    if (ownerOf(id) !== 'execute') continue;
    site = id; break;
  }
  if (!site) throw new Error('[PZ88] execute unpackto site not found');
  console.log('[PZ88] site=' + site);
  const acc = [];
  const add = (expr) => acc.push(setV('dbg', JOIN(VAR('dbg'), expr)));
  acc.push(setV('y1', VAR('bnames')));
  acc.push(setV('y2', lenOf(VAR('bnames'))));
  acc.push(setV('dbg', txt('PZ88 bnames=')));
  add(VAR('y1'));
  add(txt(' len='));
  add(VAR('y2'));
  acc.push(FAIL(VAR('dbg')));
  insertBeforeStmt(site, acc);
  console.log('[PZ88] probe installed (' + acc.length + ')');
});
