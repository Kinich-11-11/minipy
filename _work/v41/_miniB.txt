### gcd lcm euclid
def gcd(a, b):
    while b:
        a, b = b, a%b
    return a
def lcm(a, b):
    return a * b // gcd(a, b)
print(gcd(48, 36), gcd(17, 5), lcm(4, 6), lcm(21, 6), gcd(0, 7))
### quicksort recursive
def qsort(a):
    if len(a) <= 1:
        return a
    p = a[0]
    left = []
    right = []
    for x in a[1:]:
        if x < p:
            left.append(x)
        else:
            right.append(x)
    return qsort(left) + [p] + qsort(right)
print(qsort([8, 3, 9, 1, 5, 5, 2, 7]))
