### decorator basic
def double_result(f):
    def wrapper(n):
        return f(n) * 2
    return wrapper
@double_result
def val(n):
    return n + 1
print(val(5), val(0))
### decorator with args factory
def addn(k):
    def deco(f):
        def w(x):
            return f(x) + k
        return w
    return deco
@addn(10)
def inc(x):
    return x * 1
print(inc(5))
### def redef plain regression
def f(n):
    return n
print(f(3))
def f(n):
    return n * 2
print(f(3))
### closure factory regression
def mk(k):
    def ap(x):
        return x * k
    return ap
g = mk(3)
print(g(4))
