### two list literals
s = [1, 2], [3, 4]
print(s)
### calls tuple
def f():
    return 1, 2
def g():
    return 3, 4
x = f(), g()
print(x)
### return two paren tuples
def h():
    return (1, 2), (3, 4)
print(h())
### doubly wrapped
v = ((1, 2)), (3, 4)
print(v)
### second element paren
w = 1, (2, 3)
print(w)
### first element call then paren
def p():
    return 1, 2
y = p(), (3, 4)
print(y)
### two lists nested
z = [[1], [2]], [[3]]
print(z)
