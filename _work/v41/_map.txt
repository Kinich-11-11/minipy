### map str
print(list(map(str, [1, 2, 3])))
### map int
print(list(map(int, ['1', '2'])))
### filter len
print(list(filter(len, ['a', '', 'bb'])))
### filter bool values
print(list(filter(str, [0, 1, '', 'x'])))
### map over tuple
print(list(map(str, (1, 2))))
### map user function
def dbl(x):
    return x * 2
print(list(map(dbl, [1, 2, 3])))
### sum of map
print(sum(map(int, ['1', '2', '3'])))
### map with range
print(list(map(str, range(3))))
