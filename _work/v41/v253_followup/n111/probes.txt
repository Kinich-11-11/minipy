### P23 N111 closure-only generator, observe two resumes and exported alias

def outer():
    a = []
    def f():
        a.append(2)
    def g():
        f()
        yield len(a)
        f()
        yield len(a)
    def read():
        return a
    return g(), read, a

x, read, a = outer()
print(next(x), read(), a)
print(next(x), read(), a)
