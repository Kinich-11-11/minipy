### s1 str dunder
class Q:
    def __str__(self):
        return 'Q!'
print(Q(), str(Q()))
### s2 fstring dunder
class Q:
    def __str__(self):
        return 'Q!'
print(f'{Q()}')
### s3 repr dunder
class R:
    def __repr__(self):
        return 'R!'
print(repr(R()))
### s4 container repr
class R:
    def __repr__(self):
        return 'R!'
print([R()])
### s5 no dunder still class name
class N:
    pass
print(str(N()), repr(N()))
### s6 class with both
class B:
    def __str__(self):
        return 'S'
    def __repr__(self):
        return 'R'
print(str(B()), repr(B()), [B()])
