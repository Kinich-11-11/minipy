### sort ascending default
a = [3, 1, 2]
a.sort()
print(a)
### sort reverse true
a = [3, 1, 2]
a.sort(reverse=True)
print(a)
### sort reverse false
a = [3, 1, 2]
a.sort(reverse=False)
print(a)
### sort reverse true on strings
a = ['bb', 'a', 'ccc']
a.sort(reverse=True)
print(a)
### sort key only
a = ['bb', 'a', 'ccc']
a.sort(key=len)
print(a)
### sort key and reverse
a = ['bb', 'a', 'ccc']
a.sort(key=len, reverse=True)
print(a)
### sort reverse then reverse method
a = [1, 2, 3]
a.sort(reverse=True)
a.reverse()
print(a)
### sort empty and single
a = []
a.sort(reverse=True)
print(a)
b = [1]
b.sort(reverse=True)
print(b)
### sorted ascending and reverse
print(sorted([3, 1, 2]), sorted([3, 1, 2], reverse=True))
### sorted with key and reverse
print(sorted(['bb', 'a', 'ccc'], key=len), sorted(['bb', 'a', 'ccc'], key=len, reverse=True))
### sorted nested calls
print(sorted(sorted([3, 1, 2], key=lambda x: -x), reverse=True))
### sort inside function
def f(xs):
    xs.sort(reverse=True)
    return xs
print(f([1, 3, 2]))
### sort stability
a = [(1, 'b'), (1, 'a'), (0, 'c')]
a.sort()
print(a)
a.sort(reverse=True)
print(a)
### sort floats and mixed ints
a = [2.5, 1, 3]
a.sort()
print(a)
a.sort(reverse=True)
print(a)
### sort after append
a = []
a.append(5)
a.append(1)
a.sort(reverse=True)
print(a)
### two sorts in a row
a = [3, 1, 2]
a.sort(reverse=True)
b = [9, 7, 8]
b.sort()
print(a, b)
### sort reverse does not leak to later sorts
a = [3, 1, 2]
a.sort(reverse=True)
c = [4, 6, 5]
c.sort()
print(a, c)
### sorted reverse does not leak to sort
print(sorted([1, 3, 2], reverse=True))
d = [1, 3, 2]
d.sort()
print(d)
### sort reverse with extra whitespace in source
a = [3, 1, 2]
a.sort( reverse = True )
print(a)
### sort with tuple list reverse
a = [(2, 'b'), (1, 'a')]
a.sort(reverse=True)
print(a)
