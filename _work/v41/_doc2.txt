### triple single quote one line
def f():
    '''doc'''
    return 1

print(f())
### triple single quote multi line
def g():
    '''line one
    line two
    '''
    return 2

print(g())
### docstring with apostrophe
def h():
    """it's fine"""
    return 3

print(h())
### docstring with double quotes inside
def k():
    """has "" inside"""
    return 4

print(k())
### string with escaped quote
s = "a\"b"
print(s)
### string with backslash
s = "a\\b"
print(s)
### docstring with equals and brackets
def m():
    """a = [1, 2] (x) {y}"""
    return 5

print(m())
### three quotes used as empty string
s = "" + ""
print(len(s))
### docstring then code after
def n():
    """doc"""
    x = 1
    return x + 1

print(n())
### nested triple in expression
print(len("""abc"""))
