### inheritance chain override
class Animal:
    def __init__(self, name):
        self.name = name
    def speak(self):
        return self.name + ' makes a sound'
    def intro(self):
        return 'I am ' + self.name
class Dog(Animal):
    def speak(self):
        return self.name + ' barks'
class Puppy(Dog):
    def speak(self):
        return Dog.speak(self) + ' softly'
a = Animal('Generic')
d = Dog('Rex')
p = Puppy('Bit')
print(a.speak(), '|', d.speak(), '|', p.speak())
print(d.intro(), p.intro())
### polymorphism loop
class Shape:
    def area(self):
        return 0
class Circle(Shape):
    def __init__(self, r):
        self.r = r
    def area(self):
        return 3 * self.r * self.r
class Rect(Shape):
    def __init__(self, w, h):
        self.w = w
        self.h = h
    def area(self):
        return self.w * self.h
shapes = [Circle(2), Rect(3, 4), Shape()]
for s in shapes:
    print(s.area())
total = 0
for s in shapes:
    total += s.area()
print('total', total)
### operator overloading vector
class Vec:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    def __add__(self, o):
        return Vec(self.x + o.x, self.y + o.y)
    def __sub__(self, o):
        return Vec(self.x - o.x, self.y - o.y)
    def __mul__(self, k):
        return Vec(self.x * k, self.y * k)
    def __eq__(self, o):
        return self.x == o.x and self.y == o.y
    def __str__(self):
        return '(' + str(self.x) + ',' + str(self.y) + ')'
v1 = Vec(1, 2)
v2 = Vec(3, 4)
print(v1 + v2, v2 - v1, v1 * 3)
print(v1 + v2 == Vec(4, 6), v1 == v2)

### custom exception hierarchy
class AppError(Exception):
    pass
class NotFound(AppError):
    pass
class BadInput(AppError):
    pass
def find(d, k):
    if k not in d:
        raise NotFound('key ' + k)
    if not isinstance(d[k], int):
        raise BadInput('value bad')
    return d[k]
store = {'a': 1, 'b': 'x'}
for k in ['a', 'b', 'c']:
    try:
        print(find(store, k))
    except NotFound as e:
        print('nf', e)
    except BadInput as e:
        print('bi', e)
    except AppError:
        print('app')
### bank account validation
class Account:
    def __init__(self, owner, balance):
        self.owner = owner
        self.balance = balance
        self.history = []
    def deposit(self, amt):
        if amt <= 0:
            raise ValueError('bad amount')
        self.balance += amt
        self.history.append('+' + str(amt))
        return self.balance
    def withdraw(self, amt):
        if amt > self.balance:
            raise ValueError('insufficient')
        self.balance -= amt
        self.history.append('-' + str(amt))
        return self.balance
acc = Account('ann', 100)
print(acc.deposit(50), acc.withdraw(30), acc.balance)
try:
    acc.withdraw(1000)
except ValueError as e:
    print('err', e)
print(acc.history, len(acc.history))

### linked list class
class Node:
    def __init__(self, val):
        self.val = val
        self.next = None
class LinkedList:
    def __init__(self):
        self.head = None
    def append(self, v):
        n = Node(v)
        if self.head is None:
            self.head = n
            return
        cur = self.head
        while cur.next is not None:
            cur = cur.next
        cur.next = n
    def to_list(self):
        out = []
        cur = self.head
        while cur is not None:
            out.append(cur.val)
            cur = cur.next
        return out
    def length(self):
        cnt = 0
        cur = self.head
        while cur is not None:
            cnt += 1
            cur = cur.next
        return cnt
ll = LinkedList()
for v in [1, 2, 3, 4]:
    ll.append(v)
print(ll.to_list(), ll.length())
empty = LinkedList()
print(empty.to_list(), empty.length())
### bst insert and inorder
class TNode:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None
class BST:
    def __init__(self):
        self.root = None
    def insert(self, key):
        if self.root is None:
            self.root = TNode(key)
            return
        cur = self.root
        while True:
            if key < cur.key:
                if cur.left is None:
                    cur.left = TNode(key)
                    return
                cur = cur.left
            else:
                if cur.right is None:
                    cur.right = TNode(key)
                    return
                cur = cur.right
    def inorder(self):
        out = []
        def walk(n):
            if n is None:
                return
            walk(n.left)
            out.append(n.key)
            walk(n.right)
        walk(self.root)
        return out
t = BST()
for k in [5, 3, 8, 1, 4, 7, 9]:
    t.insert(k)
print(t.inorder())
### method chaining builder
class Builder:
    def __init__(self):
        self.parts = []
    def add(self, s):
        self.parts.append(s)
        return self
    def upper(self):
        self.parts = [p.upper() for p in self.parts]
        return self
    def result(self):
        return ' '.join(self.parts)
b = Builder()
r = b.add('hello').add('world').upper().add('end').result()
print(r)

### container dunder protocol
class Bag:
    def __init__(self, items):
        self.items = items
    def __len__(self):
        return len(self.items)
    def __contains__(self, x):
        return x in self.items
bg = Bag([1, 2, 3])
print(len(bg), 2 in bg, 9 in bg)
print(len(Bag([])), 'x' in Bag(['x', 'y']))
### templated game characters
class Character:
    def __init__(self, hp, atk):
        self.hp = hp
        self.atk = atk
    def hit(self, other):
        other.hp -= self.atk
        return other.hp <= 0
class Mage(Character):
    def __init__(self, hp, atk, mana):
        Character.__init__(self, hp, atk)
        self.mana = mana
    def fireball(self, other):
        self.mana -= 10
        dmg = self.atk * 2
        other.hp -= dmg
        return other.hp <= 0
m = Mage(50, 7, 30)
g = Character(40, 5)
print(m.hit(g), g.hp)
print(m.fireball(g), g.hp, m.mana)
