### probe any all genexp filter
def side(x):
    print('side', x)
    return x
print(any(side(x) > 2 for x in range(5) if x % 2 == 0))
print(all(side(x) >= 0 for x in range(3) if x != 1))
### probe any all genexp multi for
print(any(x + y == 3 for x in [1, 2] for y in [1, 2]))
print(all(x + y >= 2 for x in [1, 2] for y in [1, 2]))
print(any(x * y > 2 for x in [0, 1, 2] for y in [0, 1, 2]))
### probe any all range three args
print(any(x > 4 for x in range(0, 10, 3)))
print(all(x > 0 for x in range(9, 0, -3)))
print(any(x == 5 for x in range(10, 0, -2)))
