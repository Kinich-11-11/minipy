### M with 2 instance attrs
class M:
    def __init__(self, v, x, y):
        self.v = v
        self.x = x
        self.y = y
    def vals(self):
        return [self.v]
a1 = M(0, None, None)
b2 = M(2, None, None)
m1 = M(1, a1, b2)
print(m1.x.vals())
print(m1.y.vals())
### chained arg
class M2:
    def __init__(self, v):
        self.v = v
    def vals(self):
        return [self.v]
c1 = M2(0)
d2 = M2(2)
e1 = M2(1)
print(e1.v, c1.vals(), d2.vals())
