// ---- P79: dictionaries are writable -- d[k] = v and del d[k] (E38/E39) ----
//
// The `D` encoding is a flat run of (lsep + type + escaped text) records holding
// key, value, key, value, ...  indexget already reads it; indexset/indexdel only knew
// how to splice a list.  Both procedures now get a `D` branch that goes in *before* the
// list path and stops the script, so the list code stays byte-identical.
//
//   d[k] = v   find the key (flat index 2j-1, so the value sits at 2j) and splice the
//              value record; when the key is absent append key + value at the end.
//   del d[k]   drop the key record and its value record in one substring splice.
patch('P79 dict write and delete', () => {
  const litOf = (d) => {
    if (!d) return null;
    if (Array.isArray(d) && (d[0] === 10 || d[0] === 4) && typeof d[1] !== 'object') return String(d[1]);
    if (Array.isArray(d[1]) && (d[1][0] === 10 || d[1][0] === 4)) return String(d[1][1]);
    return null;
  };
  const blk = (d) => (d && typeof d[1] === 'string') ? BL()[d[1]] : null;
  const argName = (d) => { const b = blk(d); return b && b.opcode === 'argument_reporter_string_number' ? b.fields.VALUE[0] : null; };
  const argN = (b, i) => { const ids = JSON.parse(b.mutation.argumentids); return b.inputs[ids[i]]; };
  const isAddArg = (d, name, lit) => {
    const b = blk(d);
    return !!b && b.opcode === 'operator_add' && argName(b.inputs.NUM1) === name && litOf(b.inputs.NUM2) === lit;
  };
  const varName = (d) => (Array.isArray(d) && Array.isArray(d[1]) && d[1][0] === 12) ? d[1][1] : null;
  // the guard "if not(xb == 'L'): fail" runs *before* the key eval, so the dict branch has
  // to be spliced in front of that guard, not in front of the eval.
  const notLGuard = (set) => find(set, (b) => {
    if (b.opcode !== 'control_if' || !b.inputs.CONDITION || typeof b.inputs.CONDITION[1] !== 'string') return false;
    const nb = BL()[b.inputs.CONDITION[1]];
    if (!nb || nb.opcode !== 'operator_not') return false;
    const e = blk(nb.inputs.OPERAND);
    return !!e && e.opcode === 'operator_equals' && varName(e.inputs.OPERAND1) === 'xb' && litOf(e.inputs.OPERAND2) === 'L';
  });
  const STOPTHIS = () => I('control_stop', { fields: { STOP_OPTION: ['this script', null] }, mutation: { tagName: 'mutation', children: [], hasnext: 'false' } });
  const taken = new Set(Object.values(T(IT).variables).map((x) => x[0]));
  if (taken.has('dgf')) throw new Error('[P79] variable name taken: dgf');
  newVar('dgf', 0);
  // NB jf is a *thunk*: the flat index expression is needed twice, and reusing one block
  // id in two inputs is what broke P71 (one Scratch block has exactly one parent).
  const scanPos = (jf) => [
    setV('lcount', num(0)),
    setV('lstart', num(0)),
    setV('lend', num(0)),
    setV('lx', num(1)),
    REPEATU(GT(VAR('lx'), lenOf(VAR('xa'))), [
      IF(EQ(letterOf(VAR('lx'), VAR('xa')), VAR('lsep')), [
        chgV('lcount', num(1)),
        IF(EQ(VAR('lcount'), jf()), [setV('lstart', VAR('lx'))]),
        IF(EQ(VAR('lcount'), ADD(jf(), num(2))), [setV('lend', VAR('lx'))])
      ]),
      chgV('lx', num(1))
    ]),
    IF(EQ(VAR('lstart'), num(0)), [setV('lstart', num(1))])
  ];
  const itemSplice = () => [
    callProc('substr %s %s %s', [VAR('xa'), num(1), SUB(VAR('lstart'), num(1))]),
    setV('lnew', VAR('ev')),
    setV('lnew', JOIN(VAR('lnew'), JOIN(VAR('lsep'), JOIN(VAR('ltyp'), VAR('lval'))))),
    callProc('substr %s %s %s', [VAR('xa'), VAR('lend'), lenOf(VAR('xa'))]),
    setV('lnew', JOIN(VAR('lnew'), VAR('ev'))),
    callProc('assign %s %s %s', [item('Tok', argOf('lo')), VAR('lnew'), txt('D')]),
    setV('ev', VAR('lval')),
    setV('evtype', VAR('ltyp'))
  ];

  // ---------- (a) d[k] = v
  const ix = inDef('indexset %s %s');
  const keyGuard = notLGuard(ix);
  if (keyGuard.length !== 1) throw new Error('[P79] indexset dict guard = ' + keyGuard.length);
  const writeBranch = IF(EQ(VAR('xb'), txt('D')), [
    callProc('eval %s %s', [ADD(argOf('lo'), num(2)), SUB(VAR('lsp'), num(1))]),
    IF(NOT(EQ(VAR('error'), txt(''))), [STOPTHIS()]),
    setV('dgkey', VAR('ev')),
    setV('dgkt', VAR('evtype')),
    callProc('lesc %s', [VAR('dgkey')]),
    setV('dgesc', VAR('ev')),
    callProc('eval %s %s', [ADD(VAR('lsp'), num(2)), argOf('hi')]),
    IF(NOT(EQ(VAR('error'), txt(''))), [STOPTHIS()]),
    setV('lval', VAR('ev')),
    setV('ltyp', VAR('evtype')),
    callProc('lesc %s', [VAR('lval')]),
    setV('lval', VAR('ev')),
    callProc('listlen %s', [VAR('xa')]),
    setV('dgn', VAR('ev')),
    setV('dgi', num(1)),
    setV('dgf', num(0)),
    REPEATU(OR(GT(VAR('dgi'), VAR('dgn')), GT(VAR('dgf'), num(0))), [
      callProc('listitem %s %s', [VAR('xa'), VAR('dgi')]),
      IF(EQ(VAR('evtype'), VAR('dgkt')), [
        callProc('compareText %s %s', [VAR('ev'), VAR('dgkey')]),
        IF(EQ(VAR('cmp'), num(0)), [setV('dgf', VAR('dgi'))])
      ]),
      chgV('dgi', num(2))
    ]),
    IFELSE(GT(VAR('dgf'), num(0)), [
      ...scanPos(() => ADD(VAR('dgf'), num(1))),
      IF(EQ(VAR('lend'), num(0)), [setV('lend', ADD(lenOf(VAR('xa')), num(1)))]),
      ...itemSplice()
    ], [
      callProc('assign %s %s %s', [item('Tok', argOf('lo')),
        JOIN(JOIN(VAR('xa'), JOIN(VAR('lsep'), JOIN(VAR('dgkt'), VAR('dgesc')))),
             JOIN(VAR('lsep'), JOIN(VAR('ltyp'), VAR('lval')))), txt('D')]),
      setV('ev', VAR('lval')),
      setV('evtype', VAR('ltyp'))
    ]),
    STOPTHIS()
  ]);
  insertBeforeStmt(keyGuard[0], [writeBranch]);
  console.log('[P79] d[k] = v installed');

  // ---------- (b) del d[k]
  const idl = inDef('indexdel %s %s');
  const delGuard = notLGuard(idl);
  if (delGuard.length !== 1) throw new Error('[P79] indexdel dict guard = ' + delGuard.length);
  const delBranch = IF(EQ(VAR('xb'), txt('D')), [
    callProc('eval %s %s', [ADD(argOf('lo'), num(3)), SUB(VAR('lsp'), num(1))]),
    IF(NOT(EQ(VAR('error'), txt(''))), [STOPTHIS()]),
    setV('dgkey', VAR('ev')),
    setV('dgkt', VAR('evtype')),
    callProc('listlen %s', [VAR('xa')]),
    setV('dgn', VAR('ev')),
    setV('dgi', num(1)),
    setV('dgf', num(0)),
    REPEATU(OR(GT(VAR('dgi'), VAR('dgn')), GT(VAR('dgf'), num(0))), [
      callProc('listitem %s %s', [VAR('xa'), VAR('dgi')]),
      IF(EQ(VAR('evtype'), VAR('dgkt')), [
        callProc('compareText %s %s', [VAR('ev'), VAR('dgkey')]),
        IF(EQ(VAR('cmp'), num(0)), [setV('dgf', VAR('dgi'))])
      ]),
      chgV('dgi', num(2))
    ]),
    IF(EQ(VAR('dgf'), num(0)), [FAILB(JOIN(txt('KeyError: '), VAR('dgkey')))]),
    IF(EQ(VAR('error'), txt('')), [
      setV('lcount', num(0)),
      setV('lstart', num(0)),
      setV('lend', num(0)),
      setV('lx', num(1)),
      REPEATU(GT(VAR('lx'), lenOf(VAR('xa'))), [
        IF(EQ(letterOf(VAR('lx'), VAR('xa')), VAR('lsep')), [
          chgV('lcount', num(1)),
          IF(EQ(VAR('lcount'), VAR('dgf')), [setV('lstart', VAR('lx'))]),
          IF(EQ(VAR('lcount'), ADD(VAR('dgf'), num(2))), [setV('lend', VAR('lx'))])
        ]),
        chgV('lx', num(1))
      ]),
      IF(EQ(VAR('lstart'), num(0)), [setV('lstart', num(1))]),
      IF(EQ(VAR('lend'), num(0)), [setV('lend', ADD(lenOf(VAR('xa')), num(1)))]),
      callProc('substr %s %s %s', [VAR('xa'), num(1), SUB(VAR('lstart'), num(1))]),
      setV('lnew', VAR('ev')),
      callProc('substr %s %s %s', [VAR('xa'), VAR('lend'), lenOf(VAR('xa'))]),
      setV('lnew', JOIN(VAR('lnew'), VAR('ev'))),
      callProc('assign %s %s %s', [item('Tok', ADD(argOf('lo'), num(1))), VAR('lnew'), txt('D')]),
      setV('ev', txt('')),
      setV('evtype', txt('Z'))
    ]),
    STOPTHIS()
  ]);
  insertBeforeStmt(delGuard[0], [delBranch]);
  console.log('[P79] del d[k] installed');
});
