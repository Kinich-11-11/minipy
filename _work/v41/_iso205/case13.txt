### recursive_flatten
def flatten(x):
    out = []
    for item in x:
        if isinstance(item, list):
            for sub in flatten(item):
                out.append(sub)
        else:
            out.append(item)
    return out
print(flatten([1, [2, [3, [4, 5]], 6], 7]))
print(flatten([]), flatten([[[[]]]]))
print(flatten([1, [2], [[3], 4], 'ab']))
