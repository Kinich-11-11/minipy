### palindrome checker
def clean(s):
    out = ''
    for ch in s.lower():
        if ch.isalnum():
            out += ch
    return out
def is_pal(s):
    c = clean(s)
    return c == c[::-1]
print(is_pal('A man, a plan, a canal: Panama'), is_pal('race a car'), is_pal(''))
### word frequency
text = 'the cat and the dog and the bird'
freq = {}
for w in text.split():
    if w in freq:
        freq[w] += 1
    else:
        freq[w] = 1
items = sorted(freq.items(), key=lambda kv: -kv[1])
print(items)
### anagram grouping
words = ['listen', 'silent', 'enlist', 'google', 'banana']
groups = {}
for w in words:
    k = ''.join(sorted(w))
    if k in groups:
        groups[k].append(w)
    else:
        groups[k] = [w]
print(sorted(groups.values(), key=len, reverse=True))
