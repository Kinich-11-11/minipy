class A:
    def who(self):
        return 'A'
    def hello(self):
        return 'hello ' + self.who()
class B(A):
    def who(self):
        return 'B'
class C(B):
    def __init__(self):
        self.tag = 'cee'
    def who(self):
        return 'C/' + super().who()
print(A().hello(), B().hello(), C().hello(), C().tag)
