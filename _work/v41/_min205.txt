### M1 local list assign then append in for
def f(a):
    out = []
    for x in a:
        out.append(x)
    return out
print(f([1, 2, 3]))
### M2 gcd style while tuple assign
def g(a, b):
    while b:
        a, b = b, a % b
    return a
print(g(48, 36))
### M3 tuple assign in loop then mod
def g2(a, b):
    while b:
        t = a % b
        a = b
        b = t
    return a
print(g2(48, 36), g2(4, 6))

### M6 class add returns new instance
class V:
    def __init__(self, x):
        self.x = x
    def __add__(self, o):
        return V(self.x + o.x)
a = V(1)
b = V(2)
c = a + b
print(c.x)
### M7 method call after class attr bump

### M8 instance attr holding object then method call
class E:
    def go(self):
        return 'ego'
class W:
    def __init__(self):
        self.e = E()
    def drive(self):
        return self.e.go()
w = W()
print(w.drive())
### M9 is None check
class N:
    def __init__(self):
        self.h = None
    def set(self, v):
        if self.h is None:
            self.h = v
            return 'set'
        return 'kept'
n = N()
print(n.set(1), n.set(2), n.h)

### M16 two arg lambda inline in call
def fold(fn, xs, init):
    acc = init
    for x in xs:
        acc = fn(acc, x)
    return acc
print(fold(lambda a, b: a + b, [1, 2, 3], 0))
### M17 lambda single arg in call works
print(fold(lambda a: a * 10, [1, 2], 0))

### M19 exception base init
class E2(Exception):
    def __init__(self, code):
        Exception.__init__(self, 'msg' + str(code))

    raise E2(7)

    print(e.code, str(e))
