### with_inside_handler
class Manager:
    def __enter__(self):
        print('enter')
        return self
    def __exit__(self,t,e,tb):
        print('exit',t.__name__)
        return True
try:
    raise ExceptionGroup('g',[ValueError('v'), TypeError('t')])
except* ValueError:
    with Manager():
        raise RuntimeError('r')
    print('after')
except* TypeError:
    print('T')
