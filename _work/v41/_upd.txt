### update from list of pairs
d = {'a': 1}
d.update([('b', 2)])
print(d)
### update from list of pairs new and existing
d = {'a': 1}
d.update([('a', 9), ('b', 2)])
print(d)
### update from tuple of pairs
d = {'a': 1}
d.update((('b', 2), ('c', 3)))
print(d)
### update from dict still works
d = {'a': 1}
d.update({'b': 2})
print(d)
### update from list of lists
d = {}
d.update([['x', 1]])
print(d)
### update keyword args
d = {'a': 1}
d.update(b=2)
print(d)
### update bad length pairs
d = {}
try:
    d.update([('a', 1, 2)])
except ValueError as e:
    print('VE', e)
### update non sequence element
d = {}
try:
    d.update([5])
except TypeError as e:
    print('TE', e)
### update empty list
d = {'a': 1}
d.update([])
print(d)
### update of dict with mixed value types
d = {'a': 1}
d.update([('b', [1, 2]), ('c', 'x')])
print(d)
### update result is None
d = {}
r = d.update([('a', 1)])
print(r, d)
### update from zip
d = {}
d.update(zip('ab', [1, 2]))
print(d)
