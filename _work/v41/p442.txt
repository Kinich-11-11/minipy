// ---- P442: `sorted` 的 `__lt__`（先给 calluser 的"按区间建实参表"加守卫，再在 lcmp 分派）----
//
// 第 173/174 轮的结论：
//   * `lcmp` 是对的分派点（探针：`LC lcat=O lca=[1] lcbt=O lcb=[2]`），
//     但 `calluser` **无论如何都会先按区间建 `UArgV/UArgT`**（求值）⇒ 传空区间 `(1,0)` 会
//     `SyntaxError: parentheses`，传有效一格又会把那格 token 当实参求值（`NameError: print`）。
//   * 因此先给 `calluser` 的"建表段"加守卫：**值槽武装（`vmOn==1`）时整段跳过**。
//
// 本补丁两件事：
//   1. 在 `calluser` 里，把 **5 处 `addtolist UArgV`（剔除值槽自身那处）的最外层公共祖先语句**
//      用 `IF(NOT(vmOn == 1))` 包起来（只是把那条语句搬进分支，不改其它语句）；
//   2. 在 `lcmp` 最前面武装值槽（绑 self → `vmA=lcb`、`vmT=lcbt`、`vmOn=1` → `calluser(cfull,1,0)`；
//      恢复 self；存 `lcres`/`lcdone`；把 `lca/lcat/lcb/lcbt` 中和成 `N/N` 的 0），
//      并在 `lcmp` 链尾用 `cmp = (lcres==1) ? -1 : 1` 覆盖（**不用 stop this script**）。
patch('P442 sorted __lt__ (guard calluser build + lcmp dispatch)', () => {
  const BB = T(IT).blocks;
  const taken = new Set(Object.values(T(IT).variables).map((x) => x[0]));
  for (const n of ['lcres', 'lcdone']) if (taken.has(n)) throw new Error('[P442] var taken: ' + n);
  newVar('lcres', ''); newVar('lcdone', 0);
  const lastItem = (list) => I('data_itemoflist', { fields: B.LISTF(list), inputs: { INDEX: B.REP(llen(list), [10, '']) } });
  const popList = (list) => I('data_deleteoflist', { fields: B.LISTF(list), inputs: { INDEX: B.REP(txt('last'), [10, '']) } });
  // ---------- 1) calluser：给"建表段"加守卫 ----------
  {
    const cu = defIdOf('calluser %s %s %s');
    // 收集写入 UArgV 的位置及其祖先链
    const chains = [];
    const walk = (id, chain) => {
      let c = id;
      while (c) {
        const q = BB[c];
        if (!q) break;
        if (q.opcode === 'data_addtolist' && q.fields && q.fields.LIST && q.fields.LIST[0] === 'UArgV') chains.push({ id: c, chain: chain.slice() });
        for (const [k, v] of Object.entries(q.inputs || {})) {
          if ((k === 'SUBSTACK' || k === 'SUBSTACK2') && typeof v[1] === 'string') walk(v[1], chain.concat([c]));
        }
        c = q.next;
      }
    };
    walk(BB[cu].next, []);
    // 剔除"值槽自身"那处：其祖先里有 IFELSE，条件文本含 lengthoflist(UArgV)
    const isOwn = (ch) => ch.chain.some((a) => {
      const w = BB[a];
      return w && w.inputs && w.inputs.CONDITION && JSON.stringify(w.inputs.CONDITION).indexOf('lengthoflist') >= 0
        && JSON.stringify(w.inputs.CONDITION).indexOf('UArgV') >= 0;
    });
    const builds = chains.filter((c) => !isOwn(c));
    if (builds.length < 4) throw new Error('[P442] expected >=4 build sites, got ' + builds.length);
    console.log('[P442] build sites=' + builds.length + ' (all=' + chains.length + ')');
    // 最外层公共祖先
    let common = null;
    const first = builds[0].chain;
    for (let i = first.length - 1; i >= 0; i--) {
      const cand = first[i];
      if (builds.every((b) => b.chain.indexOf(cand) >= 0)) { common = cand; break; }
    }
    if (!common) throw new Error('[P442] no common ancestor for the build sites');
    console.log('[P442] common ancestor=' + common + ' :: ' + BB[common].opcode);
    // 把该语句包进 IF(NOT(vmOn==1))
    const after = BB[common].next;
    let pred = null, slot = null, slotParent = null;
    for (const id of Object.keys(BB)) {
      const q = BB[id];
      if (!q) continue;
      if (q.next === common) { pred = id; break; }
      for (const [k, v] of Object.entries(q.inputs || {})) {
        if ((k === 'SUBSTACK' || k === 'SUBSTACK2') && typeof v[1] === 'string' && v[1] === common) { slot = k; slotParent = id; }
      }
    }
    const guard = IF(NOT(EQ(VAR('vmOn'), num(1))), [common]);
    BB[common].next = null;
    BB[common].parent = guard;
    BB[guard].next = after;
    if (after) BB[after].parent = guard;
    if (pred) { BB[pred].next = guard; BB[guard].parent = pred; }
    else if (slotParent) { BB[slotParent].inputs[slot] = [2, guard]; BB[guard].parent = slotParent; }
    else throw new Error('[P442] no predecessor for the build ancestor');
    console.log('[P442] calluser build guard installed');
  }
  // ---------- 2) lcmp 分派（开头武装 + 结尾覆盖） ----------
  {
    const hasLt = I('data_listcontainsitem', {
      fields: B.LISTF('FnName'),
      inputs: { ITEM: B.REP(JOIN(item('InstCls', VAR('lca')), txt('.__lt__')), [10, '']) }
    });
    const def = defIdOf('lcmp');
    const head = BB[def].next;
    if (!head) throw new Error('[P442] lcmp body missing');
    let last = head;
    while (BB[last].next) last = BB[last].next;
    insertBeforeStmt(head, [
      IF(AND(EQ(VAR('lcat'), txt('O')), hasLt), [
        callProc('lookup %s', [txt('self')]),
        setV('s1s', VAR('found')),
        IF(GT(VAR('s1s'), num(0)), [addTo('VStack', VAR('self'))]),
        setV('self', VAR('lca')),
        callProc('assign %s %s %s', [txt('self'), VAR('lca'), txt('O')]),
        setV('vmA', VAR('lcb')),
        setV('vmT', VAR('lcbt')),
        setV('vmOn', num(1)),
        callProc('calluser %s %s %s', [VAR('cfull'), B.REP(num(1), [4, '0']), B.REP(num(0), [4, '0'])]),
        setV('clf', lastItem('VStack')),
        popList('VStack'),
        setV('self', VAR('clf')),
        setV('lcres', VAR('ev')),
        setV('lcdone', num(1)),
        setV('lca', num(0)), setV('lcat', txt('N')),
        setV('lcb', num(0)), setV('lcbt', txt('N'))
      ])
    ]);
    spliceAfter(last, chain([
      IF(EQ(VAR('lcdone'), num(1)), [
        setV('cmp', num(1)),
        IF(EQ(VAR('lcres'), num(1)), [setV('cmp', num(-1))]),
        setV('lcdone', num(0))
      ])
    ]));
    console.log('[P442] lcmp dispatch installed');
  }
  console.log('[P442] done');
});
