// ---- P119: methodcall must publish the receiver it just evaluated ----
//
// P20 rewrote methodcall's receiver step as
//     if mcpre == 1 then xa = mcval; xb = mctype else eval(lo, lo)
// -- and the `else` arm never copies the evaluated receiver into `xa`/`xb`.  Every consumer
// (list methods, string methods, P91's dict methods, P55's attribute access) reads `xa`, so a
// plain `d.get(k)` was answering about whatever receiver the *previous* method call happened to
// leave there.  It usually goes unnoticed because a chain of calls on the same object agrees with
// itself; it shows up as soon as the object changes between calls (pop, then get).
//
// Fix: append the copy inside the plain arm.
patch('P119 methodcall receiver', () => {
  const mc = inDef('methodcall %s %s');
  const evals = find(mc, (b) => b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === 'eval %s %s'
    && Object.values(b.inputs).every((v) => typeof v[1] === 'string' && BL()[v[1]] && BL()[v[1]].opcode === 'argument_reporter_string_number' && BL()[v[1]].fields.VALUE[0] === 'lo'));
  if (evals.length !== 1) throw new Error('[P119] receiver eval = ' + evals.length);
  const evalCall = evals[0];
  const parent = BL()[evalCall].parent;
  if (!parent || BL()[parent].opcode !== 'control_if_else') throw new Error('[P119] receiver eval is not in an if-else');
  const slot = BL()[parent].inputs.SUBSTACK2 && BL()[parent].inputs.SUBSTACK2[1];
  if (slot !== evalCall) throw new Error('[P119] receiver eval is not the else arm');
  spliceAfter(evalCall, chain([setV('xa', VAR('ev')), setV('xb', VAR('evtype'))]));
  console.log('[P119] methodcall publishes the evaluated receiver');
});
