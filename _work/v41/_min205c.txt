### P6 ll to_list method
class Nd6:
    def __init__(self, val):
        self.val = val
        self.next = None
class LL6:
    def __init__(self):
        self.head = None
    def append(self, v):
        n = Nd6(v)
        if self.head is None:
            self.head = n
            return
        cur = self.head
        while cur.next is not None:
            cur = cur.next
        cur.next = n
    def to_list(self):
        out = []
        cur = self.head
        while cur is not None:
            out.append(cur.val)
            cur = cur.next
        return out
ll = LL6()
ll.append(1)
ll.append(2)
print(ll.to_list())
