// ---- P396: `pow(x)` 缺参时报 TypeError ----
//
// 第 142 轮探针（p395）决定性结论：**`narg` 是"顶层逗号数"，不是参数个数**。
//   round(1.5) / int(2.7) / pow(2) / max([1,2]) / min([3,1]) / abs(-3)
//   全部打印 `B[N0] … narg=0` ⇒ 单参数内建统统走 `narg=="0"` 这条分支（`bcontro_nd1f7pkr`），
//   而 `narg=="1"` 那条（`bcontro_nc7o8fzy`，条件里含 pow/round/int）**根本没被走到**
//   —— 这就是第 141 轮把守卫插错地方的原因。
//
// 修法：把守卫插到 `narg=="0"` 分支的 **SUBSTACK 头部**（用"前插到分支"的写法）：
//     IF(Tok[lo]=="pow", [ fail("TypeError: pow() missing required argument 'exp' (pos 2)") ])
//   * 该分支里 round/int/max/min/abs 都不叫 pow ⇒ 完全不受影响；
//   * `pow(2)` ⇒ 报 TypeError（而不是退回变量查找报 NameError）✅
patch('P396 pow() missing argument (narg==0 branch)', () => {
  const BB = T(IT).blocks;
  const MSG = "TypeError: pow() missing required argument 'exp' (pos 2)";
  const collect = (v, out) => {
    if (!Array.isArray(v)) return;
    if (v[0] === 12 && typeof v[1] === 'string') { out.vars.push(v[1]); return; }
    if (typeof v[1] === 'string' && BB[v[1]]) {
      const q = BB[v[1]];
      if (q.fields && q.fields.VARIABLE) out.vars.push(q.fields.VARIABLE[0]);
      for (const k of Object.keys(q.fields || {})) { const f = q.fields[k]; if (Array.isArray(f) && typeof f[0] === 'string' && f[0] !== '') out.lits.push(f[0]); }
      for (const x of Object.values(q.inputs || {})) collect(x, out);
      return;
    }
    if ((v[0] === 10 || v[0] === 4) && typeof v[1] !== 'object') { out.lits.push(String(v[1])); return; }
    if (Array.isArray(v[1])) collect(v[1], out);
  };
  let evalDef = null;
  for (const id of Object.keys(BB)) {
    const b = BB[id];
    if (b.opcode !== 'procedures_definition' || !b.inputs) continue;
    const pr = BB[b.inputs.custom_block[1]];
    if (pr && String(pr.mutation.proccode) === 'eval %s %s') { evalDef = id; break; }
  }
  if (!evalDef) throw new Error('[P396] eval not found');
  const inEval = {};
  const walk = (id) => { let c = id; while (c) { const q = BB[c]; inEval[c] = 1; for (const [k, v] of Object.entries(q.inputs || {})) if ((k === 'SUBSTACK' || k === 'SUBSTACK2') && typeof v[1] === 'string') walk(v[1]); c = q.next; } };
  walk(BB[evalDef].next);
  let n0 = null;
  for (const id of Object.keys(inEval)) {
    const q = BB[id];
    if (q.opcode !== 'control_if_else' || !q.inputs || !q.inputs.CONDITION) continue;
    const o = { vars: [], lits: [] };
    collect(q.inputs.CONDITION, o);
    if (o.vars.indexOf('narg') >= 0 && o.lits.indexOf('0') >= 0 && o.lits.length <= 2) { n0 = id; break; }
  }
  if (!n0) throw new Error('[P396] narg==0 branch not found');
  console.log('[P396] narg==0 branch = ' + n0);
  const old = BB[n0].inputs.SUBSTACK;
  if (!old) throw new Error('[P396] SUBSTACK missing');
  const oldFirst = old[1];
  const guard = IF(EQ(item('Tok', argOf('lo')), txt('pow')), [
    callProc('fail %s', [txt(MSG)])
  ]);
  BB[n0].inputs.SUBSTACK = [2, guard];
  BB[guard].parent = n0;
  BB[guard].next = oldFirst;
  if (oldFirst) BB[oldFirst].parent = guard;
  console.log('[P396] pow(x) guard prepended to the narg==0 branch');
});
