### n1 next twice
it = iter([1, 2, 3])
print(next(it), next(it), next(it))
### n2 stop iteration
it = iter([1])
print(next(it))
try:
    print(next(it))
except StopIteration:
    print('stop')
### n3 list of iterator
it = iter([1, 2])
print(list(it))
### n4 for over iterator
for x in iter([1, 2]):
    print(x)
### n5 iter string twice
it = iter('ab')
print(next(it), next(it))
### n6 exhausted then iter again
it = iter([7])
print(next(it))
it2 = iter([7])
print(next(it2))
