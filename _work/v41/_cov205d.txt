### closure counter
def make_counter():
    n = 0
    def inc():
        nonlocal n
        n += 1
        return n
    return inc
a = make_counter()
b = make_counter()
print(a(), a(), a(), b(), a())

### generator pipeline
def evens(n):
    i = 0
    while i < n:
        yield i
        i += 2
def double(gen):
    for v in gen:
        yield v * 2
def take(gen, k):
    cnt = 0
    for v in gen:
        if cnt >= k:
            return
        yield v
        cnt += 1
print(list(evens(7)))
print(list(double(evens(5))))
print(list(take(double(evens(10)), 3)))
### recursive flatten
def flatten(x):
    out = []
    for item in x:
        if isinstance(item, list):
            for sub in flatten(item):
                out.append(sub)
        else:
            out.append(item)
    return out
print(flatten([1, [2, [3, [4, 5]], 6], 7]))
print(flatten([]), flatten([[[[]]]]))
print(flatten([1, [2], [[3], 4], 'ab']))
### manual fold reduce
def fold(fn, xs, init):
    acc = init
    for x in xs:
        acc = fn(acc, x)
    return acc
print(fold(lambda a, b: a + b, [1, 2, 3, 4], 0))
print(fold(lambda a, b: a * b, [1, 2, 3, 4], 1))
print(fold(lambda a, b: a if a > b else b, [3, 9, 2], 0))
print(fold(lambda a, b: a + [b], [1, 2, 3], []))

### decorator manual wrap
def shout(fn):
    def wrapped(x):
        return '!!' + str(fn(x)) + '!!'
    return wrapped
def double(x):
    return x * 2
loud = shout(double)
print(loud(21))
def compose(f, g):
    return lambda x: f(g(x))
inc = lambda x: x + 1
print(compose(double, inc)(5), compose(inc, double)(5))
### partial application manual
def partial(fn, *fixed):
    def applied(*rest):
        all_args = []
        for a in fixed:
            all_args.append(a)
        for a in rest:
            all_args.append(a)
        return fn(all_args)
    return applied
def mysum(args):
    return sum(args)
f = partial(mysum, 1, 2)
print(f(3, 4), f(), f(100))
