// ---- P605: NaN 比较语义 + inf/nan 算术（覆盖报告 P1-3）----
//
// (1) 比较：rightside 的通用比较分支里，任一侧是 "nan"/"NaN"（N 型）时
//     cmp=2——对 ==/!=/</>/<=/>= 六种算子恰好给出 CPython 语义（==False、
//     !=True、其余全 False）。
// (2) 算术：Scratch Cast.toNumber("inf")=0、"nan"=0，inf-1e308 之类全变 0。
//     新过程 fltop：N-N 算术任一侧是 inf/-inf/nan 时按 CPython 语义给结果
//     （nan 传染；inf±inf 同号/异号；inf*0=nan；inf/x、x/inf；x%inf=x；
//     inf%x=nan）。挂点：rightside bigHook 之后（J 路径已 stop，不会到达）。
// 声明边界：单目负号对 inf（0-0=0）、负 inf 与负数的大小比较（字符串序）。
patch('P605 nan semantics', () => {
  newVar('fasg', 0);
  newVar('fbth', 0);
  const STOP = () => I('control_stop', { fields: { STOP_OPTION: ['this script', null] }, mutation: { tagName: 'mutation', children: [], hasnext: 'false' } });
  const CONT = (x, lit) => I('operator_contains', { inputs: { STRING1: B.REP(x, [10, '']), STRING2: txt(lit) } });
  // ---- fltop 过程 ----
  const d = newProc('fltop %s %s %s', ['op', 'a', 'b']);
  const A = argOf('a'), Bop = argOf('b'), OP = argOf('op');
  const isNan = (x) => OR(EQ(x, txt('nan')), EQ(x, txt('NaN')));
  const isInf = (x) => OR(EQ(x, txt('inf')), EQ(x, txt('-inf')));
  const isNeg = (x) => EQ(letterOf(num(1), x), txt('-'));
  const isZero = (x) => EQ(x, txt('0'));
  const negbit = (x) => AND(isNeg(x), num(1));
  const setNan = () => [setV('ev', txt('nan')), setV('evtype', txt('N')), setV('evK', txt('')), STOP()];
  const setInf = (negExpr) => [
    setV('ev', txt('inf')),
    IF(negExpr, [setV('ev', txt('-inf'))]),
    setV('evtype', txt('N')),
    setV('evK', txt('')),
    STOP()
  ];
  attachBody(d.defId, [
    IF(OR(isNan(A), isNan(Bop)), setNan()),
    setV('fasg', MOD(ADD(negbit(A), negbit(Bop)), num(2))),
    setV('fbth', AND(AND(isInf(A), isInf(Bop)), num(1))),
    IF(EQ(OP, txt('+')), [
      IFELSE(AND(EQ(VAR('fbth'), num(1)), EQ(VAR('fasg'), num(1))), setNan(), [
        IFELSE(EQ(VAR('fbth'), num(1)), setInf(isNeg(A)), [
          IFELSE(isInf(A), setInf(isNeg(A)), setInf(isNeg(Bop)))
        ])
      ]),
      STOP()
    ]),
    IF(EQ(OP, txt('-')), [
      IFELSE(AND(EQ(VAR('fbth'), num(1)), EQ(VAR('fasg'), num(0))), setNan(), [
        IFELSE(EQ(VAR('fbth'), num(1)), setInf(isNeg(A)), [
          IFELSE(isInf(A), setInf(isNeg(A)), setInf(NOT(isNeg(Bop))))
        ])
      ]),
      STOP()
    ]),
    IF(EQ(OP, txt('*')), [
      IFELSE(OR(isZero(A), isZero(Bop)), setNan(), setInf(EQ(VAR('fasg'), num(1)))),
      STOP()
    ]),
    IF(OR(EQ(OP, txt('/')), EQ(OP, txt('//'))), [
      IFELSE(EQ(VAR('fbth'), num(1)), setNan(), [
        IFELSE(isInf(A), setInf(isNeg(A)), [
          setV('ev', txt('0')),
          setV('evtype', txt('N')),
          setV('evK', txt('')),
          callProc('mkfloat %s', [VAR('ev')]),
          STOP()
        ])
      ]),
      STOP()
    ]),
    IF(EQ(OP, txt('%')), [
      IFELSE(isInf(A), setNan(), [
        setV('ev', A),
        setV('evtype', txt('N')),
        setV('evK', txt('')),
        STOP()
      ])
    ])
  ]);
  // ---- 挂点：rightside bigHook 之后 ----
  const rs = inDef('rightside %s %s %s %s %s');
  const bgCalls = find(rs, (b) => b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === 'bgops %s %s %s %s %s');
  if (bgCalls.length !== 1) throw new Error('[P605] bgops call = ' + bgCalls.length);
  const hookIf = BL()[bgCalls[0]].parent;
  if (!hookIf || BL()[hookIf].opcode !== 'control_if') throw new Error('[P605] bigHook if not found (parent=' + hookIf + ')');
  if (!BL()[hookIf].next) throw new Error('[P605] bigHook has no next');
  const opin = OR(
    OR(OR(EQ(argOf('op'), txt('+')), OR(EQ(argOf('op'), txt('-')), OR(EQ(argOf('op'), txt('*')), EQ(argOf('op'), txt('/')))))),
    OR(EQ(argOf('op'), txt('//')), EQ(argOf('op'), txt('%'))));
  const isinfv = (x) => OR(OR(EQ(x, txt('inf')), EQ(x, txt('-inf'))), isNan(x));
  const cond = AND(opin, AND(AND(EQ(VAR('lt2'), txt('N')), EQ(VAR('evtype'), txt('N'))), OR(isinfv(argOf('leftval')), isinfv(VAR('ev')))));
  const fltopIf = IF(cond, [
    callProc('fltop %s %s %s', [argOf('op'), argOf('leftval'), VAR('ev')]),
    STOP()
  ]);
  insertBeforeStmt(BL()[hookIf].next, [fltopIf]);
  // ---- (1) 比较 cmp=2 ----
  const cmpSets = find(rs, (b) => {
    if (!(b.opcode === 'data_setvariableto' && b.fields.VARIABLE
      && b.fields.VARIABLE[0] === 'cmp' && b.next
      && (BL()[b.next].opcode === 'control_if' || BL()[b.next].opcode === 'control_if_else')
      && BL()[b.next].inputs.CONDITION && typeof BL()[b.next].inputs.CONDITION[1] === 'string')) return false;
    const cj = [...reach(BL()[b.next].inputs.CONDITION[1])].map((x) => JSON.stringify(BL()[x])).join('');
    return cj.includes('"lt2"') && cj.includes('"evtype"') && cj.includes('"S"');
  });
  if (cmpSets.length !== 1) throw new Error('[P605] cmp=0 sites = ' + cmpSets.length);
  // nanhit 标志 + cmp=2（cmp=2 单独只能压住 </<=，>/>= 仍会误真）
  newVar('nanhit', 0);
  const nanSet = IF(AND(EQ(VAR('lt2'), txt('N')), AND(EQ(VAR('evtype'), txt('N')), OR(isNan(argOf('leftval')), isNan(VAR('ev'))))), [
    setV('nanhit', num(1)),
    setV('cmp', num(2))
  ]);
  spliceAfter(cmpSets[0], setV('nanhit', num(0)));
  spliceAfter(BL()[cmpSets[0]].next, nanSet);
  // 终值 if（条件引用 cmp 且 ev=1）包上 NOT(nanhit)——NaN 六种比较全 False
  const bigIfs = find(rs, (b) => {
    if (b.opcode !== 'control_if' || !b.inputs.CONDITION || typeof b.inputs.CONDITION[1] !== 'string') return false;
    if (!b.inputs.SUBSTACK || typeof b.inputs.SUBSTACK[1] !== 'string') return false;
    const cj = [...reach(b.inputs.CONDITION[1])].map((x) => JSON.stringify(BL()[x])).join('');
    if (!cj.includes('"cmp"') || !cj.includes('"=="')) return false;
    const sj = [...reach(b.inputs.SUBSTACK[1])];
    return sj.some((x) => BL()[x].opcode === 'data_setvariableto' && BL()[x].fields.VARIABLE
      && BL()[x].fields.VARIABLE[0] === 'ev' && String(BL()[x].inputs.VALUE[1][1]) === '1');
  });
  if (bigIfs.length !== 1) throw new Error('[P605] ev=1 big if = ' + bigIfs.length);
  const bigIf = bigIfs[0];
  const origCond = BL()[bigIf].inputs.CONDITION[1];
  // NaN：六种比较全 False、!= 为 True——终值条件 AND 上
  // (非 nan 或 op=='!=')：== cmp=2→False；!= not(cmp==0)→True；>/>= 被 nanhit 压住
  const newCond = AND(origCond, OR(NOT(EQ(VAR('nanhit'), num(1))), EQ(argOf('op'), txt('!='))));
  const andBlk = newCond; // AND(...) 根块
  BL()[origCond].parent = andBlk;
  BL()[andBlk].parent = bigIf;
  BL()[bigIf].inputs.CONDITION = [2, andBlk];
  console.log('[P605] nan semantics installed');
});
