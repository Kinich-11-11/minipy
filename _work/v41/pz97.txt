// PZ97（临时调试）：在 callfn 里那两个"守卫 = Tok[p18lo] == '**'"的 eval 调用点之前，
// 打印 p18lo / Tok[p18lo] / p18hi / Tok[p18hi] / count，看守卫为什么会误判为真
patch('PZ97 callfn ** guard values', () => {
  const BB = T(IT).blocks;
  const ownerOf = (id) => {
    let cur = id;
    for (let i = 0; i < 400 && cur; i++) {
      const q = BB[cur];
      if (!q) return '?';
      if (q.opcode === 'procedures_definition') {
        const pr = BB[q.inputs.custom_block[1]];
        return pr && pr.mutation ? String(pr.mutation.proccode) : '?';
      }
      cur = q.parent;
    }
    return '(toplevel)';
  };
  const nameOf = (v) => {
    if (!Array.isArray(v)) return null;
    if (v[0] === 12 && typeof v[1] === 'string') return v[1];
    const id = typeof v[1] === 'string' ? v[1] : null;
    const q = id ? BB[id] : null;
    if (q && q.opcode === 'data_variable') return q.fields.VARIABLE[0];
    if (q && q.opcode === 'argument_reporter_string_number' && q.fields.VALUE) return '#' + q.fields.VALUE[0];
    if (Array.isArray(v[1])) return nameOf(v[1]);
    return null;
  };
  let n = 0;
  for (const id of Object.keys(BB)) {
    const b = BB[id];
    if (b.opcode !== 'procedures_call' || !b.mutation) continue;
    if (String(b.mutation.proccode) !== 'eval %s %s') continue;
    const ow = ownerOf(id);
    if (ow.indexOf('callfn') < 0) continue;
    const vals = Object.values(b.inputs || {});
    if (vals.length < 2) continue;
    if (nameOf(vals[0]) !== 'p18lo' || nameOf(vals[1]) !== 'p18hi') continue;
    n++;
    const tag = 'G' + n;
    insertBeforeStmt(id, [
      callProc('log %s', [JOIN(txt(tag + ' p18lo='), VAR('p18lo'))]),
      callProc('log %s', [JOIN(txt(tag + ' Tok[p18lo]='), item('Tok', VAR('p18lo')))]),
      callProc('log %s', [JOIN(txt(tag + ' p18hi='), VAR('p18hi'))]),
      callProc('log %s', [JOIN(txt(tag + ' Tok[p18hi]='), item('Tok', VAR('p18hi')))]),
      callProc('log %s', [JOIN(txt(tag + ' count='), VAR('count'))]),
      callProc('log %s', [JOIN(txt(tag + ' hi='), argOf('hi'))]),
      callProc('log %s', [JOIN(txt(tag + ' lo='), argOf('lo'))])
    ]);
    console.log('[PZ97] marked ' + tag + ' @' + id);
  }
  if (!n) throw new Error('[PZ97] no callfn(p18lo,p18hi) eval site');
});
