### format negative and sign
print(f'{-3.14159:10.2f}|', f'{-3.14159:+.2f}', f'{3.0:+d}', f'{3.0: d}')
### format floats width and fill
print(f'{3.14159:10.3f}|', f'{3.14159:<10.3f}|', f'{3.14159:^10.3f}|', f'{3.14159:010.3f}')
### pow one arg
print(pow(2))
### pow three args
print(pow(2, 3, 5))
### pow single arg
print(pow(3))
### pow builtin listed in docs
print(pow(2, 3))
### assert with message
assert 1 == 2, 'msg'
### tuple of generator
print(tuple(x for x in range(3)))
