patch('P473 log every fail message', () => {
  const BB = T(IT).blocks;
  const LOGX = (expr) => callProc('log %s', [expr]);
  const targets = [];
  for (const id of Object.keys(BB)) {
    const q = BB[id];
    if (q.opcode !== 'procedures_call' || !q.mutation) continue;
    if (String(q.mutation.proccode) !== 'fail %s') continue;
    targets.push(id);
  }
  console.log('[P473] fail sites=' + targets.length);
  let skipped = 0;
  for (const id of targets) {
    const q = BB[id];
    const keys = Object.keys(q.inputs || {});
    if (!keys.length) continue;
    const arg = q.inputs[keys[0]];               // 原始输入数组，直接复用
    const ctx = I('operator_join', {
      inputs: { STRING1: B.REP(txt(' | pc='), [10, '']), STRING2: B.REP(VAR('pc'), [10, '']) }
    });
    const ctx2 = I('operator_join', {
      inputs: { STRING1: ctx, STRING2: I('operator_join', { inputs: { STRING1: B.REP(txt(' fi='), [10, '']), STRING2: B.REP(VAR('fi'), [10, '']) } }) }
    });
    const ctx3 = I('operator_join', {
      inputs: { STRING1: ctx2, STRING2: I('operator_join', { inputs: { STRING1: B.REP(txt(' head=['), [10, '']), STRING2: I('operator_join', { inputs: { STRING1: B.REP(VAR('head'), [10, '']), STRING2: B.REP(txt('] count='), [10, '']) } }) } }) }
    });
    const ctx4 = I('operator_join', {
      inputs: { STRING1: ctx3, STRING2: B.REP(VAR('count'), [10, '']) }
    });
    const msg = I('operator_join', {
      inputs: { STRING1: I('operator_join', { inputs: { STRING1: B.REP(txt('FAIL ' + id + ' :: '), [10, '']), STRING2: arg } }), STRING2: ctx4 }
    });
    try { insertBeforeStmt(id, [LOGX(msg)]); } catch (e) { skipped++; }
  }
  console.log('[P473] probes installed, skipped=' + skipped);
});
