// ---- P55: classes -- class A: / __init__ / methods / self.attr   [gap row A4]
//
// Minimal object model that fits the string-encoded value scheme:
//   * `class A:` registers every `def` in its body as a *function* named A.m
//     (lookup/calluser/regdef then do the work); curcls marks it as a method so the
//     leading `self` parameter is dropped from its recorded signature;
//   * an instance is a small integer id -- its class is InstCls[id], its attributes
//     are rows of IAttrId/IAttrName/IAttrVal/IAttrType;
//   * `A(5)` constructs: callfn sees a class name, allocates the id, binds `self`,
//     runs A.__init__ when present, and returns the id with type 'O';
//   * `obj.m(args)` -> methodcall (guard widened to 'O') -> callmeth -> calluser;
//   * `self.v` reads through a new eval branch, `self.v = x` through a new execute
//     branch.
// Not supported (documented): inheritance, class attributes, __repr__, bound-method
// values, attribute access on list elements (a[0].v), operators on instances.
patch('P55 classes', () => {
  for (const v of ['iid', 'cfull', 'ccls', 'ck', 'cax', 'cax2', 'cX', 'clf', 'self', 'mdot', 'mdep', 'mk']) newVar(v, 0);
  // NB: 0 == '' is FALSE in scratch-vm, so string-valued flags must start as ''
  newVar('curcls', '');
  for (const l of ['ClsName', 'InstCls', 'IAttrId', 'IAttrName', 'IAttrVal', 'IAttrType']) B.addList(IT, l, []);
  const STOPTHIS = () => I('control_stop', { fields: { STOP_OPTION: ['this script', null] }, mutation: { tagName: 'mutation', children: [], hasnext: 'false' } });
  const delLast = (list) => I('data_deleteoflist', { fields: B.LISTF(list), inputs: { INDEX: txt('last') } });
  const delAt = (list, idx) => I('data_deleteoflist', { fields: B.LISTF(list), inputs: { INDEX: B.REP(idx, [4, '0']) } });
  const at = (list, idx) => item(list, idx);
  const setAt = (list, idx, v) => I('data_replaceitemoflist', { fields: B.LISTF(list), inputs: { INDEX: B.REP(idx, [4, '0']), ITEM: B.REP(v, [4, '0']) } });
  const tokAt = (i) => item('Tok', i);
  const lcont = (listName) => I('data_listcontents', { fields: B.LISTF(listName) });
  const contains = (listName, v) => I('operator_contains', { inputs: { STRING1: B.REP(lcont(listName), [10, '']), STRING2: B.REP(v, [10, '']) } });
  const varNameOf = (d) => {
    if (!d) return null;
    if (Array.isArray(d[1]) && d[1][0] === 12) return d[1][1];
    const b = (typeof d[1] === 'string') ? BL()[d[1]] : null;
    return b && b.opcode === 'data_variable' ? b.fields.VARIABLE[0] : null;
  };
  const litOf = (d) => {
    if (!d) return null;
    if (Array.isArray(d) && (d[0] === 10 || d[0] === 4) && typeof d[1] !== 'object') return String(d[1]);
    if (Array.isArray(d[1]) && (d[1][0] === 10 || d[1][0] === 4)) return String(d[1][1]);
    return null;
  };
  const eqConst = (orId, name, lit) => [...reach(orId)].some((id) => {
    const b = BL()[id];
    if (!b || b.opcode !== 'operator_equals') return false;
    const a = b.inputs.OPERAND1, c = b.inputs.OPERAND2;
    return (varNameOf(a) === name && litOf(c) === lit) || (varNameOf(c) === name && litOf(a) === lit);
  });
  const hasVarEq = eqConst;
  const callsProc = (rootId, proccode) => [...reach(rootId)].some((x) => {
    const b = BL()[x];
    return b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === proccode;
  });
  const subOf = (id) => (BL()[id].inputs.SUBSTACK && typeof BL()[id].inputs.SUBSTACK[1] === 'string') ? BL()[id].inputs.SUBSTACK[1] : null;

  // ---------- 1) prepare(): class is a block head
  const prep = inDef('prepare');
  const notFree = (id) => ![...reach(id)].some((x) => BL()[x].opcode === 'operator_not');
  const headAll = find(prep, (b, id) => b.opcode === 'operator_or' && hasVarEq(id, 'prevhead', 'def') && notFree(id));
  const headChains = headAll.filter((id) => !headAll.some((o) => o !== id && reach(o).has(id)));
  if (!headChains.length) throw new Error('[P55] prepare head chains = 0');
  for (const id of headChains) replaceExpr(id, () => OR(id, EQ(VAR('prevhead'), txt('class'))));
  console.log('[P55] prepare: class is a block head (' + headChains.length + ' chains)');

  // ---------- 2) regdef(): a method loses its leading `self` parameter
  const rd = inDef('regdef %s');
  const pnAppend = find(rd, (b) => b.opcode === 'data_addtolist' && b.fields.LIST && b.fields.LIST[0] === 'FnPN');
  if (pnAppend.length !== 1) throw new Error('[P55] FnPN append = ' + pnAppend.length);
  insertBeforeStmt(pnAppend[0], [
    IF(NOT(EQ(VAR('curcls'), txt(''))), [delAt('DefPar', num(1)), delAt('DefDef', num(1))])
  ]);
  console.log('[P55] regdef: methods drop self from the signature');

  // ---------- 3) execute(): class statement + `self.v = expr`
  const ex = inDef('execute');
  const forIf = find(ex, (b) => b.opcode === 'control_if' && b.inputs.CONDITION && typeof b.inputs.CONDITION[1] === 'string' && eqConst(b.inputs.CONDITION[1], 'head', 'for'));
  if (forIf.length !== 1) throw new Error('[P55] execute for-branch = ' + forIf.length);
  const classBranch = IF(EQ(VAR('head'), txt('class')), [
    IF(NOT(EQ(tokAt(num(3)), txt(':'))), [FAIL('SyntaxError: class Name:')]),
    setV('ccls', tokAt(num(2))),
    addTo('ClsName', VAR('ccls')),
    setV('ck', ADD(VAR('pc'), num(1))),
    REPEATU(OR(GT(VAR('ck'), SUB(item('Ends', VAR('pc')), num(1))), NOT(EQ(VAR('error'), txt('')))), [
      callProc('tokenize %s', [item('Code', VAR('ck'))]),
      setV('count', I('data_lengthoflist', { fields: B.LISTF('Tok') })),
      IF(EQ(tokAt(num(1)), txt('def')), [
        setV('cax', VAR('pc')),
        setV('pc', VAR('ck')),
        setV('curcls', VAR('ccls')),
        callProc('regdef %s', [JOIN(JOIN(VAR('ccls'), txt('.')), tokAt(num(2)))]),
        setV('curcls', txt('')),
        setV('pc', VAR('cax'))
      ]),
      chgV('ck', num(1))
    ]),
    callProc('tokenize %s', [item('Code', VAR('pc'))]),
    setV('dest', item('Ends', VAR('pc'))),
    setV('pending', num(1))
  ]);
  const attrSetBranch = IF(AND(EQ(tokAt(num(2)), txt('.')), AND(EQ(tokAt(num(4)), txt('=')), EQ(item('Kind', num(1)), txt('I')))), [
    callProc('lookup %s', [tokAt(num(1))]),
    IFELSE(AND(NOT(EQ(VAR('found'), num(0))), EQ(item('Types', VAR('found')), txt('O'))), [
      setV('iid', item('Values', VAR('found'))),
      callProc('eval %s %s', [num(5), VAR('count')]),
      setV('cax', VAR('ev')),
      setV('cax2', VAR('evtype')),
      setV('cX', num(0)),
      setV('ck', num(1)),
      REPEATU(GT(VAR('ck'), llen('IAttrId')), [
        IF(AND(EQ(at('IAttrId', VAR('ck')), VAR('iid')), EQ(at('IAttrName', VAR('ck')), tokAt(num(3)))), [
          setV('cX', VAR('ck')),
          setV('ck', llen('IAttrId'))
        ]),
        chgV('ck', num(1))
      ]),
      IFELSE(EQ(VAR('cX'), num(0)), [
        addTo('IAttrId', VAR('iid')),
        addTo('IAttrName', tokAt(num(3))),
        addTo('IAttrVal', VAR('cax')),
        addTo('IAttrType', VAR('cax2'))
      ], [
        setAt('IAttrVal', VAR('cX'), VAR('cax')),
        setAt('IAttrType', VAR('cX'), VAR('cax2'))
      ])
    ], [
      FAIL('AttributeError: only instances have attributes')
    ]),
    setV('pending', num(1))
  ]);
  // `return [1, 2]` (and friends) must not be taken for an index assignment
  const isTok2Bracket = (orId) => [...reach(orId)].some((id) => {
    const b = BL()[id];
    if (!b || b.opcode !== 'operator_equals') return false;
    const a = b.inputs.OPERAND1, c = b.inputs.OPERAND2;
    const link = (d) => {
      const bb = (d && typeof d[1] === 'string') ? BL()[d[1]] : null;
      if (!bb || bb.opcode !== 'data_itemoflist' || !bb.fields.LIST || bb.fields.LIST[0] !== 'Tok') return false;
      const ix = bb.inputs.INDEX;
      return (Array.isArray(ix) && ix[0] === 4 && String(ix[1]) === '2') || (Array.isArray(ix[1]) && ix[1][0] === 4 && String(ix[1][1]) === '2');
    };
    return (link(a) && litOf(c) === '[') || (link(c) && litOf(a) === '[');
  });
  const hasItemEq = (orId, listName, want, lit) => [...reach(orId)].some((id) => {
    const b = BL()[id];
    if (!b || b.opcode !== 'operator_equals') return false;
    const a = b.inputs.OPERAND1, c = b.inputs.OPERAND2;
    const link = (d) => {
      const bb = (d && typeof d[1] === 'string') ? BL()[d[1]] : null;
      if (!bb || bb.opcode !== 'data_itemoflist' || !bb.fields.LIST || bb.fields.LIST[0] !== listName) return false;
      const ix = bb.inputs.INDEX;
      return (Array.isArray(ix) && ix[0] === 4 && String(ix[1]) === String(want)) || (Array.isArray(ix[1]) && ix[1][0] === 4 && String(ix[1][1]) === String(want));
    };
    return (link(a) && litOf(c) === lit) || (link(c) && litOf(a) === lit);
  });
  const idxIf = find(ex, (b, id) => b.opcode === 'control_if' && b.inputs.CONDITION && typeof b.inputs.CONDITION[1] === 'string'
    && hasItemEq(b.inputs.CONDITION[1], 'Kind', 1, 'I') && isTok2Bracket(b.inputs.CONDITION[1]));
  if (idxIf.length !== 1) throw new Error('[P55] indexset branch = ' + idxIf.length);
  {
    const condId = BL()[idxIf[0]].inputs.CONDITION[1];
    const kw = () => OR(OR(EQ(tokAt(num(1)), txt('return')), OR(EQ(tokAt(num(1)), txt('yield')), EQ(tokAt(num(1)), txt('raise')))), EQ(tokAt(num(1)), txt('print')));
    replaceExpr(condId, () => AND(condId, NOT(kw())));
  }
  console.log('[P55] indexset branch guarded against keyword heads');
  insertBeforeStmt(forIf[0], [classBranch, attrSetBranch]);
  console.log('[P55] execute: class + attribute-set branches installed');

  // ---------- 4) eval(): `self.v` (exactly three tokens with a dot) reads an attribute
  const ev = inDef('eval %s %s');
  const mcallIf = find(ev, (b, id) => b.opcode === 'control_if' && callsProc(subOf(id) || id, 'methodcall %s %s'));
  if (mcallIf.length !== 1) throw new Error('[P55] eval methodcall branch = ' + mcallIf.length);
  const callIf = find(ev, (b, id) => b.opcode === 'control_if' && callsProc(subOf(id) || id, 'callfn %s %s %s'));
  if (callIf.length !== 1) throw new Error('[P55] eval call branch = ' + callIf.length);
  // obj.method(...) with a receiver that is itself an expression: find the first
  // top-level dot, hand it to methodcall2 (which evaluates the receiver properly)
  const mc2 = newProc('methodcall2 %s %s %s', ['lo', 'hi', 'mdot']);
  attachBody(mc2.defId, [
    callProc('eval %s %s', [argOf('lo'), SUB(argOf('mdot'), num(1))]),
    IF(EQ(VAR('error'), txt('')), [
      setV('mcval', VAR('ev')),
      setV('mctype', VAR('evtype')),
      setV('mcpre', num(1)),
      callProc('methodcall %s %s', [SUB(argOf('mdot'), num(1)), argOf('hi')]),
      setV('mcpre', num(0))
    ])
  ]);
  insertBeforeStmt(callIf[0], [
    setV('mdot', num(0)),
    IF(NOT(LT(argOf('hi'), ADD(argOf('lo'), num(2)))), [
      setV('mdep', num(0)),
      setV('mk', argOf('lo')),
      REPEATU(GT(VAR('mk'), argOf('hi')), [
        IF(OR(EQ(tokAt(VAR('mk')), txt('(')), EQ(tokAt(VAR('mk')), txt('['))), [chgV('mdep', num(1))]),
        IF(OR(EQ(tokAt(VAR('mk')), txt(')')), EQ(tokAt(VAR('mk')), txt(']'))), [chgV('mdep', num(-1))]),
        IF(AND(EQ(VAR('mdep'), num(0)), EQ(tokAt(VAR('mk')), txt('.'))), [IF(EQ(VAR('mdot'), num(0)), [setV('mdot', VAR('mk'))])]),
        chgV('mk', num(1))
      ])
    ]),
    // only a MULTI-token receiver needs methodcall2; `math.floor(x)` keeps going through
    // the classic methodcall branch (its math special case reads Tok[lo+2])
    IF(AND(GT(VAR('mdot'), ADD(argOf('lo'), num(1))), EQ(tokAt(argOf('hi')), txt(')'))), [
      callProc('methodcall2 %s %s %s', [argOf('lo'), argOf('hi'), VAR('mdot')]),
      STOPTHIS()
    ])
  ]);
  console.log('[P55] eval: methodcall2 route installed');
  insertBeforeStmt(mcallIf[0], [
    IF(AND(EQ(tokAt(ADD(argOf('lo'), num(1))), txt('.')), EQ(argOf('hi'), ADD(argOf('lo'), num(2)))), [
      callProc('eval %s %s', [argOf('lo'), argOf('lo')]),
      IFELSE(EQ(VAR('evtype'), txt('O')), [
        setV('iid', VAR('ev')),
        setV('cX', num(0)),
        setV('ck', num(1)),
        REPEATU(GT(VAR('ck'), llen('IAttrId')), [
          IF(AND(EQ(at('IAttrId', VAR('ck')), VAR('iid')), EQ(at('IAttrName', VAR('ck')), tokAt(ADD(argOf('lo'), num(2))))), [
            setV('cX', VAR('ck')),
            setV('ck', llen('IAttrId'))
          ]),
          chgV('ck', num(1))
        ]),
        IFELSE(EQ(VAR('cX'), num(0)), [
          FAILB(JOIN(txt('AttributeError: '), tokAt(ADD(argOf('lo'), num(2)))))
        ], [
          setV('ev', at('IAttrVal', VAR('cX'))),
          setV('evtype', at('IAttrType', VAR('cX')))
        ])
      ], [
        FAIL('SyntaxError: attribute access')
      ]),
      STOPTHIS()
    ])
  ]);
  console.log('[P55] eval: attribute read installed');

  // ---------- 5) callmeth proc + methodcall dispatch for instances
  const cm = newProc('callmeth %s %s %s %s', ['inst', 'meth', 'lo', 'hi']);
  attachBody(cm.defId, [
    setV('ccls', item('InstCls', argOf('inst'))),
    setV('cfull', JOIN(JOIN(VAR('ccls'), txt('.')), argOf('meth'))),
    IFELSE(contains('FnName', VAR('cfull')), [
      addTo('VStack', VAR('self')),
      setV('self', argOf('inst')),
      callProc('assign %s %s %s', [txt('self'), argOf('inst'), txt('O')]),
      callProc('calluser %s %s %s', [VAR('cfull'), argOf('lo'), argOf('hi')]),
      setV('clf', item('VStack', llen('VStack'))),
      delLast('VStack'),
      IFELSE(EQ(VAR('clf'), txt('')), [
        callProc('assign %s %s %s', [txt('self'), txt(''), txt('Z')])
      ], [
        callProc('assign %s %s %s', [txt('self'), VAR('clf'), txt('O')])
      ]),
      setV('self', VAR('clf'))
    ], [
      FAILB(JOIN(JOIN(VAR('ccls'), txt(' has no attribute ')), argOf('meth')))
    ])
  ]);
  const mc = inDef('methodcall %s %s');
  const lmStmt = find(mc, (b) => b.opcode === 'data_setvariableto' && b.fields.VARIABLE[0] === 'lmeth');
  if (lmStmt.length !== 1) throw new Error('[P55] methodcall lmeth = ' + lmStmt.length);
  spliceAfter(lmStmt[0], chain([
    IF(EQ(VAR('xb'), txt('O')), [
      callProc('callmeth %s %s %s %s', [VAR('xa'), VAR('lmeth'), ADD(argOf('lo'), num(4)), SUB(argOf('hi'), num(1))]),
      STOPTHIS()
    ])
  ]));
  const guard = find(mc, (b, id) => b.opcode === 'operator_or' && eqConst(id, 'xb', 'L') && eqConst(id, 'xb', 'S'));
  if (guard.length !== 1) throw new Error('[P55] methodcall type guard = ' + guard.length);
  replaceExpr(guard[0], () => OR(guard[0], EQ(VAR('xb'), txt('O'))));
  console.log('[P55] methodcall: instance dispatch installed');

  // ---------- 6) callfn(): a class name constructs an instance
  const cf = inDef('callfn %s %s %s');
  const argEval = find(cf, (b) => b.opcode === 'procedures_call' && b.mutation && b.mutation.proccode === 'eval %s %s');
  if (argEval.length !== 1) throw new Error('[P55] callfn arg eval = ' + argEval.length);
  insertBeforeStmt(argEval[0], [
    IF(contains('ClsName', argOf('fn')), [
      setV('iid', ADD(llen('InstCls'), num(1))),
      addTo('InstCls', argOf('fn')),
      setV('cfull', JOIN(JOIN(argOf('fn'), txt('.')), txt('__init__'))),
      IF(contains('FnName', VAR('cfull')), [
        addTo('VStack', VAR('self')),
        setV('self', VAR('iid')),
        callProc('assign %s %s %s', [txt('self'), VAR('iid'), txt('O')]),
        callProc('calluser %s %s %s', [VAR('cfull'), argOf('lo'), argOf('hi')]),
        setV('clf', item('VStack', llen('VStack'))),
        delLast('VStack'),
        IFELSE(EQ(VAR('clf'), txt('')), [
          callProc('assign %s %s %s', [txt('self'), txt(''), txt('Z')])
        ], [
          callProc('assign %s %s %s', [txt('self'), VAR('clf'), txt('O')])
        ]),
        setV('self', VAR('clf'))
      ]),
      IF(EQ(VAR('error'), txt('')), [setV('ev', VAR('iid')), setV('evtype', txt('O'))]),
      STOPTHIS()
    ])
  ]);

  // ---------- 7) reset the new tables at run start
  const runInit = find(allIds(), (b) => b.opcode === 'data_deletealloflist' && b.fields.LIST && b.fields.LIST[0] === 'LoopText');
  if (runInit.length !== 1) throw new Error('[P55] run init anchor = ' + runInit.length);
  insertBeforeStmt(runInit[0], [clearL('ClsName'), clearL('InstCls'), clearL('IAttrId'), clearL('IAttrName'), clearL('IAttrVal'), clearL('IAttrType')]);
  console.log('[P55] classes ready');
});
