### paren target comparison
a = (1, 2)
print(a == (1, 2))
### paren tuple compare
pa = (1, 2)
print(pa == (1, 2))
### expression statement paren
(1 + 2)
print('ok')
### paren target with function values
def f():
    return 7, 8
(x, y) = f()
print(x, y)
### tuple rhs of two parens
p = (1, 2), (3, 4)
print(p)
### nested both sides
(a, b), (c, d) = (1, 2), (3, 4)
print(a, b, c, d)
### bracket nested
[(a, b), c] = [(1, 2), 3]
print(a, b, c)
### paren target 3 elements
(x, y, z) = (1, 2, 3)
print(x, y, z)
### paren target from list
(a, b) = [4, 5]
print(a, b)
### paren target mismatch
try:
    (a, b) = (1, 2, 3)
except ValueError as e:
    print('ve', e)
### star in paren target
(a, *rest) = (1, 2, 3)
print(a, rest)
### swap with parens
a, b = 1, 2
(a, b) = (b, a)
print(a, b)
### flat still works
a, b = 1, 2
print(a, b)
### nested flat still works
a, (b, c) = 1, (2, 3)
print(a, b, c)
