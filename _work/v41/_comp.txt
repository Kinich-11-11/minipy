### dict comprehension tuple target
print({k: v for k, v in [('a', 1)]})
### dict comprehension over items
d = {'a': 1, 'b': 2}
print({k: v for k, v in d.items()})
### dict comprehension full
print({k: v for k, v in [('a', 1), ('b', 2)]})
print({k: k * 2 for k in range(4)})
print({k: v for k, v in {'a': 1, 'b': 2}.items() if v > 1})
print({v: k for k, v in {'a': 1, 'b': 2}.items()})
### list comprehension tuple target
print([k for k, v in [('a', 1), ('b', 2)]])
### list comprehension over items
d = {'x': 1, 'y': 2}
print([k for k, v in d.items()])
### set comprehension tuple target
print({v for k, v in [('a', 1), ('b', 1)]})
### comprehension over dict keys
d = {'a': 1, 'b': 2}
print([k for k in d])
### dict comprehension over dict
d = {'a': 1, 'b': 2}
print({k: v * 2 for k, v in d.items()})
### tuple target with parens
print([a + b for (a, b) in [(1, 2), (3, 4)]])
### generator style comprehension single name still works
print([x * x for x in range(3)])
### nested comprehension
print([[y for y in range(x)] for x in range(3)])
### comprehension with two for clauses
print([(x, y) for x in range(2) for y in range(2)])
