### literal paren compare
print('(' == '(')
### variable paren compare
x = '('
print(x == '(')
### indexing paren
s = '(a,b)'
print(s[0])
print(s[0] == '(')
### letter of via slicing
print(s[1])
### bracket compare
print('[' == '[')
### comma compare
print(',' == ',')
### paren in list
t = ['(', 'a']
print(t[0] == '(')
