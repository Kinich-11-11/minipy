### closure counter
def make_counter():
    n = 0
    def inc():
        nonlocal n
        n += 1
        return n
    return inc
c1 = make_counter()
c2 = make_counter()
print(c1(), c1(), c2(), c1())
### decorator basic
def double_result(f):
    def wrapper(n):
        return f(n) * 2
    return wrapper
@double_result
def val(n):
    return n + 1
print(val(5), val(0))
