### h1 ordinary fn call with self.x
def f(v): return v
class Q:
    def __init__(self, x): self.x = x
    def mk(self): return f(self.x)
print(Q(3).mk())
### h2 print self.x in method
class Q:
    def __init__(self, x): self.x = x
    def show(self): print(self.x)
Q(4).show()
### h3 class call with self.x
class P:
    def __init__(self, x): self.x = x
    def mk(self): return P(self.x)
print(P(5).mk().x)
### h4 class call with const only
class P:
    def __init__(self, x): self.x = x
    def mk(self): return P(7)
print(P(5).mk().x)
### h5 method call with self.x as arg
class R:
    def __init__(self, x): self.x = x
    def add(self, n): return self.x + n
    def mk(self): return self.add(self.x)
print(R(6).mk())
