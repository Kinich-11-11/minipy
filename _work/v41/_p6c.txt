### operator overloading vector
class Vec:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    def __add__(self, o):
        return Vec(self.x + o.x, self.y + o.y)
    def __sub__(self, o):
        return Vec(self.x - o.x, self.y - o.y)
    def __mul__(self, k):
        return Vec(self.x * k, self.y * k)
    def __eq__(self, o):
        return self.x == o.x and self.y == o.y
    def __str__(self):
        return '(' + str(self.x) + ',' + str(self.y) + ')'
v1 = Vec(1, 2)
v2 = Vec(3, 4)
print(v1 + v2, v2 - v1, v1 * 3)
print(v1 + v2 == Vec(4, 6), v1 == v2)
