### star target basic
a, *rest = [1, 2, 3, 4]
print(a, rest)
### star middle
a, *mid, z = [1, 2, 3, 4]
print(a, mid, z)
### star target tuple rhs
a, *rest = (1, 2, 3)
print(a, rest)
### star target empty rest
a, *rest = [1]
print(a, rest)
### star nested unpack
a, (b, *c) = 1, (2, 3, 4)
print(a, b, c)
### star to list
first, *others = "abc"
print(first, others)
### subscript targets flat
d = {}
d["a"], d["b"] = 1, 2
print(d)
### subscript targets parens
d = {}
(d["a"], d["b"]) = (3, 4)
print(d)
### subscript and name mixed
d = {}
x, d["k"] = 10, 20
print(x, d)
### star with subscript
d = {}
a, *d["r"] = 1, 2, 3
print(a, d)
### starred call with keyword
def f(a, b, c):
    return a + b + c
args = [1, 2]
print(f(*args, c=3))
### double star call
def g(a, b):
    return a * b
kw = {"a": 2, "b": 5}
print(g(**kw))
