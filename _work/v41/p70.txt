// ---- P70: bitwise augmented assignment (E5/E6) ----
//
// `b &= 10` never merges in the tokenizer ('&' is not in its "can be followed by ="
// set), so it arrives as four tokens [name, '&', '=', value].  `b <<= 1` takes the same
// shape because `<`+`<` merges into '<<' but the '=' stays behind.  So one branch in
// execute covers all five operators: reuse `rightside` (it already implements & | ^ << >>)
// and hand it the operator token plus the value starting at index 4.
patch('P70 bitwise augmented assign', () => {
  const litOf = (d) => {
    if (!d) return null;
    if (Array.isArray(d) && (d[0] === 10 || d[0] === 4) && typeof d[1] !== 'object') return String(d[1]);
    if (Array.isArray(d[1]) && (d[1][0] === 10 || d[1][0] === 4)) return String(d[1][1]);
    return null;
  };
  const isTokAt = (d, idx) => {
    const b = (d && typeof d[1] === 'string') ? BL()[d[1]] : null;
    if (!b || b.opcode !== 'data_itemoflist') return false;
    if (!b.fields.LIST || b.fields.LIST[0] !== 'Tok') return false;
    return litOf(b.inputs.INDEX) === String(idx);
  };
  const eqTokAt = (condId, idx, lit) => [...reach(condId)].some((id) => {
    const b = BL()[id];
    if (!b || b.opcode !== 'operator_equals') return false;
    const a = b.inputs.OPERAND1, c = b.inputs.OPERAND2;
    return (isTokAt(a, idx) && litOf(c) === lit) || (isTokAt(c, idx) && litOf(a) === lit);
  });
  const ex = inDef('execute');
  // the "+=" / "-=" / ... branch has a unique condition (item 2 of Tok == "+=")
  const plusEq = find(ex, (b) => b.opcode === 'control_if' && b.inputs.CONDITION && typeof b.inputs.CONDITION[1] === 'string' && eqTokAt(b.inputs.CONDITION[1], 2, '+='));
  if (plusEq.length !== 1) throw new Error('[P70] "+=" branch = ' + plusEq.length);
  const opIsBit = I('operator_contains', { inputs: { STRING1: B.REP(txt('&|^<<>>'), [10, '']), STRING2: B.REP(item('Tok', num(2)), [10, '']) } });
  const branch = IF(
    AND(AND(EQ(item('Kind', num(1)), txt('I')), EQ(item('Tok', num(3)), txt('='))), opIsBit),
    [
      setV('name', item('Tok', num(1))),
      callProc('lookup %s', [VAR('name')]),
      IFELSE(EQ(VAR('found'), num(0)),
        [FAILB(JOIN(txt('NameError: '), VAR('name')))],
        [
          callProc('rightside %s %s %s %s %s', [item('Values', VAR('found')), item('Types', VAR('found')), item('Tok', num(2)), num(4), VAR('count')]),
          IF(EQ(VAR('error'), txt('')), [callProc('assign %s %s %s', [VAR('name'), VAR('ev'), VAR('evtype')])])
        ]),
      setV('pending', num(1))
    ]
  );
  insertBeforeStmt(plusEq[0], [branch]);
  console.log('[P70] & | ^ << >> augmented assignment installed');
});
