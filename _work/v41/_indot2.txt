### list literal with float
print([1.5])
### dict literal with float
print({'a': 1.5})
### dict getitem float
d = {'a': 1.5}
print(d['a'])
### dict getitem float type
d = {'a': 1.5}
print(type(d['a']))
### list getitem float type
a = [1.5]
print(a[0], type(a[0]))
### float literal in list then in
print(1.5 in [1.5])
### float variable in list
x = 1.5
print(x in [1.5])
### float literal in list variable
a = [1.5]
print(1.5 in a)
### list element in its own list
a = [1.5]
print(a[0] in a)
### dict membership float value
d = {'a': 1.5}
print('a' in d)
### string dot in string dot literal
print('1.5' in '1.5')
### string dot variable in literal
x = '1.5'
print(x in '1.5')
### dot inside a longer string
print('a.b' in 'xa.by')
### int in list
print(1 in [1, 2])
### float 0.5 in list
print(0.5 in [0.5])
### float 1.0 in list
print(1.0 in [1.0])
### dict items with float
d = {'a': 1.5}
for k, v in d.items():
    print(k, v, type(v))
### dict keys with float value
d = {'a': 1.5}
print(d.keys())
### len of float dict
print(len({'a': 1.5}))
### float equality inside dict
print({'a': 1.5}['a'] == 1.5)
### float in tuple
print(1.5 in (1.5,))
### float in set
print(1.5 in {1.5})
### string without dot in string
print('ab' in 'xaby')
### dot at end of string
print('a.' in 'xa.y')
### number then attribute-like dot in string
s = '1.5'
print(s, len(s), type(s))
### list of float printed and indexed
a = [0.5, 1.25]
print(a, a[0], a[1], type(a[1]))
### dict of float printed and typed
d = {'x': 0.5, 'y': 1.25}
print(d, d['x'], type(d['y']))
