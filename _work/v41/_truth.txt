### if empty list
if []:
    print('truthy')
else:
    print('falsy')
### if empty dict
if {}:
    print('truthy')
else:
    print('falsy')
### if empty tuple
if ():
    print('truthy')
else:
    print('falsy')
### if empty set
if set():
    print('truthy')
else:
    print('falsy')
### not empty list
print(not [], not [1], not {}, not '')
### and or with empty containers
print([] and 1, [] or 2, [1] and 3, {} or 'x')
### while empty list
n = 0
while []:
    n += 1
print(n)
### bool builtin
print(bool([]), bool([1]), bool({}), bool(()), bool(''), bool(set()), bool(0), bool(1))
### if empty string
if '':
    print('truthy')
else:
    print('falsy')
### empty list equality in condition
print([] == 0, [] == False, bool([]) == False)
### ternary with empty list
print('a' if [] else 'b')
### list of empties
print([bool(x) for x in [[], {}, (), 0, '', 1, [1]]])
