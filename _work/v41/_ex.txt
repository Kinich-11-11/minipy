### x1 type(e)
class E(Exception):
    pass
try:
    raise E('x')
except E as e:
    print(type(e))
### x2 type(e).__name__
class E(Exception):
    pass
try:
    raise E('x')
except E as e:
    print(type(e).__name__)
### x3 type equality
class E(Exception):
    pass
try:
    raise E('x')
except E as e:
    print(type(e) == E)
### x4 custom attr
class E(Exception):
    def __init__(self, code):
        super().__init__('boom')
        self.code = code
try:
    raise E(42)
except E as e:
    print(e.code)
### x5 str of user exception
class E(Exception):
    pass
try:
    raise E('msg')
except E as e:
    print(str(e))
### x6 builtin exc name
try:
    1 / 0
except ZeroDivisionError as e:
    print(type(e).__name__)
### x7 builtin exc str
try:
    1 / 0
except ZeroDivisionError as e:
    print(str(e))
### x8 isinstance
class E(Exception):
    pass
try:
    raise E('x')
except Exception as e:
    print(isinstance(e, E))
