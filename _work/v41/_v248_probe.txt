### I01 委托生成器 close 传播
def a():
    try:
        yield 1
    finally:
        print("inner")
def b():
    try:
        yield from a()
    finally:
        print("outer")
g = b()
print(next(g))
g.close()

### I09 多层闭包大小写隔离
def f(x,X):
    def g():
        def h():
            nonlocal x,X
            x += 1
            X += 10
            return x,X
        return h
    return g()
a = f(1,2)
b = f(3,4)
print(a(),b(),a())
### I10 闭包同帧共享大小写
def f():
    x = 1
    X = 2
    def inc():
        nonlocal X
        X += 3
    def read():
        return x,X
    return inc,read
a,b = f()
a()
print(b())

### I24 next default custom iterator
class A:
    def __init__(self):
        self.n = 0
    def __iter__(self):
        return self
    def __next__(self):
        self.n += 1
        if self.n > 1:
            raise StopIteration
        return self.n
a = iter(A())
print(next(a,9),next(a,9))
