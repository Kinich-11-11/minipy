### catch zero division
try:
    x = 1 / 0
except ZeroDivisionError:
    print('zd caught')
### catch int value error
try:
    x = int('abc')
except ValueError:
    print('int caught')
### catch index error
try:
    x = [1, 2][5]
except IndexError:
    print('ie caught')
### catch slice step zero read
try:
    x = 'abc'[::0]
    print('no error', x)
except ValueError:
    print('step caught')
### catch slice step type
try:
    x = [1, 2, 3][0:2:'a']
except TypeError:
    print('type caught')
### catch list index error in function
def f():
    return [1][9]
try:
    f()
except IndexError:
    print('fn caught')
### catch slice step zero finally
try:
    x = 'abc'[::0]
except ValueError:
    print('caught')
finally:
    print('done')
