### escapes newline tab backslash
print(repr('\n'), repr('\t'), repr('\\'))
### escape carriage return
print(repr('\r'))
### escapes v f a b
print(repr('\v'), repr('\f'), repr('\a'), repr('\b'))
### escape zero
print(repr('\0'))
### escape single quote
print(repr('\''))
### escape double quote
print(repr("\""))
### escape unknown letter
print(repr('\q'))
### escape hex byte
print(repr('\x41'))
### escape octal
print(repr('\101'))
### escape lengths
print(len('\n'), len('\t'), len('\r'), len('\\'), len('\x41'))
### newline prints two rows
print('a\nb')
### crlf repr
print(repr('a\r\nb'))
### splitlines crlf
print('a\r\nb'.splitlines())
### splitlines cr only
print('a\rb'.splitlines())
### strip cr
print(repr('\rx\r'.strip()))
### split on tab
print('a\tb'.split('\t'))
### backslash in middle
print(repr('a\\b'))
### cr inside fstring
print(repr(f'{"\r"}'))
### cr count
print('a\rb'.count('\r'))
### cr find
print('a\rb'.find('\r'))
### tab repr inside text
print(repr('x\ty'))
### newline at end
print(repr('x\n'))
