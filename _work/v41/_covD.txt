### pow two args
print(pow(2, 3))
### pow two args computed
x = 10
print(pow(2, x))
### pow three args
print(pow(2, 3, 5))
### !pow one arg
print(pow(2))
### pow name alone
print(pow)
### pow float exp
print(pow(2, 0.5))
### sep empty alone
print('x', 'y', sep='')
### sep empty with end newline
print('x', 'y', sep='', end='\n')
### sep empty with end empty
print('x', 'y', sep='', end='')
### end only
print('a', end='-')
print('b')
### sep dash and end
print('a', 'b', sep='-', end='!')

print()
