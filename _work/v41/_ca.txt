### x1 class attr list shared
class A:
    xs = []
a1 = A()
a2 = A()
a1.xs.append(1)
print(a2.xs)
### x2 assign via class
class B:
    n = 1
B.n = 5
print(B.n, B().n)
### x3 instance shadows
class C:
    n = 1
c = C()
c.n = 2
print(C.n, c.n)
### x4 method assigns class attr
class D:
    n = 0
    def inc(self):
        D.n = D.n + 1
d = D()
d.inc()
d.inc()
print(D.n, d.n)
### x5 shared list via method
class E:
    xs = []
    def add(self, v):
        self.xs.append(v)
e1 = E()
e2 = E()
e1.add(1)
print(e2.xs)
### x6 count instances
class F:
    n = 0
    def __init__(self):
        F.n = F.n + 1
f1 = F()
f2 = F()
print(F.n)
