### p1 function value direct
def mk(n):
    def g(x):
        return x * n
    return g
f = mk(2)
print(f, type(f))
### p2 through a list
def mk2(n):
    def g(x):
        return x * n
    return g
fs = [mk2(3)]
print(fs[0], type(fs[0]))
### p3 call via parameter
def mk3(n):
    def g(x):
        return x * n
    return g
def callit(fn, v):
    return fn(v)
print(callit(mk3(4), 5))
### p4 non-closure function value in list
def plain(x):
    return x + 1
ps = [plain]
print(ps[0](3))
### p5 plain function value direct
f5 = plain
print(f5(3))
