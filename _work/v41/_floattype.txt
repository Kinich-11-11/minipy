### type of float literals
print(type(1.5), type(1), type(0.5), type(2.0), type(-1.5))
### type of scientific literal
print(type(1e3), type(2.5e-3))
### type of float variable
x = 1.5
print(type(x))
### type of float arithmetic
print(type(1.0 + 1.0), type(1 / 2), type(2 * 0.5), type(3.0 - 1.0))
### type of float from function
def f():
    return 1.5
print(type(f()))
### isinstance float
print(isinstance(1.5, float), isinstance(1.5, int), isinstance(1, float))
### float still prints right
print(1.5, 2.0, 0.5, -1.5)
### int and float compare
print(1.5 == 1.5, 1.5 > 1, 1.0 == 1)
### float in dict membership no dot
print(1 in {1: 'x'})
### int in list no dot
print(1 in [1, 2])
### type of string
print(type('a'), type('1.5'), type('1'))
### int in tuple
print(1 in (1, 2))
### float arithmetic results
print(1.0 / 2, 0.1 + 0.2, 1.5 * 2)
### float key in dict
d = {1.5: 'x'}
print(d[1.5])
### str of float
print(str(1.5), str(2.0))
### float comparisons chain
print(1.0 < 1.5 < 2.0)
### float in fstring spec
print(f'{1.5:.2f}')
### float sum type
print(type(sum([1.5, 2.5])))
### float abs type
print(type(abs(-1.5)))
