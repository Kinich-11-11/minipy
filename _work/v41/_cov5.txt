### closure captures variable not value
def make():
    n = 0
    def inc():
        return n + 1
    return inc
print(make()())
### closure counter attempt without nonlocal
def make_counter():
    n = 0
    def inc():
        n = n + 1
        return n
    return inc
c = make_counter()
try:
    print(c(), c())
except UnboundLocalError as e:
    print('ule', e)
### nonlocal keyword
def outer():
    n = 0
    def inc():
        nonlocal n
        n += 1
        return n
    return inc
c = outer()
print(c(), c(), c())

### recursion
def fact(n):
    if n <= 1:
        return 1
    return n * fact(n - 1)
print(fact(6))
def fib(n):
    if n < 2:
        return n
    return fib(n - 1) + fib(n - 2)
print(fib(12))
### mutual recursion
def is_even(n):
    if n == 0:
        return True
    return is_odd(n - 1)
def is_odd(n):
    if n == 0:
        return False
    return is_even(n - 1)
print(is_even(10), is_odd(7))
