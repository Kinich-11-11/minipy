### generator max alone
data = ['apple', 'fig', 'banana', 'kiwi', 'plum']
print(max(len(w) for w in data))
### generator list alone
data = ['apple', 'fig', 'banana', 'kiwi', 'plum']
print(list(len(w) for w in data))
### generator max with map
data = ['apple', 'fig', 'banana', 'kiwi', 'plum']
print(sum(map(len, data)), max(len(w) for w in data))
### generator max separated
data = ['apple', 'fig', 'banana', 'kiwi', 'plum']
g = (len(w) for w in data)
print(max(g))
### generator comprehension split
g = (x for x in range(3))
a = [v for v in g]
b = [v for v in g]
print(a, b)
### generator list twice
g = (x for x in range(3))
print(list(g), list(g))
### nested alias simple
def h(out):
    def inner():
        out.append(1)
    inner()
    print(out)
o = []
h(o)
print(o)
### parameter alias nested without capture
def h(out):
    def inner(a):
        a.append(1)
    inner(out)
o = []
h(o)
print(o)
### list nested item assignment
def h(a):
    def inner(b):
        b[0], b[1] = b[1], b[0]
    inner(a)
    print(a)
a = [1,2]
h(a)
print(a)
### parameter shallow slice item assignment
def f(a):
    b = a[:]
    b[0] = 9
    print(a, b)
a = [1, 2]
f(a)
print(a)
### parameter shallow slice tuple swap
def f(a):
    b = a[:]
    b[0], b[1] = b[1], b[0]
    print(a, b)
a = [1, 2]
f(a)
print(a)

### zero division integer and float operations
for a in [1, 1.0]:
    for b in [0, 0.0]:
        try:
            print(a/b)
        except ZeroDivisionError as e:
            print(e)
        try:
            print(a//b)
        except ZeroDivisionError as e:
            print(e)
        try:
            print(a%b)
        except ZeroDivisionError as e:
            print(e)
        try:
            print(divmod(a,b))
        except ZeroDivisionError as e:
            print(e)

### parameter rebind isolates caller

    a = []
    a.append(2)
    return a
base = [1]
print(f(base), base)
### same-name parameter alias

    def g(a):
        a.append(3)
    g(a)
    return a
a = [1]
print(f(a), a)
### unrelated same name local

        a = []

### same object twice as parameter

print(a)
### dictionary alias across calls

### set alias across calls

    def g(b):

    g(a)

### returned captured parameter mutation
def make(a):
    def push(x):
        a.append(x)
    return push
a = []
push = make(a)
push(2)
push(3)
print(a)
### caller rebind disconnects closed parameter
def make(a):
    def push():
        a.append(2)
        return a
    return push
a = [1]
push = make(a)
a = [9]
print(push(), a)
### parameter shallow copy isolates original
def f(a):
    b = a[:]
    c = a.copy()
    b.append(2)
    c.append(3)
    print(a, b, c)
a = [1]
f(a)
print(a)
### parameter shallow nested copy
def f(a):
    b = a[:]
    b[0].append(2)
    print(a, b)
a = [[1]]
f(a)
print(a)
