def qsort(xs):
    if len(xs) <= 1:
        return xs
    p = xs[0]
    lo = []
    eq = []
    hi = []
    for x in xs:
        if x < p:
            lo.append(x)
        elif x > p:
            hi.append(x)
        else:
            eq.append(x)
    return qsort(lo) + eq + qsort(hi)

### I2 mergesort and binary search
def msort(xs):
    if len(xs) <= 1:
        return xs
    m = len(xs) // 2
    left = msort(xs[:m])
    right = msort(xs[m:])
    out = []
    i = 0
    j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            out.append(left[i])
            i += 1
        else:
            out.append(right[j])
            j += 1
    out.extend(left[i:])
    out.extend(right[j:])
    return out
a = msort([9, 3, 7, 1, 8, 2])
print(a)
def bsearch(xs, t):
    lo, hi = 0, len(xs)
    while lo < hi:
        mid = (lo + hi) // 2
        if xs[mid] < t:
            lo = mid + 1
        else:
            hi = mid
    if lo < len(xs) and xs[lo] == t:
        return lo
    return -1
print(bsearch(a, 7), bsearch(a, 4), bsearch(a, 9))
### I3 dp lcs
def lcs(a, b):
    n = len(a)
    m = len(b)
    dp = []
    for i in range(n + 1):
        dp.append([0] * (m + 1))
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            if a[i - 1] == b[j - 1]:
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    return dp[n][m]
print(lcs('ABCBDAB', 'BDCABA'))
print(lcs('', 'abc'), lcs('abc', 'abc'))

def bfs(g, s):
    q = [s]
    seen = [s]
    order = []
    while q:
        cur = q.pop(0)
        order.append(cur)
        for nxt in g[cur]:
            if nxt not in seen:
                seen.append(nxt)
                q.append(nxt)
    return order
print(bfs(graph, 'A'))
def path(g, s, t):
    prev = {}
    q = [s]
    seen = [s]
    while q:
        cur = q.pop(0)
        if cur == t:
            break
        for nxt in g[cur]:
            if nxt not in seen:
                seen.append(nxt)
                prev[nxt] = cur
                q.append(nxt)
    if t not in seen:
        return None
    out = []
    cur = t
    while cur != s:
        out.append(cur)
        cur = prev[cur]
    out.append(s)
    out.reverse()
    return out

### I5 prime sieve
def sieve(n):
    flags = [True] * (n + 1)
    flags[0] = False
    flags[1] = False
    p = 2
    while p * p <= n:
        if flags[p]:
            k = p * p
            while k <= n:
                flags[k] = False
                k += p
        p += 1
    out = []
    for i in range(n + 1):
        if flags[i]:
            out.append(i)
    return out

def mul_str(a, b):
    res = [0] * (len(a) + len(b))
    for i in range(len(a) - 1, -1, -1):
        for j in range(len(b) - 1, -1, -1):
            res[i + j + 1] += int(a[i]) * int(b[j])
    for i in range(len(res) - 1, 0, -1):
        res[i - 1] += res[i] // 10
        res[i] = res[i] % 10
    out = ''
    for d in res:
        if not (out == '' and d == 0):
            out += str(d)
    if out == '':
        return '0'
    return out

### I8 rpn evaluator
def rpn(expr):
    st = []
    for tok in expr.split():
        if tok in ['+', '-', '*', '/']:
            b = st.pop()
            a = st.pop()
            if tok == '+':
                st.append(a + b)
            elif tok == '-':
                st.append(a - b)
            elif tok == '*':
                st.append(a * b)
            else:
                st.append(a // b)
        else:
            st.append(int(tok))
    return st.pop()

print(rpn('5 1 2 + 4 * + 3 -'))

### I9 knapsack dp
def knap(w, vals, weights):
    dp = [0] * (w + 1)
    for i in range(len(vals)):
        for cap in range(w, weights[i] - 1, -1):
            cand = dp[cap - weights[i]] + vals[i]
            if cand > dp[cap]:
                dp[cap] = cand
    return dp[w]

### I14 tokenizer state machine
def tokenize(s):
    toks = []
    i = 0
    while i < len(s):
        ch = s[i]
        if ch == ' ':
            i += 1
        elif ch >= '0' and ch <= '9':
            j = i
            while j < len(s) and s[j] >= '0' and s[j] <= '9':
                j += 1
            toks.append(('num', int(s[i:j])))
            i = j
        else:
            toks.append(('op', ch))
            i += 1
    return toks

### I15 nested pretty printer
def render(v, ind):
    pad = ' ' * ind
    if isinstance(v, dict):
        if len(v) == 0:
            return '{}'
        out = '{\n'
        for k in sorted(v):
            out += pad + '  ' + str(k) + ': ' + render(v[k], ind + 2) + '\n'
        return out + pad + '}'
    if isinstance(v, list):
        if len(v) == 0:
            return '[]'
        out = '[\n'
        for x in v:
            out += pad + '  ' + render(x, ind + 2) + '\n'
        return out + pad + ']'
    return str(v)

print(render(data, 0))
