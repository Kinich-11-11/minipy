### l1 lambda inline two params
print((lambda a, b: a + b)(1, 2))
### l2 lambda default
f = lambda x, y=10: x + y
print(f(1), f(1, 2))
### l3 builtin as value
def apply(f, v):
    return f(v)
print(apply(str, 4))
print(apply(len, [1, 2]))
### l4 lambda no params
print((lambda: 7)())
### l5 lambda stored then called
g = lambda x: x * 2
print(g(21))
### l6 builtin as value direct
h = abs
print(h(-3))
