### probe perm recursion minimal
def perms(xs):
    if len(xs) <= 1:
        return [xs]
    out = []
    for i in range(len(xs)):
        rest = xs[:i] + xs[i + 1:]
        for p in perms(rest):
            out.append([xs[i]] + p)
    return out
print(perms(['a', 'b']))
print(perms(['a', 'b', 'c']))

### probe heap sort minimal
def sift(a, n, i):
    while True:
        big = i
        l = 2 * i + 1
        if l < n and a[l] > a[big]:
            big = l
        if big == i:
            return
        a[i], a[big] = a[big], a[i]
        i = big
def hsort(a):
    h = a[:]
    n = len(h)
    for i in range(n // 2 - 1, -1, -1):
        sift(h, n, i)
    for end in range(n - 1, 0, -1):
        h[0], h[end] = h[end], h[0]
        sift(h, end, 0)
    return h
print(hsort([3, 1, 2]))
print(hsort([9, 4, 7, 1, 8]))
