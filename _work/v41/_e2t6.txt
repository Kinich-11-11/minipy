### order swap
class N:
    def __init__(self, v, a, b):
        self.v = v
        self.a = a
        self.b = b
    def vals(self):
        return [self.v]
l0 = N(0, None, None)
r2 = N(2, None, None)
l1 = N(1, l0, r2)
top = N(5, l1, None)
print(top.b.vals())
print(top.a.vals())
### via alias var
q = top
print(q.a.vals())
print(q.b.vals())
### method call inside method arg context
def f(x):
    return x
print(f(top.a.vals()), top.b.vals())
