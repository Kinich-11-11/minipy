### tuple-form subscript assignment
a = [5, 2]
a[0], a[1] = a[1], a[0]
print(a)
### ret alias nested append
def build2(out):
    def add():
        out.append(1)
    add()
    return out
print(build2([]))
### ret alias global list
items = []
def additem(v):
    items.append(v)
    return items
print(additem(1), additem(2))
