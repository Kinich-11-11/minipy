### V1 pop in append
a = [1, 2, 3]
b = []
b.append(a.pop(0))
print(b, a)
### V2 two statements
a2 = [1, 2, 3]
b2 = []
x = a2.pop(0)
b2.append(x)
print(b2, a2)
### V3 inner nonmutating
a3 = [1, 2, 3]
b3 = []
b3.append(a3.index(1))
print(b3, a3)
### V4 str upper in append
b4 = []
b4.append('abc'.upper())
print(b4)
### V5 str upper standalone
print('abc'.upper())
### V6 str upper assigned then appended
u = 'abc'.upper()
b6 = []
b6.append(u)
print(b6)
### V7 append concat with upper
b7 = []
b7.append('a' + 'bc'.upper())
print(b7)
### V8 two lists
lst = [1, 2, 3]
acc = []
acc.append(lst.pop(1))
print(acc, lst)
