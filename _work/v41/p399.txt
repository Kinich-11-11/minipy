// ---- P399: `max/min` 的 `default=` 支持（在 minmax 内实现，修正第 146 轮的错误）----
//
// 第 146 轮失败原因：段模型是"MComma 里已有的逗号 + 一个哨兵"。我把哨兵设成 `mmdc`（=关键字前那个逗号），
// 而那个逗号**已经在 MComma 里** ⇒ 重复 ⇒ 多出空区间段。
//
// 本轮正确做法：
//   * 扫描阶段识别 `Tok[p]=="default" && Tok[p+1]=="="`（深度 0），记 `mmdf=p`、`mmdc=` 它前面最近的顶层逗号；
//   * 扫描**之后**：若 `mmdc` 正是 MComma 的最后一个元素 ⇒ 把它**删掉**（这样可迭代对象就只有"一个参数"，
//     关键字那一段不再被当成第二个参数）；并令 `mmsent = mmdc - 1`；
//   * 把原来的 `addTo(MComma, hi+1)` 换成
//       若 mmdf>0 ⇒ addTo(MComma, mmsent)      ← 单参数情形（等价于原来的"无逗号"哨兵位置）
//       否则     ⇒ addTo(MComma, hi+1)         ← 与改动前完全一致
//   * 第一段求值**之前**先求 default 的值（`eval(mmdf+2, hi)`）存进 mmdfv/mmdft；
//   * 第一段求值之后（`mbest = ev` 处）：若给了 default 且 `ev == ""`（空可迭代）⇒ 用 default 作结果。
//
// 判定：`print(max([1, 2], default=0), min([], default=-1))` ⇒ `2 -1`；
//   且 `max([1,2])`、`min([])`（仍 ValueError）、`max([1,-5], key=abs)`（仍 -5）不回归。
patch('P399 max/min default= (fixed sentinel)', () => {
  const taken = new Set(Object.values(T(IT).variables).map((x) => x[0]));
  for (const n of ['mmdf', 'mmdc', 'mmclast', 'mmdfv', 'mmdft', 'mmsent']) {
    if (taken.has(n)) throw new Error('[P399] var taken: ' + n);
  }
  for (const n of ['mmdf', 'mmdc', 'mmclast', 'mmsent']) newVar(n, 0);
  newVar('mmdfv', ''); newVar('mmdft', '');
  const BB = T(IT).blocks;
  const deleteLast = (list) => I('data_deleteoflist', { fields: B.LISTF(list), inputs: { INDEX: B.REP(txt('last'), [10, '']) } });
  const defId = defIdOf('minmax %s %s %s');
  const head = BB[defId].next;
  if (!head) throw new Error('[P399] minmax body not found');
  let scanLoop = null, addHi1 = null, firstEval = null, mbestSet = null;
  let c = head;
  while (c) {
    const q = BB[c];
    if (!scanLoop && q.opcode === 'control_repeat_until') scanLoop = c;
    if (!addHi1 && q.opcode === 'data_addtolist' && q.fields && q.fields.LIST && q.fields.LIST[0] === 'MComma' && q.inputs && q.inputs.ITEM) {
      const it = q.inputs.ITEM[1];
      const w = typeof it === 'string' ? BB[it] : null;
      if (w && w.opcode === 'operator_add') addHi1 = c;
    }
    if (!firstEval && q.opcode === 'procedures_call' && q.mutation && String(q.mutation.proccode) === 'eval %s %s') firstEval = c;
    if (!mbestSet && q.opcode === 'data_setvariableto' && q.fields && q.fields.VARIABLE && q.fields.VARIABLE[0] === 'mbest') mbestSet = c;
    c = q.next;
  }
  if (!scanLoop || !addHi1 || !firstEval || !mbestSet) {
    throw new Error('[P399] pattern miss');
  }
  console.log('[P399] scan=' + scanLoop + ' addHi1=' + addHi1 + ' firstEval=' + firstEval + ' mbest=' + mbestSet);
  // (0) 初始化
  insertBeforeStmt(head, [setV('mmdf', num(0)), setV('mmdc', num(0)), setV('mmclast', num(0))]);
  // (1) 扫描体里识别 default 并追踪"最近的顶层逗号"
  const scanBody = BB[scanLoop].inputs.SUBSTACK[1];
  insertBeforeStmt(scanBody, [
    IF(EQ(VAR('ed'), num(0)), [
      IF(EQ(item('Tok', VAR('ek')), txt(',')), [setV('mmclast', VAR('ek'))])
    ]),
    IF(EQ(item('Tok', VAR('ek')), txt('default')), [
      IF(EQ(item('Tok', ADD(VAR('ek'), num(1))), txt('=')), [
        setV('mmdf', VAR('ek')),
        setV('mmdc', VAR('mmclast'))
      ])
    ])
  ]);
  // (2) 扫描之后、addTo(MComma, hi+1) 之前：删掉那个逗号并算好哨兵
  insertBeforeStmt(addHi1, [
    IF(GT(VAR('mmdf'), num(0)), [
      IF(EQ(item('MComma', llen('MComma')), VAR('mmdc')), [deleteLast('MComma')]),
      setV('mmsent', VAR('mmdc'))
    ])
  ]);
  // (3) 哨兵条件化
  {
    const nx = BB[addHi1].next;
    const gate = IFELSE(GT(VAR('mmdf'), num(0)),
      [addTo('MComma', VAR('mmsent'))],
      [addTo('MComma', ADD(argOf('hi'), num(1)))]);
    insertBeforeStmt(addHi1, [gate]);
    const w = BB[addHi1].parent;
    if (w && BB[w].next === addHi1) BB[w].next = nx;
    if (nx) BB[nx].parent = w;
  }
  // (4) 第一段求值之前先求 default 的值
  insertBeforeStmt(firstEval, [
    IF(GT(VAR('mmdf'), num(0)), [
      callProc('eval %s %s', [ADD(VAR('mmdf'), num(2)), argOf('hi')]),
      setV('mmdfv', VAR('ev')),
      setV('mmdft', VAR('evtype'))
    ])
  ]);
  // (5) 第一段求值之后：空可迭代 ⇒ 用 default
  spliceAfter(mbestSet, chain([
    IF(GT(VAR('mmdf'), num(0)), [
      IF(EQ(VAR('ev'), txt('')), [
        setV('mbest', VAR('mmdfv')),
        setV('mtype', VAR('mmdft'))
      ])
    ])
  ]));
  console.log('[P399] default= support installed (sentinel fixed)');
});
