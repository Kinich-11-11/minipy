// PZ106（临时调试）：找 eval 里那条"调用之后再调用"守卫（其实参为 `eval(?, cpe)`），
// 在它之前打印 cpe / Tok[cpe] / Tok[cpe+1] / hi / Tok[hi] / lo / Tok[lo]
patch('PZ106 chained-call guard values', () => {
  const BB = T(IT).blocks;
  const nameOf = (v) => {
    if (!Array.isArray(v)) return null;
    if (v[0] === 12 && typeof v[1] === 'string') return v[1];
    const id = typeof v[1] === 'string' ? v[1] : null;
    const q = id ? BB[id] : null;
    if (q && q.opcode === 'data_variable') return q.fields.VARIABLE[0];
    if (q && q.opcode === 'argument_reporter_string_number' && q.fields.VALUE) return '#' + q.fields.VALUE[0];
    if (Array.isArray(v[1])) return nameOf(v[1]);
    return null;
  };
  const ownerOf = (id) => {
    let cur = id;
    for (let i = 0; i < 400 && cur; i++) {
      const q = BB[cur];
      if (!q) return '?';
      if (q.opcode === 'procedures_definition') {
        const pr = BB[q.inputs.custom_block[1]];
        return pr && pr.mutation ? String(pr.mutation.proccode) : '?';
      }
      cur = q.parent;
    }
    return '(toplevel)';
  };
  let site = null;
  for (const id of Object.keys(BB)) {
    const b = BB[id];
    if (b.opcode !== 'procedures_call' || !b.mutation) continue;
    if (String(b.mutation.proccode) !== 'eval %s %s') continue;
    if (ownerOf(id) !== 'eval %s %s') continue;
    const vals = Object.values(b.inputs || {});
    if (vals.length < 2) continue;
    if (nameOf(vals[1]) === 'cpe') { site = id; break; }
  }
  if (!site) throw new Error('[PZ106] eval(?, cpe) site not found');
  console.log('[PZ106] site=' + site);
  // 找它外层第一层 IF
  let cur = BB[site].parent, g = null;
  for (let i = 0; i < 14 && cur; i++) {
    const q = BB[cur];
    if (!q) break;
    if (q.opcode === 'control_if' || q.opcode === 'control_if_else') { g = cur; break; }
    cur = q.parent;
  }
  const log = (label, expr) => callProc('log %s', [JOIN(txt(label), expr)]);
  const probes = [
    log('G cpe=', VAR('cpe')),
    log('G Tok[cpe]=', item('Tok', VAR('cpe'))),
    log('G Tok[cpe+1]=', item('Tok', ADD(VAR('cpe'), num(1)))),
    log('G hi=', argOf('hi')),
    log('G Tok[hi]=', item('Tok', argOf('hi'))),
    log('G lo=', argOf('lo')),
    log('G Tok[lo]=', item('Tok', argOf('lo')))
  ];
  if (g) {
    try {
      insertBeforeStmt(g, probes);
      console.log('[PZ106] probe inserted before guard ' + g);
    } catch (e) {
      console.log('[PZ106] insertBefore failed: ' + e.message);
    }
  } else {
    console.log('[PZ106] WARN: no enclosing IF');
  }
});
