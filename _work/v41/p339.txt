// ---- P339: 调用 lambda 时保存/恢复 Tok（#4 闭包/链式调用的根因）----
//
// 第 90/91 轮定位：`calluser` 里 `FnKind[fi] == "L"`（调用 lambda）那条分支是
//
//     0: set <var>
//     1: tokenize <lambda 体>      ← ★ 把**全局 Tok 列表整个换成 lambda 体的记号**
//     2: eval(lo, hi)              ← 在这份被换掉的 Tok 上求值
//     3: set <var>
//
// 而没有人在事后恢复调用者的 Tok。于是"调用之后再调用"（如 `adder(5)(1)`：
// 外层 eval 正在扫 `adder ( 5 ) ( 1 )`，中途调用 lambda 把 Tok 换走）
// 返回后外层继续按旧的 lo/hi 读**已经变了的** Tok → 括号深度变负 → `SyntaxError: parentheses`。
// （第 91 轮的对照实验：把 10 道"常量折叠写回 Tok"的门关掉，`_cl` 毫无变化 ⇒ 不是折叠的问题。）
//
// 修法：在该 tokenize **之前**把 Tok 整表存进 `TokSave`，在三步之后**写回**。
patch('P339 calluser restores Tok after tokenizing a lambda', () => {
  const taken = new Set(Object.values(T(IT).variables).map((x) => x[0]));
  for (const n of ['tsi', 'tsn']) if (taken.has(n)) throw new Error('[P339] var taken: ' + n);
  newVar('tsi', 0); newVar('tsn', 0);
  const have = new Set(Object.values(T(IT).lists).map((x) => x[0]));
  if (!have.has('TokSave')) B.addList(IT, 'TokSave', []);
  const BB = T(IT).blocks;
  const litOf = (d) => {
    if (!Array.isArray(d)) return null;
    if ((d[0] === 10 || d[0] === 4) && typeof d[1] !== 'object') return String(d[1]);
    if ((d[0] === 1 || d[0] === 2 || d[0] === 3) && Array.isArray(d[1])) return litOf(d[1]);
    return null;
  };
  const nameOf = (v) => {
    if (!Array.isArray(v)) return null;
    if (v[0] === 12 && typeof v[1] === 'string') return v[1];
    const id = typeof v[1] === 'string' ? v[1] : null;
    const q = id ? BB[id] : null;
    if (q && q.opcode === 'data_variable') return q.fields.VARIABLE[0];
    if (Array.isArray(v[1])) return nameOf(v[1]);
    return null;
  };
  // 收集 calluser / atom 里所有 tokenize 调用（求值中途会换掉 Tok 的站点）
  const sites = [];
  for (const id of Object.keys(BB)) {
    const b = BB[id];
    if (b.opcode !== "procedures_call" || !b.mutation) continue;
    if (String(b.mutation.proccode).indexOf("tokenize") < 0) continue;
    let cur = id, own = "?";
    for (let i = 0; i < 400 && cur; i++) {
      const q = BB[cur];
      if (!q) break;
      if (q.opcode === "procedures_definition") {
        const pr = BB[q.inputs.custom_block[1]];
        own = pr && pr.mutation ? String(pr.mutation.proccode) : "?";
        break;
      }
      cur = q.parent;
    }
    if (!own) continue;
    if (own.indexOf("calluser") < 0 && own.indexOf("atom") < 0) continue;
    sites.push(id);
  }
  if (!sites.length) throw new Error("[P339] no tokenize site in calluser/atom");
  if (!site) throw new Error('[P339] lambda tokenize site not found');
  console.log('[P339] tokenize site=' + site);
  const save = [
    clearL("TokSave"),
    setV("tsi", num(1)),
    REPEATU(GT(VAR("tsi"), llen("Tok")), [
      addTo("TokSave", item("Tok", VAR("tsi"))),
      chgV("tsi", num(1))
    ])
  ];
  const restore = [
    setV("tsn", llen("TokSave")),
    setV("tsi", num(1)),
    REPEATU(GT(VAR("tsi"), VAR("tsn")), [
      I("data_replaceitemoflist", {
        fields: B.LISTF("Tok"),
        inputs: { INDEX: B.REP(VAR("tsi"), [4, "0"]), ITEM: B.REP(item("TokSave", VAR("tsi")), [10, ""]) }
      }),
      chgV("tsi", num(1))
    ])
  ];
  let nSite = 0;
  for (const site of sites) {
    try {
      insertBeforeStmt(site, save.slice());
      const a1 = BB[site].next;
      const a2 = a1 ? BB[a1].next : null;
      if (!a2) { console.log("[P339] skip (no room) " + site); continue; }
      spliceAfter(a2, chain(restore));
      nSite++;
    } catch (e) {
      console.log("[P339] skip " + site + " : " + e.message);
    }
  }
  console.log('[P339] tokenize site=' + site);
  const save = [
    clearL('TokSave'),
    setV('tsi', num(1)),
    REPEATU(GT(VAR('tsi'), llen('Tok')), [
      addTo('TokSave', item('Tok', VAR('tsi'))),
      chgV('tsi', num(1))
    ])
  ];
  const restore = [
    setV('tsn', llen('TokSave')),
    setV('tsi', num(1)),
    REPEATU(GT(VAR('tsi'), VAR('tsn')), [
      I('data_replaceitemoflist', {
        fields: B.LISTF('Tok'),
        inputs: { INDEX: B.REP(VAR('tsi'), [4, '0']), ITEM: B.REP(item('TokSave', VAR('tsi')), [10, '']) }
      }),
      chgV('tsi', num(1))
    ])
  ];
  insertBeforeStmt(site, save);
  // 还原：接在 tokenize + eval(lo,hi) + set 之后
  const after1 = BB[site].next;                       // eval(lo,hi)
  if (!after1) throw new Error('[P339] no eval after tokenize');
  const after2 = BB[after1].next;                     // set
  if (!after2) throw new Error('[P339] no set after eval');
  spliceAfter(after2, chain(restore));
  console.log("[P339] Tok save/restore installed at " + nSite + " site(s)");
});
