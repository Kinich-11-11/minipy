### A7 gcd and lcm
def gcd(a, b):
    while b:
        a, b = b, a % b
    return a
def lcm(a, b):
    return a // gcd(a, b) * b
print(gcd(48, 18), gcd(17, 5), gcd(-24, 36))
print(lcm(12, 18), lcm(7, 13))
print(lcm(2, 3 * 5 * 7 * 11 * 13))
print(gcd(0, 5), gcd(5, 0))

    out = ''
    for ch in s.lower():
        if ch.isalpha():
            out += ch
    return out

### B6 caesar cipher
def caesar(s, k):
    out = ''
    for ch in s:
        if 'a' <= ch <= 'z':
            out += chr((ord(ch) - 97 + k) % 26 + 97)
        elif 'A' <= ch <= 'Z':
            out += chr((ord(ch) - 65 + k) % 26 + 65)
        else:
            out += ch
    return out
enc = caesar('Hello, World!', 3)
print(enc)
print(caesar(enc, -3))
print(caesar('abc xyz', 1))

### D4 generator pipeline
def nums(n):
    i = 0
    while i < n:
        yield i
        i += 1
def evens(src):
    for v in src:
        if v % 2 == 0:
            yield v
def squares(src):
    for v in src:
        yield v * v
print(list(squares(evens(nums(10)))))
print(sum(squares(evens(nums(8)))))

### D6 fibonacci generator
def fibgen(n):
    a, b = 0, 1
    i = 0
    while i < n:
        yield a
        a, b = b, a + b
        i += 1
print(list(fibgen(10)))
print(sum(fibgen(12)))
print([v for v in fibgen(8) if v % 2 == 0])
### D7 infinite source take
def naturals():
    n = 1
    while True:
        yield n
        n += 1
g = naturals()
out = []
for _ in range(5):
    out.append(next(g))
print(out)
def take(gen, k):
    res = []
    i = 0
    while i < k:
        res.append(next(gen))
        i += 1
    return res
print(take(naturals(), 6))
print(take((x * x for x in naturals()), 4))
