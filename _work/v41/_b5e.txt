### D1 plain f g 1
def dbl(x):
    return x * 2
def inc(x):
    return x + 1
def ap6(f, g):
    return f(g(1))
print(ap6(dbl, inc))
### D2 global vars lambda nested
def dbl2(x):
    return x * 2
def inc2(x):
    return x + 1
ff = dbl2
gg = inc2
lam = lambda x: ff(gg(x))
print(lam(5))
### D3 in-def nested lambda one level
def ap7(f):
    def inc7(x):
        return x + 1
    return lambda x: f(inc7(x))
h7 = ap7(dbl2)
print(h7(5))
