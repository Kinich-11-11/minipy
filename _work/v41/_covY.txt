### append in recursion pre order
def f(n, out):
    if n == 0:
        return
    out.append(n)
    f(n - 1, out)
out = []
f(3, out)
print(out)
### append in recursion post order
def g(n, out):
    if n == 0:
        return
    g(n - 1, out)
    out.append(n)
out = []
g(3, out)
print(out)
### append in nested function
def h(out):
    def inner():
        out.append(1)
    inner()
    return out
o = []
print(h(o), o)

### sorted custom objects needs lt
class P:
    def __init__(self, n):
        self.n = n
    def __lt__(self, other):
        return self.n < other.n
ps = [P(3), P(1)]
try:
    print([p.n for p in sorted(ps)])
except Exception as e:
    print('sort failed')
