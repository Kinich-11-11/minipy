### fstring escaped braces pair
print(f'{{a}}')
### fstring escaped braces with value
n = 5
print(f'{{{n}}}')
### fstring empty braces escaped
print(f'{{}}')
### fstring plain substitution
n = 5
print(f'{n}')
### fstring escaped name
print(f'{{n}}')
### fstring mixed text
n = 1
print(f'x{{y}}z{n}')
### fstring value then escaped
n = 1
print(f'{n}{{}}')
### fstring double escaped pair
print(f'{{{{}}}}')
### fstring spec still works
n = 5
print(f'{n:>3}|')
### fstring single open brace escaped
print(f'{{')
### fstring single close brace escaped
print(f'}}')
### fstring escaped with format spec
x = 3.14159
print(f'{{{x:.2f}}}')
### fstring escaped with dict access
d = {'k': 1}
print(f'{{{d["k"]}}}')
### fstring centred spec
n = 42
print(f'[{n:^5}]')
### fstring adjacent substitutions
a = 1
b = 2
print(f'{a}{b}')
### fstring escaped braces around text
print(f'a{{b}}c')
### fstring escaped brace at start and end
n = 7
print(f'{{{n}}}')
### fstring literal brace in spec
n = 3
print(f'{n:{{>4}}')
