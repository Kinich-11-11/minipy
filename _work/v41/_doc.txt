### d13 diamond second branch
class A:
    pass
class B(A):
    pass
class C(A):
    pass
class D(B, C):
    pass
print(isinstance(D(), C), issubclass(D, C))
### d4 closure returned
def adder(n):
    return lambda x: x + n
print(adder(5)(1))
### d4b nonlocal
def o():
    n = 0
    def i():
        nonlocal n
        n += 1
        return n
    return i
print(o()())
