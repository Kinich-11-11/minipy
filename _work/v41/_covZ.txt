### finally return overrides
def h():
    try:
        return 'a'
    finally:
        return 'b'
print(h())
### finally with break
out = []
for i in range(3):
    try:
        if i == 1:
            break
    finally:
        out.append(i)
print(out)

### custom exception isinstance
class MyErr(Exception):
    pass
try:
    raise MyErr('y')
except MyErr as e:
    print(isinstance(e, Exception), isinstance(e, MyErr))
### custom exception with init attrs
class AppErr(Exception):
    def __init__(self, code):
        self.code = code
try:
    raise AppErr(7)
except AppErr as e:
    print(e.code)

### return container mutated in nested function
def build(out):
    def add():
        out.append(1)
    add()
    return out
r = build([])
print(r)
### return list mutated in loop
def build2():
    xs = []
    for i in range(3):
        xs.append(i)
    return xs
print(build2())
### return dict mutated in loop
def build3():
    d = {}
    for i in range(3):
        d[i] = i
    return d
print(build3())
### recursion appending to passed list
def rec(n, out):
    if n == 0:
        return out
    out.append(n)
    return rec(n - 1, out)
print(rec(3, []))
### recursion building local list
def rec2(n):
    if n == 0:
        return []
    xs = rec2(n - 1)
    xs.append(n)
    return xs
print(rec2(3))
### append then return in same function
def addone(xs):
    xs.append(9)
    return xs
print(addone([1]), addone([1, 2]))
### mutate then read in same function
def m(xs):
    xs.append(9)
    return len(xs)
print(m([1]))
### dict set then read same function
def s(d, k, v):
    d[k] = v
    return d.get(k)
print(s({}, 'a', 1))
### list set then read same function
def t(a):
    a[0] = 9
    return a[0]
print(t([1, 2]))
