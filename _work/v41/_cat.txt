### str concat simple
print("(" + str(3) + ")")
### str concat comma
print("a" + "," + "b")
### return concat call
def f(x):
    return "(" + str(x) + ")"
print(f(3))
### return concat attr
class P:
    def __init__(self):
        self.x = 1
    def s(self):
        return str(self.x)
print(P().s())
### return concat attr plus
class Q:
    def __init__(self):
        self.x = 2
    def s(self):
        return "(" + str(self.x) + "," + str(self.x) + ")"
print(Q().s())
### paren string three
print(("(" + ")"))
### str of str call chain
print(str("a") + str("b"))
### concat with paren literal
print("(" + "x" + ")")
