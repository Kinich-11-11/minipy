### list slicing and mutation
a = [1, 2, 3, 4, 5]
print(a[1:3], a[:2], a[3:], a[::-1], a[::2])
### slice assignment
a = [1, 2, 3, 4, 5]
a[1:3] = [9]
print(a)
a = [1, 2, 3, 4, 5]
a[1:3] = [9, 8, 7]
print(a)
a = [1, 2, 3]
a[1:1] = [0]
print(a)
### slice step assignment
a = [1, 2, 3, 4, 5]
a[::2] = [0, 0, 0]
print(a)
### slice assignment length mismatch caught
a = [1, 2, 3, 4]
try:
    a[::2] = [0]
except ValueError as e:
    print('ve', e)

### sort and sorted
a = [3, 1, 2]
print(sorted(a), a)
a.sort()
print(a)
a.sort(reverse=True)
print(a)
print(sorted([1, 2, 3], reverse=True), sorted([]))
### sort with key
print(sorted(['bb', 'a', 'ccc'], key=len), sorted(['bb', 'a', 'ccc'], key=len, reverse=True))
print(sorted([-3, 1, -2], key=abs), sorted([(2, 'b'), (1, 'a')], key=lambda p: p[0]))
### sort stability
print(sorted([(1, 'b'), (1, 'a'), (0, 'c')]))
print(sorted(['b1', 'a2', 'b3', 'a4'], key=lambda s: s[0]))
### sort mixed types caught
try:
    print(sorted([1, 'a']))
except TypeError as e:
    print('te')

### extend and concatenation

### slice copy vs alias
a = [[1], [2]]

### outer list mutate does not affect copy
a = [1, 2]

print(a, b)
### shallow copy method

print(a, b)

### tuple basics
t = (1, 2, 3)
print(t[0], t[-1], t[1:], t[::-1], len(t), t + (4,), t * 2)
print((1,), type((1,)), type((1)), bool(()), bool((0,)))
### tuple immutability caught
t = (1, 2)
try:
    t[0] = 9

### tuple methods and comparison

### tuple unpacking
