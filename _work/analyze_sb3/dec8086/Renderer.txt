DEFINE FAST RGBToHSV (r) (g) (b):
  set cMax to (r)
  set cMin to (r)
  IF (g) > cMax THEN:
    set cMax to (g)
  IF (g) < cMin THEN:
    set cMin to (g)
  IF (b) > cMax THEN:
    set cMax to (b)
  IF (b) < cMin THEN:
    set cMin to (b)
  set delta to (cMax - cMin)
  set V to cMax
  IF cMax = "0" THEN:
    set S to "0"
  ELSE:
    set S to (delta / cMax)
  IF delta = "0" THEN:
    set H to "0"
  ELSE:
    IF cMax = (r) THEN:
      set H to (((g) - (b)) / delta)
      set H to (H mod 6)
    ELSE:
      IF cMax = (g) THEN:
        set H to ((((b) - (r)) / delta) + 2)
      ELSE:
        set H to ((((r) - (g)) / delta) + 4)
  set H to (H / 6)

DEFINE setSpriteColor (h) (s) (v):
  set ghost effect to 0
  IF (s) = "0" THEN:
    set color effect to INF
    set brightness effect to ((v) * 100)
  ELSE:
    set color effect to ((h) * 200)
    set brightness effect to (((v) - 1) * 100)
