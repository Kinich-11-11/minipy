WHEN flag clicked:
  hide

WHEN I receive "HALT":
  stop other scripts in sprite
  set running to 0
  set busy to 0
  set dirty to 1

WHEN I receive "RUN":
  set error to ""
  set status to "Running"
  set steps to 0
  set pc to 1
  set fnlimit to 0
  delete all of Output
  delete all of Names
  delete all of Values
  delete all of Types
  delete all of LoopHead
  delete all of LoopEnd
  delete all of LoopKind
  delete all of LoopName
  delete all of LoopValue
  delete all of LoopStop
  delete all of LoopStep
  delete all of LoopText
  set outpage to 1
  set varpage to 1
  prepare
  REPEAT UNTIL ((running = 0) or ((pc > (length of Code)) and ((length of LoopHead) = 0))) or (not (error = "")):
    boundary
    IF ((not (pc > (length of Code))) and (running = 1)) and (error = "") THEN:
      tokenize (item pc of Code)
      IF (length of Tok) > 0 THEN:
        set selected to pc
        IF (pc < page) or (pc > (page + 15)) THEN:
          set page to pc
        set dirty to 1
        IF mode = 1 THEN:
          set status to "Paused / STEP to advance"
          wait until ((ticket > 0) or (mode = 0)) or (running = 0)
          change ticket by -1
        IF running = 1 THEN:
          set status to "Running"
          change steps by 1
          IF steps > (10000 + (speed * 90000)) THEN:
            fail "LimitError: step limit reached"
          ELSE:
            execute
      ELSE:
        change pc by 1
      IF (speed = 0) or ((steps mod 50) = 0) THEN:
        set dirty to 1
        wait 0 seconds
  IF not (error = "") THEN:
    set status to "Error / see console"
    log ("Line " join (pc join (": " join error)))
  ELSE:
    IF stopreq = 0 THEN:
      set status to "Finished"
  set running to 0
  set busy to 0
  set dirty to 1

DEFINE FAST fail (message):
  IF error = "" THEN:
    set error to (message)

DEFINE FAST log (message):
  set lj to 1
  set lt to ""
  REPEAT (length of (message)):
    set lt to (lt join (letter lj of (message)))
    IF (length of lt) = 30 THEN:
      add lt to Output
      set lt to ""
    change lj by 1
  IF (not (lt = "")) or ((length of (message)) = 0) THEN:
    add lt to Output
  IF (length of Output) > 300 THEN:
    delete 1 of Output
  set otgt to ((length of Output) - 8)
  IF otgt < 1 THEN:
    set otgt to 1
  IF outpage > (otgt - 3) THEN:
    set outpage to otgt
  IF outpage < 1 THEN:
    set outpage to 1
  set dirty to 1

DEFINE FAST prepare:
  set cs to " !"#$%&'()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\]^_`abcdefghijklmnopqrstuvwxyz{|}~"
  IF (length of Code) > 200 THEN:
    fail "LimitError: 200 lines maximum"
  set branchjump to 0
  delete all of Indents
  delete all of Ends
  delete all of Elses
  set ei to 1
  set previndent to 0
  set prevhead to ""
  set backline to 0
  REPEAT (length of Code):
    set es to (item ei of Code)
    IF (length of es) > 240 THEN:
      set pc to ei
      fail "LimitError: 240 chars per line"
    set ej to 1
    REPEAT UNTIL (ej > (length of es)) or (not ((letter ej of es) = " ")):
      change ej by 1
    set indent to (ej - 1)
    IF (ej > (length of es)) or ((letter ej of es) = "#") THEN:
      set indent to -1
    ELSE:
      IF not ((indent mod 4) = 0) THEN:
        set pc to ei
        fail "IndentationError: use 4 spaces"
    IF (not (indent < 0)) and (error = "") THEN:
      set pc to ei
      tokenize es
      IF indent > previndent THEN:
        IF (not (indent = (previndent + 4))) or (not (((((prevhead = "if") or (prevhead = "else")) or (prevhead = "while")) or (prevhead = "for")) or (prevhead = "elif"))) THEN:
          fail "IndentationError: unexpected indent"
      ELSE:
        IF ((((prevhead = "if") or (prevhead = "else")) or (prevhead = "while")) or (prevhead = "for")) or (prevhead = "elif") THEN:
          set pc to backline
          fail "IndentationError: expected block"
      set count to (length of Tok)
      set prevhead to (item 1 of Tok)
      set previndent to indent
      set backline to ei
      IF ((((prevhead = "if") or (prevhead = "else")) or (prevhead = "while")) or (prevhead = "for")) or (prevhead = "elif") THEN:
        IF not ((item count of Tok) = ":") THEN:
          fail "SyntaxError: expected colon"
      IF (prevhead = "else") and (not ((length of Tok) = 2)) THEN:
        fail "SyntaxError: else:"
    add indent to Indents
    add 0 to Ends
    add 0 to Elses
    change ei by 1
  IF ((((prevhead = "if") or (prevhead = "else")) or (prevhead = "while")) or (prevhead = "for")) or (prevhead = "elif") THEN:
    set pc to backline
    fail "IndentationError: expected block"
  set ei to 1
  REPEAT (length of Code):
    set indent to (item ei of Indents)
    set ej to (ei + 1)
    REPEAT UNTIL (ej > (length of Code)) or ((not ((item ej of Indents) < 0)) and (not ((item ej of Indents) > indent))):
      change ej by 1
    replace item ei of Ends with ej
    IF (not (ej > (length of Code))) and ((item ej of Indents) = indent) THEN:
      tokenize (item ej of Code)
      IF ((item 1 of Tok) = "else") or ((item 1 of Tok) = "elif") THEN:
        replace item ei of Elses with ej
    change ei by 1
  IF error = "" THEN:
    set ei to 1
    REPEAT (length of Code):
      tokenize (item ei of Code)
      IF ((item 1 of Tok) = "else") or ((item 1 of Tok) = "elif") THEN:
        set ej to (ei - 1)
        REPEAT UNTIL (ej < 1) or ((item ej of Indents) = (item ei of Indents)):
          change ej by -1
        IF ej < 1 THEN:
          set pc to ei
          fail "SyntaxError: elif/else without if"
        ELSE:
          tokenize (item ej of Code)
          IF not (((item 1 of Tok) = "if") or ((item 1 of Tok) = "elif")) THEN:
            set pc to ei
            fail "SyntaxError: elif/else without if"
      change ei by 1
  IF error = "" THEN:
    set pc to 1

DEFINE FAST boundary:
  set lc to (length of LoopHead)
  REPEAT UNTIL (lc = 0) or (not (pc = (item lc of LoopEnd))):
    IF (item lc of LoopKind) = "while" THEN:
      set pc to (item lc of LoopHead)
      popLoop
    ELSE:
      IF (item lc of LoopKind) = "list" THEN:
        set lv to ((item lc of LoopValue) + 1)
        IF lv > (item lc of LoopStop) THEN:
          popLoop
        ELSE:
          replace item lc of LoopValue with lv
          listitem (item lc of LoopText) lv
          assign (item lc of LoopName) ev evtype
          set pc to ((item lc of LoopHead) + 1)
      ELSE:
        IF (item lc of LoopKind) = "str" THEN:
          set lv to ((item lc of LoopValue) + 1)
          IF (item lc of LoopStop) < lv THEN:
            popLoop
          ELSE:
            replace item lc of LoopValue with lv
            assign (item lc of LoopName) (letter lv of (item lc of LoopText)) "S"
            set pc to ((item lc of LoopHead) + 1)
        ELSE:
          set lv to ((item lc of LoopValue) + (item lc of LoopStep))
          set ls to (item lc of LoopStep)
          set le to (item lc of LoopStop)
          IF ((ls > 0) and (lv < le)) or ((ls < 0) and (lv > le)) THEN:
            replace item lc of LoopValue with lv
            assign (item lc of LoopName) lv "N"
            set pc to ((item lc of LoopHead) + 1)
          ELSE:
            popLoop
    set lc to (length of LoopHead)

DEFINE FAST popLoop:
  set lc to (length of LoopHead)
  delete lc of LoopHead
  delete lc of LoopEnd
  delete lc of LoopKind
  delete lc of LoopName
  delete lc of LoopValue
  delete lc of LoopStop
  delete lc of LoopStep
  delete (length of LoopText) of LoopText

DEFINE FAST pushLoop (typ) (vn) (vstart) (vstop) (vstep):
  add pc to LoopHead
  add (item pc of Ends) to LoopEnd
  add (typ) to LoopKind
  add (vn) to LoopName
  add (vstart) to LoopValue
  add (vstop) to LoopStop
  add (vstep) to LoopStep
  add "" to LoopText

DEFINE FAST compareText (va) (vb):
  set cmp to 0
  set ci to 1
  REPEAT UNTIL ((ci > (length of (va))) or (ci > (length of (vb)))) or (not (cmp = 0)):
    switch costume to ("_" join (letter ci of (va)))
    set ca to (costume number)
    switch costume to ("_" join (letter ci of (vb)))
    set cb to (costume number)
    IF ca < cb THEN:
      set cmp to -1
    IF ca > cb THEN:
      set cmp to 1
    change ci by 1
  IF cmp = 0 THEN:
    IF (length of (va)) < (length of (vb)) THEN:
      set cmp to -1
    IF (length of (va)) > (length of (vb)) THEN:
      set cmp to 1

DEFINE FAST truth (vv) (vt):
  set truthv to 1
  IF (vt) = "S" THEN:
    IF (length of (vv)) = 0 THEN:
      set truthv to 0
  ELSE:
    IF (vv) = 0 THEN:
      set truthv to 0

DEFINE FAST lookup (vn):
  set found to 1
  set cmp to 1
  REPEAT UNTIL (found > (length of Names)) or (cmp = 0):
    IF ((length of (item found of Names)) = (length of (vn))) and ((item found of Names) = (vn)) THEN:
      compareText (item found of Names) (vn)
      IF not (cmp = 0) THEN:
        change found by 1
    ELSE:
      set cmp to 1
    IF not (cmp = 0) THEN:
      change found by 1
  IF found > (length of Names) THEN:
    set found to 0

DEFINE FAST assign (vn) (vv) (vt):
  lookup (vn)
  IF found = 0 THEN:
    add (vn) to Names
    add (vv) to Values
    add (vt) to Types
  ELSE:
    replace item found of Values with (vv)
    replace item found of Types with (vt)

DEFINE FAST tokenize (source):
  checkASCII (source)
  delete all of Tok
  delete all of Kind
  set ep to 1
  REPEAT UNTIL (ep > (length of (source))) or (not (error = "")):
    set ec to (letter ep of (source))
    IF ec = " " THEN:
      change ep by 1
    ELSE:
      IF ec = "#" THEN:
        set ep to ((length of (source)) + 1)
      ELSE:
        IF (ec = dq) or (ec = "'") THEN:
          set eq to ec
          set et to ""
          change ep by 1
          REPEAT UNTIL (ep > (length of (source))) or ((letter ep of (source)) = eq):
            set ec to (letter ep of (source))
            IF ec = slash THEN:
              change ep by 1
              set ec to (letter ep of (source))
            set et to (et join ec)
            change ep by 1
          IF ep > (length of (source)) THEN:
            fail "SyntaxError: unclosed string"
          add et to Tok
          add "S" to Kind
          change ep by 1
        ELSE:
          IF "0123456789." contains ec THEN:
            set et to ""
            REPEAT UNTIL (ep > (length of (source))) or (not ("0123456789." contains (letter ep of (source)))):
              set et to (et join (letter ep of (source)))
              change ep by 1
            add et to Tok
            add "N" to Kind
          ELSE:
            IF "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ_" contains ec THEN:
              set et to ""
              REPEAT UNTIL (ep > (length of (source))) or (not ("abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ_0123456789" contains (letter ep of (source)))):
                set et to (et join (letter ep of (source)))
                change ep by 1
              add et to Tok
              add "I" to Kind
            ELSE:
              set et to ec
              change ep by 1
              IF not (ep > (length of (source))) THEN:
                set en to (letter ep of (source))
                IF (((((((((((ec = "=") or (ec = "!")) or (ec = "<")) or (ec = ">")) or (ec = "+")) or (ec = "-")) or (ec = "*")) or (ec = "/")) or (ec = "%")) and (en = "=")) or ((ec = "/") and (en = "/"))) or ((ec = "*") and (en = "*")) THEN:
                  set et to (et join en)
                  change ep by 1
              add et to Tok
              add "O" to Kind

DEFINE FAST rank (op):
  set erank to 0
  IF (op) = "or" THEN:
    set erank to 1
  IF (op) = "and" THEN:
    set erank to 2
  IF ((((((op) = "==") or ((op) = "!=")) or ((op) = "<")) or ((op) = ">")) or ((op) = "<=")) or ((op) = ">=") THEN:
    set erank to 3
  IF ((op) = "+") or ((op) = "-") THEN:
    set erank to 4
  IF ((((op) = "*") or ((op) = "/")) or ((op) = "//")) or ((op) = "%") THEN:
    set erank to 5
  IF (op) = "**" THEN:
    set erank to 6
  IF (op) = "in" THEN:
    set erank to 3
  IF (op) = "notin" THEN:
    set erank to 3

DEFINE FAST eval (lo) (hi):
  change fnlimit by 1
  IF fnlimit > 64 THEN:
    fail "LimitError: expression depth"
  set ev to 0
  set evtype to "N"
  IF error = "" THEN:
    IF (lo) > (hi) THEN:
      fail "SyntaxError: missing expression"
    ELSE:
      set ed to 0
      set ecut to 0
      set emin to 100
      set ek to (lo)
      REPEAT (((hi) - (lo)) + 1):
        IF not ((item ek of Kind) = "S") THEN:
          set eop to (item ek of Tok)
          IF eop = "(" THEN:
            change ed by 1
          IF eop = ")" THEN:
            change ed by -1
          IF eop = "[" THEN:
            change ed by 1
          IF eop = "]" THEN:
            change ed by -1
          IF ed < 0 THEN:
            fail "SyntaxError: parentheses"
          IF ed = 0 THEN:
            IF (eop = "not") and ((item (ek + 1) of Tok) = "in") THEN:
              set eop to "notin"
            ELSE:
              IF (eop = "in") and ((item (ek - 1) of Tok) = "not") THEN:
                set eop to "skip"
            rank eop
            IF (eop = "+") or (eop = "-") THEN:
              IF ek = (lo) THEN:
                set erank to 0
              ELSE:
                set eprev to (item (ek - 1) of Tok)
                IF (((((((((((((((eprev = "(") or (eprev = "+")) or (eprev = "-")) or (eprev = "*")) or (eprev = "/")) or (eprev = "%")) or (eprev = "//")) or (eprev = "==")) or (eprev = "<")) or (eprev = ">")) or (eprev = "<=")) or (eprev = ">=")) or (eprev = "!=")) or (eprev = "and")) or (eprev = "or")) or (eprev = "**") THEN:
                  set erank to 0
            IF (erank > 0) and (not (erank > emin)) THEN:
              set emin to erank
              set ecut to ek
        change ek by 1
      IF not (ed = 0) THEN:
        fail "SyntaxError: parentheses"
      IF ((item (lo) of Tok) = "not") and (not (emin < 3)) THEN:
        unary "not" ((lo) + 1) (hi)
      ELSE:
        IF ecut > 0 THEN:
          pair (lo) (hi) ecut (item ecut of Tok)
        ELSE:
          IF (lo) = (hi) THEN:
            atom (lo)
          ELSE:
            IF ((item (lo) of Tok) = "(") and ((item (hi) of Tok) = ")") THEN:
              eval ((lo) + 1) ((hi) - 1)
            ELSE:
              IF (((item (lo) of Tok) = "-") or ((item (lo) of Tok) = "+")) or ((item (lo) of Tok) = "not") THEN:
                unary (item (lo) of Tok) ((lo) + 1) (hi)
              ELSE:
                IF ((item ((lo) + 1) of Tok) = "(") and ((item (hi) of Tok) = ")") THEN:
                  set ed to 0
                  set narg to 0
                  set qcut to 0
                  set argi to ((lo) + 2)
                  REPEAT UNTIL argi > ((hi) - 1):
                    set eop to (item argi of Tok)
                    IF (eop = "(") or (eop = "[") THEN:
                      change ed by 1
                    IF (eop = ")") or (eop = "]") THEN:
                      change ed by -1
                    IF (ed = 0) and (eop = ",") THEN:
                      change narg by 1
                      IF qcut = 0 THEN:
                        set qcut to argi
                    change argi by 1
                  IF narg = 0 THEN:
                    callfn (item (lo) of Tok) ((lo) + 2) ((hi) - 1)
                  ELSE:
                    IF (narg = 1) and ((((item (lo) of Tok) = "max") or ((item (lo) of Tok) = "min")) or (((item (lo) of Tok) = "pow") or ((item (lo) of Tok) = "round"))) THEN:
                      builtin2 (item (lo) of Tok) ((lo) + 2) qcut ((hi) - 1)
                    ELSE:
                      fail "boperat_n7hrawsp"
                ELSE:
                  IF ((item ((lo) + 1) of Tok) = ".") and ((item (hi) of Tok) = ")") THEN:
                    methodcall (lo) (hi)
                  ELSE:
                    IF ((item (lo) of Tok) = "[") and ((item (hi) of Tok) = "]") THEN:
                      listlit ((lo) + 1) ((hi) - 1)
                    ELSE:
                      IF ((item ((lo) + 1) of Tok) = "[") and ((item (hi) of Tok) = "]") THEN:
                        indexget (lo) (hi)
                      ELSE:
                        fail "SyntaxError: expression"
  IF (length of ev) > 2048 THEN:
    fail "LimitError: value exceeds 2048 chars"
  change fnlimit by -1

DEFINE FAST atom (idx):
  set ev to (item (idx) of Tok)
  set evtype to (item (idx) of Kind)
  IF evtype = "N" THEN:
    numbercheck ev
    IF en = 0 THEN:
      fail "SyntaxError: invalid number"
  IF evtype = "I" THEN:
    IF ev = "True" THEN:
      set ev to 1
      set evtype to "N"
    ELSE:
      IF ev = "False" THEN:
        set ev to 0
        set evtype to "N"
      ELSE:
        lookup ev
        IF found = 0 THEN:
          fail ("NameError: " join ev)
          set ev to 0
        ELSE:
          set ev to (item found of Values)
          set evtype to (item found of Types)
  ELSE:
    IF (not (evtype = "N")) and (not (evtype = "S")) THEN:
      fail "SyntaxError: unexpected token"

DEFINE FAST pair (lo) (hi) (cut) (op):
  eval (lo) ((cut) - 1)
  truth ev evtype
  IF not ((((op) = "and") and (truthv = 0)) or (((op) = "or") and (truthv = 1))) THEN:
    IF ((op) = "not") and ((item ((cut) + 1) of Tok) = "in") THEN:
      rightside ev evtype "notin" ((cut) + 2) (hi)
    ELSE:
      rightside ev evtype (op) ((cut) + 1) (hi)

DEFINE FAST rightside (leftval) (lefttype) (op) (lo) (hi):
  eval (lo) (hi)
  IF error = "" THEN:
    IF (op) = "+" THEN:
      IF ((lefttype) = "S") and (evtype = "S") THEN:
        set ev to ((leftval) join ev)
      ELSE:
        IF ((lefttype) = "N") and (evtype = "N") THEN:
          set ev to ((leftval) + ev)
        ELSE:
          fail "TypeError: use str(number)"
    IF ((op) = "*") and (((lefttype) = "S") or (not (evtype = "N"))) THEN:
      set yb to ""
      IF (lefttype) = "S" THEN:
        IF ev > 0 THEN:
          REPEAT ev:
            set yb to (yb join (leftval))
      ELSE:
        IF (leftval) > 0 THEN:
          REPEAT (leftval):
            set yb to (yb join ev)
      set ev to yb
      set evtype to "S"
    IF (op) = "**" THEN:
      IF (not ((lefttype) = "N")) or (not (evtype = "N")) THEN:
        fail "TypeError: numeric operands"
      IF ((ev mod 1) = 0) and (not ((abs of ev) > 1024)) THEN:
        set qpow to 1
        REPEAT (abs of ev):
          set qpow to (qpow * (leftval))
        IF ev < 0 THEN:
          set ev to (1 / qpow)
        ELSE:
          set ev to qpow
      ELSE:
        IF not ((leftval) > 0) THEN:
          fail "ValueError: math domain error"
        set ev to (e ^ of (ev * (ln of (leftval))))
      set evtype to "N"
    IF ((((((op) = "-") or ((op) = "*")) or ((op) = "/")) or ((op) = "//")) or ((op) = "%")) and (not (((op) = "*") and (((lefttype) = "S") or (not (evtype = "N"))))) THEN:
      IF (not ((lefttype) = "N")) or (not (evtype = "N")) THEN:
        fail "TypeError: numeric operands"
      ELSE:
        IF (op) = "-" THEN:
          set ev to ((leftval) - ev)
        IF (op) = "*" THEN:
          set ev to ((leftval) * ev)
        IF (((op) = "/") or ((op) = "//")) or ((op) = "%") THEN:
          IF ev = 0 THEN:
            fail "ZeroDivisionError"
          ELSE:
            IF (op) = "/" THEN:
              set ev to ((leftval) / ev)
            IF (op) = "//" THEN:
              set ev to (floor of ((leftval) / ev))
            IF (op) = "%" THEN:
              set ev to ((leftval) mod ev)
    IF ((((((op) = "==") or ((op) = "!=")) or ((op) = "<")) or ((op) = ">")) or ((op) = "<=")) or ((op) = ">=") THEN:
      set cmp to 0
      IF ((lefttype) = "S") and (evtype = "S") THEN:
        compareText (leftval) ev
      ELSE:
        IF (leftval) < ev THEN:
          set cmp to -1
        IF (leftval) > ev THEN:
          set cmp to 1
      IF not ((lefttype) = evtype) THEN:
        IF ((op) = "==") or ((op) = "!=") THEN:
          set cmp to 1
        ELSE:
          fail "TypeError: incompatible comparison"
      set ev to 0
      IF (((((((op) = "==") and (cmp = 0)) or (((op) = "!=") and (not (cmp = 0)))) or (((op) = "<") and (cmp < 0))) or (((op) = ">") and (cmp > 0))) or (((op) = "<=") and (not (cmp > 0)))) or (((op) = ">=") and (not (cmp < 0))) THEN:
        set ev to 1
      set evtype to "N"
  IF ((data_variable) = "") and (((op) = "in") or ((op) = "notin")) THEN:
    IF ((lefttype) = "S") and (evtype = "S") THEN:
      set tmp to ev
      set found to 0
      set si to 1
      REPEAT UNTIL (si > (((length of tmp) - (length of (leftval))) + 1)) or (found = 1):
        substr tmp si (si + ((length of (leftval)) - 1))
        compareText ev (leftval)
        IF cmp = 0 THEN:
          set found to 1
        change si by 1
      set ev to found
      set evtype to "N"
      IF (op) = "notin" THEN:
        set ev to (1 - ev)
    ELSE:
      fail "TypeError: in requires strings"

DEFINE FAST unary (op) (lo) (hi):
  eval (lo) (hi)
  IF (op) = "not" THEN:
    truth ev evtype
    IF truthv = 0 THEN:
      set ev to 1
    ELSE:
      set ev to 0
    set evtype to "N"
  ELSE:
    IF not (evtype = "N") THEN:
      fail "TypeError: numeric operand"
    IF (op) = "-" THEN:
      set ev to (0 - ev)

DEFINE FAST callfn (fn) (lo) (hi):
  eval (lo) (hi)
  IF error = "" THEN:
    set pending to 0
    IF (fn) = "str" THEN:
      IF evtype = "L" THEN:
        listfmt ev
        set pending to 1
      ELSE:
        set ev to ("" join ev)
        set evtype to "S"
        set pending to 1
    IF (fn) = "sum" THEN:
      IF not (evtype = "L") THEN:
        fail "TypeError: sum() requires a list"
      set ltmp to ev
      listlen ltmp
      set ln to ev
      set lsum to 0
      set lpi to 1
      REPEAT UNTIL lpi > ln:
        listitem ltmp lpi
        IF not (evtype = "N") THEN:
          fail "TypeError: sum() requires a list of numbers"
        change lsum by ev
        change lpi by 1
      set ev to lsum
      set evtype to "N"
      set pending to 1
    IF (fn) = "len" THEN:
      IF evtype = "L" THEN:
        listlen ev
        set pending to 1
      ELSE:
        IF not (evtype = "S") THEN:
          fail "TypeError: len requires string"
        set ev to (length of ev)
        set evtype to "N"
        set pending to 1
    IF ((fn) = "int") or ((fn) = "float") THEN:
      numbercheck ev
      IF en = 0 THEN:
        fail "ValueError: invalid number"
      ELSE:
        set ev to (ev + 0)
        IF (fn) = "int" THEN:
          IF ev < 0 THEN:
            set ev to (ceiling of ev)
          ELSE:
            set ev to (floor of ev)
      set evtype to "N"
      set pending to 1
    IF ((fn) = "abs") or ((fn) = "round") THEN:
      IF not (evtype = "N") THEN:
        fail "TypeError: numeric argument"
