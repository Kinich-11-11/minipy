DEFINE FAST text (s) (tx) (ty) (limit) (offset):
  set rj to ((offset) + 1)
  set rx to (tx)
  REPEAT (limit):
    IF not (rj > (length of (s))) THEN:
      set rc to (letter rj of (s))
      switch costume to 
      switch costume to ()
      go to x: rx y: (ty)
      stamp
    change rx by 5.1
    change rj by 1
