DEFINE FAST reveal:
  IF selected < page THEN:
    set page to selected
  IF selected > (page + 15) THEN:
    set page to (selected - 15)
  set dirty to 1
  set saved to 0
  keepCursor

DEFINE FAST backup:
  delete all of Snapshot
  set ui to 1
  REPEAT (length of Code):
    add (item ui of Code) to Snapshot
    change ui by 1

DEFINE FAST undo:
  IF (length of Snapshot) > 0 THEN:
    delete all of Code
    set ui to 1
    REPEAT (length of Snapshot):
      add (item ui of Snapshot) to Code
      change ui by 1
    set selected to 1
    set page to 1
    set status to "Restored last edit"

DEFINE editLine:
  set busy to 1
  ask ("Line " join (selected join (": " join (item selected of Code)))) and wait
  IF not ((answer) = ":cancel") THEN:
    backup
    replace item selected of Code with (answer)
    set status to "Edited"
  set busy to 0
  set dirty to 1

DEFINE importCode:
  set busy to 1
  ask "Paste code: use | between lines. :cancel keeps current code" and wait
  IF not ((answer) = ":cancel") THEN:
    loadCode (answer)
  set busy to 0
  set dirty to 1

DEFINE FAST loadCode (source):
  backup
  delete all of Code
  set ui to 1
  set ut to ""
  REPEAT (length of (source)):
    set uc to (letter ui of (source))
    IF uc = "|" THEN:
      add ut to Code
      set ut to ""
    ELSE:
      IF uc = "\" THEN:
        change ui by 1
        set ut to (ut join (letter ui of (source)))
      ELSE:
        set ut to (ut join uc)
    change ui by 1
  add ut to Code
  set selected to 1
  set page to 1
  set hscroll to 0
  set status to "Imported"

DEFINE FAST keepCursor:
  IF (data_variable) < 0 THEN:
    set cursor to 0
  IF (data_variable) > (length of (item (data_variable) of Code)) THEN:
    set cursor to (length of (item (data_variable) of Code))
  IF (data_variable) < (data_variable) THEN:
    set hscroll to (data_variable)
  IF (data_variable) > ((data_variable) + 49) THEN:
    set hscroll to ((data_variable) - 49)
  IF (data_variable) < 0 THEN:
    set hscroll to 0
  set dirty to 1

DEFINE FAST prepEdit:
  IF (data_variable) = 0 THEN:
    backup
    set saved to 1

DEFINE FAST insert (ch):
  IF ((data_variable) = 0) and ((data_variable) = 0) THEN:
    prepEdit
    set ut to (item (data_variable) of Code)
    set ub to ""
    set ui to 1
    REPEAT (data_variable):
      set ub to ((data_variable) join (letter (data_variable) of (data_variable)))
      change ui by 1
    set ub to ((data_variable) join (ch))
    set ui to ((data_variable) + 1)
    REPEAT ((length of (data_variable)) - (data_variable)):
      set ub to ((data_variable) join (letter (data_variable) of (data_variable)))
      change ui by 1
    replace item (data_variable) of Code with (data_variable)
    change cursor by (length of (ch))
    set status to "Typing"
    keepCursor

DEFINE FAST newline:
  IF ((data_variable) = 0) and ((data_variable) = 0) THEN:
    set ut to (item (data_variable) of Code)
    set ub to ""
    set uc to ""
    set ui to 1
    REPEAT (data_variable):
      set ub to ((data_variable) join (letter (data_variable) of (data_variable)))
      change ui by 1
    REPEAT ((length of (data_variable)) - (data_variable)):
      set uc to ((data_variable) join (letter (data_variable) of (data_variable)))
      change ui by 1
    prepEdit
    replace item (data_variable) of Code with (data_variable)
    insert (data_variable) at ((data_variable) + 1) of Code
    change selected by 1
    set cursor to 0
    set saved to 0
    set status to "New line"
    reveal

DEFINE FAST backspace:
  IF ((data_variable) = 0) and ((data_variable) = 0) THEN:
    IF (data_variable) > 0 THEN:
      prepEdit
      set ut to (item (data_variable) of Code)
      set ub to ""
      set ui to 1
      REPEAT ((data_variable) - 1):
        set ub to ((data_variable) join (letter (data_variable) of (data_variable)))
        change ui by 1
      change ui by 1
      REPEAT ((length of (data_variable)) - (data_variable)):
        set ub to ((data_variable) join (letter (data_variable) of (data_variable)))
        change ui by 1
      replace item (data_variable) of Code with (data_variable)
      change cursor by -1
      set status to "Backspace"
      keepCursor
    ELSE:
      IF (data_variable) > 1 THEN:
        prepEdit
        set ut to (item ((data_variable) - 1) of Code)
        set cursor to (length of (data_variable))
        replace item ((data_variable) - 1) of Code with ((data_variable) join (item (data_variable) of Code))
        delete (data_variable) of Code
        change selected by -1
        set saved to 0
        set status to "Joined lines"
        reveal

DEFINE FAST moveCursor (dir):
  IF ((data_variable) = 0) and ((data_variable) = 0) THEN:
    IF (dir) < 0 THEN:
      IF (data_variable) > 0 THEN:
        change cursor by -1
      ELSE:
        IF (data_variable) > 1 THEN:
          change selected by -1
          set cursor to (length of (item (data_variable) of Code))
          reveal
    ELSE:
      IF (data_variable) < (length of (item (data_variable) of Code)) THEN:
        change cursor by 1
      ELSE:
        IF (data_variable) < (length of Code) THEN:
          change selected by 1
          set cursor to 0
          reveal
    set saved to 0
    keepCursor

DEFINE FAST exportCode:
  delete all of Export
  set ut to ""
  set ui to 1
  REPEAT (length of Code):
    IF ui > 1 THEN:
      set ut to (ut join "|")
    set ut to (ut join (item ui of Code))
    change ui by 1
  add ut to Export
  show list Export
  set exportShown to "1"
  set status to "Export: whole program on 1 line ( | = newline )"
  set dirty to "1"

DEFINE FAST scrollone (zone) (dir):
  IF (zone) = 2 THEN:
    change outpage by (dir)
    IF outpage > ((length of Output) - 8) THEN:
      set outpage to ((length of Output) - 8)
    IF outpage < 1 THEN:
      set outpage to 1
    set dirty to "1"
  IF (zone) = 1 THEN:
    change varpage by (dir)
    IF varpage > ((length of Names) - 3) THEN:
      set varpage to ((length of Names) - 3)
    IF varpage < 1 THEN:
      set varpage to 1
    set dirty to "1"
  IF (zone) = 3 THEN:
    change page by (dir)
    IF page > ((length of Code) - 15) THEN:
      set page to ((length of Code) - 15)
    IF page < 1 THEN:
      set page to 1
    set dirty to "1"
